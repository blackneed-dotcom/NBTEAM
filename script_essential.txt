FUNC_CHATLOG	{

	set @sd, mysd();

	set @month$, gettime(4,0);
	set @day$, gettime(3,0);
	set @hour$, gettime(2,0);
	set @min$, gettime(1,0);
	set @sec$, gettime(0,0,);
	set @clan$, get_clanname(@sd);

	if(gettime(5,0) < 10) set @year$, "0"+gettime(5,0);
	if(gettime(4,0) < 10) set @month$, "0"+gettime(4,0);
	if(gettime(3,0) < 10) set @day$, "0"+gettime(3,0);
	if(gettime(2,0) < 10) set @hour$, "0"+gettime(2,0);
	if(gettime(1,0) < 10) set @min$, "0"+gettime(1,0);
	if(gettime(0,0) < 10) set @sec$, "0"+gettime(0,0);

	if(mapid() < 10) set @mapid$, "00000"+mapid();
	else if(mapid() < 100) set @mapid$, "0000"+mapid();
	else if(mapid() < 1000) set @mapid$, "000"+mapid();
	else if(mapid() < 10000) set @mapid$, "00"+mapid();
	else if(mapid() < 100000) set @mapid$, "0"+mapid();

	set @x$, get_x(@id);
	set @y$, get_y(@id);

	if(get_x(@id) < 10) set @x$, "00"+get_x(@id);
	else if(get_x(@id) < 100) set @x$, "0"+get_x(@id);

	if(get_y(@id) < 10) set @y$, "00"+get_y(@id);
	else if(get_y(@id) < 100) set @y$, "0"+get_y(@id);
	
	set @mapinfo$, @mapid$+"/"+@x$+"/"+@y$+"";

	write_log "log/chat/"+@year$+"년_"+@month$+"월_"+@day$+"일_"+@hour$+"시.txt", "["+@hour$+"시 "+@min$+"분 "+@sec$+"초 - "+@mapinfo$+"] □ "+get_name(@sd)+": "+get_say(@sd)+" \\n";

	return;
}

ONTRADE	{
	set @sd, mysd();
	set @id, myid();
	set @year$, gettime(5,0);
	set @month$, gettime(4,0);
	set @day$, gettime(3,0);
	set @hour$, gettime(2,0);
	set @min$, gettime(1,0);
	set @sec$, gettime(0,0,);
	
	set #SERVER_YEAR, gettime(5,0);
	set #SERVER_MONTH, gettime(4,0);
	set #SERVER_DAY, gettime(3,0);
	set #SERVER_HOUR, gettime(2,0);
	set #SERVER_MINUTE, gettime(1,0);
	set #SERVER_SECOND, gettime(0,0);

	if(gettime(5,0) < 10) set @year$, "0"+gettime(5,0);
	if(gettime(4,0) < 10) set @month$, "0"+gettime(4,0);
	if(gettime(3,0) < 10) set @day$, "0"+gettime(3,0);
	if(gettime(2,0) < 10) set @hour$, "0"+gettime(2,0);
	if(gettime(1,0) < 10) set @min$, "0"+gettime(1,0);
	if(gettime(0,0) < 10) set @sec$, "0"+gettime(0,0);

	if(mapid() < 10) set @mapid$, "00000"+mapid();
	else if(mapid() < 100) set @mapid$, "0000"+mapid();
	else if(mapid() < 1000) set @mapid$, "000"+mapid();
	else if(mapid() < 10000) set @mapid$, "00"+mapid();
	else if(mapid() < 100000) set @mapid$, "0"+mapid();

	set @x$, get_x(@id);
	set @y$, get_y(@id);

	if(get_x(@id) < 10) set @x$, "00"+get_x(@id);
	else if(get_x(@id) < 100) set @x$, "0"+get_x(@id);

	if(get_y(@id) < 10) set @y$, "00"+get_y(@id);
	else if(get_y(@id) < 100) set @y$, "0"+get_y(@id);
	
	set @mapinfo$, @mapid$+"/"+@x$+"/"+@y$+"";


	write_log "log/trade/"+@year$+"년_"+@month$+"월_"+@day$+"일.txt", "["+@hour$+"시 "+@min$+"분 "+@sec$+"초 - "+@mapinfo$+"] [TRADE ] "+get_name(@sd)+"보유금액 "+get_money(@sd)+"원 ( "+get_itemlist(@sd)+" *황금개수: "+countitem(name2itemid("황금"))+"개 )\\n";

			set .@result$, .@result$ + " " + itemid2name(get_mixsuccess_itemid(.@mix_id, .@i)) + "(" + get_mixsuccess_amount(.@mix_id, .@i) + "개)";

		if(strpos(.@result$,"황금") > 0 || strpos(.@result$,"천억경") > 0 || strpos(.@result$,"만억경") > 0) 
			{ }
		else
			broadcast 5, "[안내] " + get_name(.@sd) + .@result$ + " 성공!";
	}
	else
	{
		set .@log$, "실패";
		set .@resultmsg$, "[실패] ";

		for(set .@i, 0; .@i < get_mixfailure_count(.@mix_id); set .@i, .@i+1)
		{
			set .@result$, .@result$ + " " + itemid2name(get_mixsuccess_itemid(.@mix_id, .@i)) + "(" + get_mixfailure_amount(.@mix_id, .@i) + "개)";
		}
		if(strpos(.@result$,"황금") > 0 || strpos(.@result$,"천억경") > 0 || strpos(.@result$,"만억경") > 0)
			{ }
		else
			broadcast 5, "[안내] " + get_name(.@sd) + .@result$ + " 실패!";
	}

	callfunc "ITEMMIXLOG", .@log$, .@result$;
}
