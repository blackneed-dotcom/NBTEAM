FUNC_ONSEMICOLON	{
	set @sd, mysd();
	set @id, myid();

	if(get_side(@id) == 0)
	{
		set @target_id, obj_getfront();
		set @mapobject, get_mapobject(get_x(@id), get_y(@id)-1);
		
		if(@mapobject == 1619 || @mapobject == 1620)
		{
			set @target_x, get_x(@id);
			set @target_y, get_y(@id)-1;
			
			if(@mapobject == 1620)
			{
				set @target_x, get_x(@id)-1;
				set @target_y, get_y(@id)-1;
			}
			
			paperpopup @sd, 18, 10, "<"+get_mapname(@sd)+" 게시판... 끄려면 ESC를 누르세요...>\\n\\n"+get_mapboarddata(@target_x, @target_y);
		}
		

	}
	else if(get_side(@id) == 1)
	{
		set @target_id, obj_getfront();
	}
	else if(get_side(@id) == 2)
	{
		set @target_id, obj_getfront();
	}
	else if(get_side(@id) == 3)
	{
		set @target_id, obj_getfront();
	}
	
	if(@target_id != 0)
	{
		if(istype(@target_id) == 1)
		{
			set @target_sd, id2sd(@target_id);
			set @tg_job, get_job(@target_sd);

			switch(@tg_job)
			{
				case 0:
					set @job$, "평민";
					break;

				case 1:
					if(get_joblevel(@target_sd) == 0) { set @job$, "전사"; }
					if(get_joblevel(@target_sd) == 1) { set @job$, "검객"; }
					if(get_joblevel(@target_sd) == 2) { set @job$, "검제"; }
					if(get_joblevel(@target_sd) == 3) { set @job$, "검황"; }
					if(get_joblevel(@target_sd) == 4) { set @job$, "검성"; }
					break;
				case 2:
					if(get_joblevel(@target_sd) == 0) { set @job$, "도적"; }
					if(get_joblevel(@target_sd) == 1) { set @job$, "자객"; }
					if(get_joblevel(@target_sd) == 2) { set @job$, "진검"; }
					if(get_joblevel(@target_sd) == 3) { set @job$, "귀검"; }
					if(get_joblevel(@target_sd) == 4) { set @job$, "태성"; }
					break;
				case 3:
					if(get_joblevel(@target_sd) == 0) { set @job$, "주술사"; }
					if(get_joblevel(@target_sd) == 1) { set @job$, "술사"; }
					if(get_joblevel(@target_sd) == 2) { set @job$, "현사"; }
					if(get_joblevel(@target_sd) == 3) { set @job$, "현인"; }
					if(get_joblevel(@target_sd) == 4) { set @job$, "현자"; }
					break;
				case 4:
					if(get_joblevel(@target_sd) == 0) { set @job$, "도사"; }
					if(get_joblevel(@target_sd) == 1) { set @job$, "도인"; }
					if(get_joblevel(@target_sd) == 2) { set @job$, "명인"; }
					if(get_joblevel(@target_sd) == 3) { set @job$, "진인"; }
					if(get_joblevel(@target_sd) == 4) { set @job$, "진선"; }
					break;
			}
			
			set_sdsegment(@target_sd);
			set @msg$, $업적칭호$;
			set_sdsegment(@sd);

			message @sd, 3, ""+get_name(@target_sd)+"("+@job$+"): "+@msg$+"";
		}
		/*
		else if(istype(@target_id) == 2)
		{
			message @sd, 3, get_mobname(@target_id);
		}
		else if(istype(@target_id) == 3)
		{
			message @sd, 3, get_npcname(@target_id);
		}
		*/
	}

return;
}

FUNC_ONATTACH	{
	set @sd, mysd();
	set @id, myid();

	if(get_side(@id) == 0)
	{
		set @mapobject, get_mapobject(get_x(@id), get_y(@id)-1);
		if(@mapobject == 1619 || @mapobject == 1620)
		{
		
			//message @sd, 0, get_mapboarddata(get_x(@id), get_y(@id)-1);
		
			set @attach_slot, get_lastslot(@id);
			set @item_id, get_item(@sd, @attach_slot);

			if(@item_id > 0)
			{
				set @item_name$, itemid2name(@item_id);
				if(@item_name$ == "노란비서") { set @check, 1; }
				else if(@item_name$ == "소환비서") { set @check, 1; }
				else if(@item_name$ == "장안성비서") { set @check, 1; }
				else if(@item_name$ == "대방성비서") { set @check, 1; }
				else if(@item_name$ == "부여성비서") { set @check, 1; }
				else if(@item_name$ == "국내성비서") { set @check, 1; }
				else if(@item_name$ == "귀환비서") { set @check, 1; }
				else if(@item_name$ == "도삭산100층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산200층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산300층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산400층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산500층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산600층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산700층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산800층비서") { set @check, 1; }
				else if(@item_name$ == "도삭산900층비서") { set @check, 1; }


				if(@check == 1)
				{
					if(countitem(name2itemid(@item_name$)) > 0)
					{
						delitem @sd, name2itemid(@item_name$), 1;
						message @sd, 3, ""+@item_name$+" 붙였습니다.";

						set @pos, input2(@msg$, 0, 12, "붙일 메시지는?", "", "");
						
						if(@pos == 0) return;
						else
						{
							set @target_x, get_x(@id);
							set @target_y, get_y(@id)-1;
							if(@mapobject == 1620)
							{
								set @target_x, get_x(@id)-1;
								set @target_y, get_y(@id)-1;
							}
							
							set @list_1$, get_name(@sd) + ": " +@msg$ + " ("+gettime(2, 0)+":"+gettime(1, 0)+":"+gettime(0, 0)+")\\n";
							set @list_2$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 0) + "\\n";
							set @list_3$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 1) + "\\n";
							set @list_4$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 2) + "\\n";
							set @list_5$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 3) + "\\n";
							set @list_6$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 4) + "\\n";
							set @list_7$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 5) + "\\n";
							set @list_8$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 6) + "\\n";
							set @list_9$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 7) + "\\n";
							set @list_10$, strtok(get_mapboarddata(@target_x, @target_y), "\\n", 8) + "\\n";
							
							set @board_data$, @list_1$ + @list_2$ + @list_3$ + @list_4$ + @list_5$ + @list_6$ + @list_7$ + @list_8$ + @list_9$ + @list_10$;
							
							set_mapboarddata @target_x, @target_y, @board_data$;
							paperpopup @sd, 18, 10, "<"+get_mapname(@sd)+" 게시판... 끄려면 ESC를 누르세요...>\\n\\n"+get_mapboarddata(@target_x, @target_y);
						}
					}
				}
				else
				{
					message @sd, 3, "붙일 수 없는 물건입니다.";
				}
			}
		}
		else
		{
			message @sd, 3, "그런 곳에는 붙일 수 없습니다.";
		}
	}


return;
}

ONLEVELUP	{
	callfunc "ONLEVELUP";
}

FUNC_ONLEVELUP	{
	set @sd, mysd();
	set @id, myid();

	if(get_job(@sd) == 0)
	{
				
		set_purewill @sd, get_will(@sd);
		set_puremight @sd, get_might(@sd);
		set_puregrace @sd, get_grace(@sd);

		set_purehp @sd, get_purehp(@sd) + ( 45 + rand(1, 11) );
		set_puremp @sd, get_puremp(@sd) + ( 30 + rand(1, 11) );

		if(get_level(@sd)%2 > 0)
		{
			set_purewill @sd, get_purewill(@sd) + 1;
			set_puregrace @sd, get_puregrace(@sd) + 1;
		}
		else
		{
			set_puremight @sd, get_puremight(@sd) + 1;
		}
	}
	else if(get_job(@sd) == 1)
	{
		set @might_val, get_might(@sd) - get_puremight(@sd);
		set @will_val, get_will(@sd) - get_purewill(@sd);
		set @grace_val, get_grace(@sd) - get_puregrace(@sd);

		set_purehp @sd, get_purehp(@sd) + ( 70 + rand(1, 11) );
		set_puremp @sd, get_puremp(@sd) + ( 10 + rand(1, 11) );

		set_puremight @sd, get_puremight(@sd) + 1;

		if(get_level(@sd)%2 > 0)
		{
			set_purewill @sd, get_purewill(@sd) + 1;
		}
		if(get_level(@sd)%3 > 0)
		{
			set_puregrace @sd, get_puregrace(@sd) + 1;
		}
	}
	else if(get_job(@sd) == 2)
	{
		set @might_val, get_might(@sd) - get_puremight(@sd);
		set @will_val, get_will(@sd) - get_purewill(@sd);
		set @grace_val, get_grace(@sd) - get_puregrace(@sd);

		set_purehp @sd, get_purehp(@sd) + ( 55 + rand(1, 11) );
		set_puremp @sd, get_puremp(@sd) + ( 20 + rand(1, 11) );

		set_purewill @sd, get_purewill(@sd) + 1;

		if(get_level(@sd)%2 > 0)
		{
		set_puremight @sd, get_puremight(@sd) + 1;
		}
		if(get_level(@sd)%3 > 0)
		{
			set_puregrace @sd, get_puregrace(@sd) + 1;
		}
	}
	else if(get_job(@sd) == 3)
	{
		set @might_val, get_might(@sd) - get_puremight(@sd);
		set @will_val, get_will(@sd) - get_purewill(@sd);
		set @grace_val, get_grace(@sd) - get_puregrace(@sd);

		set_purehp @sd, get_purehp(@sd) + ( 35 + rand(1, 11) );
		set_puremp @sd, get_puremp(@sd) + ( 35 + rand(1, 11) );

		set_puregrace @sd, get_puregrace(@sd) + 1;

		if(get_level(@sd)%2 > 0)
		{
			set_puremight @sd, get_puremight(@sd) + 1;
		}
		if(get_level(@sd)%3 > 0)
		{
			set_purewill @sd, get_purewill(@sd) + 1;
		}
	}
	else if(get_job(@sd) == 4)
	{
		set @might_val, get_might(@sd) - get_puremight(@sd);
		set @will_val, get_will(@sd) - get_purewill(@sd);
		set @grace_val, get_grace(@sd) - get_puregrace(@sd);

		set_purehp @sd, get_purehp(@sd) + ( 45 + rand(1, 11) );
		set_puremp @sd, get_puremp(@sd) + ( 30 + rand(1, 11) );

		set_puregrace @sd, get_puregrace(@sd) + 1;

		if(get_level(@sd)%3 > 0)
		{
			set_purewill @sd, get_purewill(@sd) + 1;
			set_puremight @sd, get_puremight(@sd) + 1;
		}
	}

	set_might @sd, get_puremight(@sd) + @might_val;
	set_grace @sd, get_puregrace(@sd) + @grace_val;
	set_will @sd, get_purewill(@sd) + @will_val;

	set_hp @sd, get_maxhp(@sd);
	set_mp @sd, get_maxmp(@sd);

	effect @id, 2;

	message @sd, 3, "레벨업!!";

	if(get_level(@sd) == 99)
	{
		set @year$, gettime(5,0);
		set @month$, gettime(4,0);
		set @day$, gettime(3,0);
		set @hour$, gettime(2,0);
		set @min$, gettime(1,0);
		set @sec$, gettime(0,0);

		if(gettime(5,0) < 10) set @year$, "0"+gettime(5,0);
		if(gettime(4,0) < 10) set @month$, "0"+gettime(4,0);
		if(gettime(3,0) < 10) set @day$, "0"+gettime(3,0);
		if(gettime(2,0) < 10) set @hour$, "0"+gettime(2,0);
		if(gettime(1,0) < 10) set @min$, "0"+gettime(1,0);
		if(gettime(0,0) < 10) set @sec$, "0"+gettime(0,0);

		if(get_job(@sd) == 1) { set @job$, "전사"; }
		if(get_job(@sd) == 2) { set @job$, "도적"; }
		if(get_job(@sd) == 3) { set @job$, "주술사"; }
		if(get_job(@sd) == 4) { set @job$, "도사"; }

		if(get_joblevel(@sd) == 0)
		{
			announce 5, "<< ["+get_name(@sd)+"("+@job$+")] 님께서 99레벨을 달성하셨습니다. 축하합니다! >>";

	}
	else

FUNC_ONDROP	{
	set @sd, mysd();
	set @id, myid();
	
	// 돈 버그 해결
	if(get_lastdrop_itemid(@sd) > 60000);
	{
		set $money_count, $money_count + 1;
		if($money_count > 10)
		{
			set $money_count, 0;
			for(set @loop, 0; @loop <= 20; set @loop, @loop + 1)
			{
				getitemmap @sd, get_x(@id), get_y(@id);
			}
			getoutofhere @sd;
		}
	}

	set @logtime$, gettime(4,0) + "" + gettime(3,0) + "" + gettime(2,0) + "" + gettime(1,0) + "" + gettime(0,0) + "";
	write_log "db_customlog/drop/"+get_name(@sd)+""+gettime(4,0)+".txt", get_nowtime()+ "	"+@logtime$+"\\n";
	
	set @slot, get_dropitem_slot(@sd);
	set @all, get_dropitem_all(@sd);
	set @item_num1, get_item(@sd, @slot);
	set @count1, countitem(@item_num1);





	set @list, list1("정말 버리시겠습니까?", "예", "아뇨");
	if(@list == 0) end;
	else if(@list == 1)
	{
		drop_item @sd, @slot, @all, 0;	
		action @id, 4, 50, 0;
		sound @id, 406;	
		set @item_num2, get_item(@sd, @slot);
		set @count2, countitem(@item_num2);

		callfunc "ITEMLOG", "D", itemid2name(@item_num1), ""+@count1+"→"+@count2+"";

		if($dropbug_warning > 20)
		{
			getitemmap @sd, get_x(@id), get_y(@id);
			callfunc "ITEMLOG", "[GET  D]", get_lastget_itemid(@sd);
			message @sd, 0, "[알림] 아이템 버리기 버그 사용이 감지되었습니다."; 
			message @sd, 0, "[알림] 서버의 안정성을 위하여 관련 기능이 제한 됩니다."; 
			message @sd, 0, "[알림] 운영자(향수GM)에게 문의하시기 바랍니다."; 
			set_flag @sd, "drop", 0;
		}
		
		set $droped, $droped + 1;
		if($droped == 0)
		{
			set $droped_time1, get_nowtime();
		}
		else if($droped > 0)
		{
			if($droped > 10)
			{
				set @compare, get_nowtime() - $droped_time1;
				if(@compare < 5)
				{
					set $dropbug_warning, $dropbug_warning + 1;
					
					for(set @i, 0; @i <= 10; set @i, @i + 1)
					{
						getitemmap @sd, get_x(@id), get_y(@id);	
					}
					set $droped, 0;
					getoutofhere @sd;
				}
				else
				{
					set $droped, 0;
				}
			}
			
		}
		


		//callfunc "ITEMLOG", "[DROP  ]", get_lastdrop_itemid(@sd);
	}

return;
}
