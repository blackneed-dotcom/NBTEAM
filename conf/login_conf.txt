// ------------------------------------------------------------------------------------
// 	1. 내 아이피 주소와 로그인 서버의 포트 설정하기.
// ------------------------------------------------------------------------------------
//
// 아래는 예시임.
//
// bind_ip: 0.0.0.0		<- 그냥 놔두세요 0.0.0.0으로
// login_ip: 192.168.25.61	<- 네이버에서 [내 아이피 주소]검색해서 뜨는 아이피 쓰세요.
// login_port: 2010		<- 포트 설정 할 줄 모르면 2010으로 두시고 할줄 알면 알아서 ㄱㄱ

//실제 설정은 ↓ 여기서

bind_ip: 0.0.0.0
login_ip: 127.0.0.1
login_port: 2010


// ------------------------------------------------------------------------------------
// 	2. 지금 실행하는 [로그인 서버(login.exe)]를 가장 먼저 실행한 [인터 서버(inter.exe)]와 연결해주기.
// ------------------------------------------------------------------------------------
//
// 아래는 예시임.
//
// inter_ip: 127.0.0.1		<- inter_conf.txt에 수정 안하셨으면 그대로 두시면 됨. (inter_conf.txt 내용과 동일)
// inter_port: 2500		<- 마찬가지로 따로 수정 안하셨으면 그대로. 
// inter_pwd: AngelServer	<- 놔두셈.

//실제 설정은 ↓ 여기서

inter_ip: 127.0.0.1
inter_port: 2500
inter_pwd: MyServer


// ------------------------------------------------------------------------------------
// 	3. 로그인 서버의 기능 설정. (서버선택, 아이디생성, 로그인 등)
// ------------------------------------------------------------------------------------
//
// 아래는 예시임.
//
//			* 숫자 1은 '허용함'을 의미
//
//			* 숫자 0은 '허용하지 않음'을 의미
//
// 한글자 아이디 허용여부
// allow_shortname: 0
//
// 영문아이디 허용여부
// allow_engname: 0
//
// 쉬운비번 허용여부 (숫자로만 비번 가능하고, 6글자  이하 가능)
// allow_easypwd: 1
//

//실제 설정은 ↓ 여기서

//한글자 아이디 허여부
allow_shortname: 0

//영문아이디 허용여부
allow_engname: 0

//쉬운비번 허용여부
allow_easypwd: 1

//당신의 서버 이름
server_name: 서버

//5.65 버전체크 (허용시 5.65 사용 가능)
allow_version: 1

//이용약관을 사용하고 싶다면, 아래의 주석을 해제하세요
agreement: db/agreement.txt


// ------------------------------------------------------------------------------------
// 	4. 서버 리스트의 설정.
// ------------------------------------------------------------------------------------
//
// max_gateway: 1 일 경우,
//
// gateway: 서버명	서버설명	ip주소	map서버포트
//
// 위와 같이 1개만 적어줘야 함.
//
// 
// 만약 max_gateway: 2 인데, 서버 갯수가 1개이면 login.exe가 그냥 꺼져버릴 수 있으니 조심.
//
//
//
//
//					네이버에서 님 아이피 주소 검색 ㄱㄱ
//max_gateway: 2				↓
//gateway: 불홍	2014년 04월 24일 시작	자기이아피	2010
//gateway: 괴유(TEST)	2013년 05월 12일 시작	자기이아피	2010
//								↑
//							login_conf.txt에 적힌 로그인 서버의 포트. (맨 윗부분 1번 참고)
//
//

//실제 설정은 ↓ 여기서

//서버리스트를 사용하고 싶다면, 아래의 주석을 해제하세요

max_gateway: 1
gateway: 마스터온라인	2017년 10월 28일 시작	127.0.0.1	2010
