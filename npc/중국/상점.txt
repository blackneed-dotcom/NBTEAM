장안성대장간	{
	set @mysd, mysd();
	set @myid, myid();
MAKE:
	set @pos, list2(@list, 0, 0, "금천지인패를 교환해드립니다.", "구곡검", "협가검", "영후단봉", "대모홍접선");
	if(@pos == 0) end;
	else if(@list == 1)
	{
		set @pos, list2(@list1, 0, 1, "금천지인패 1개로 교환하시겠습니까?.", "교환");
		if(@pos == 0) end;
		if(@pos == -1) goto MAKE;
		if(@list1 == 1) { set @needitem$, "금천지인패"; set @additem$, "구곡검"; }
		if(countitem(name2itemid(@needitem$)) < 1)
		{
			set @pos, dialog(0,0,0, @needitem$ + "1개(를) 구해오세요.");
			end;
		}
		else
		{
			additem @mysd, name2itemid(@additem$), 1;
			delitem @mysd, name2itemid(@needitem$), 1;
			set @pos, dialog(0,0,1, @additem$ + "만들어 드렸습니다.");
			broadcast 5, get_name(@sd) + "님이 구곡검 제작에 성공하셨습니다.";
			end;
		}
	}
	else if(@list == 2)
	{
		set @pos, list2(@list1, 0, 1, "금천지인패 1개로 교환하시겠습니까?.", "교환");

		if(@pos == -1) goto MAKE;
		if(@list1 == 1) { set @needitem$, "금천지인패"; set @additem$, "협가검"; }

			set @pos, dialog(0,0,0, @needitem$ + "1개(를) 구해오세요.");
			end;
		}
		else
		{

			broadcast 5, get_name(@sd) + "님이 협가검 제작에 성공하셨습니다.";
			end;
		}
	}
	else if(@list == 3)
	{
		set @pos, list2(@list1, 0, 1, "금천지인패 1개로 교환하시겠습니까?.", "교환");
		if(@pos == 0) end;
		if(@pos == -1) goto MAKE;
		if(@list1 == 1) { set @needitem$, "금천지인패"; set @additem$, "영후단봉"; }
		if(countitem(name2itemid(@needitem$)) < 1)
		{
			set @pos, dialog(0,0,0, @needitem$ + "1개(를) 구해오세요.");
			end;
		}
		else
		{
			additem @mysd, name2itemid(@additem$), 1;
			delitem @mysd, name2itemid(@needitem$), 1;
			set @pos, dialog(0,0,1, @additem$ + "만들어 드렸습니다.");
			broadcast 5, get_name(@sd) + "님이 영후단봉 제작에 성공하셨습니다.";
			end;
		}
	}
	else if(@list == 4)
	{
		set @pos, list2(@list1, 0, 1, "금천지인패 1개로 교환하시겠습니까?.", "교환");
