채광장입장	{
	set @sd, mysd();
	set @id, myid();

	if($produce_level_1 == 0)
	{
		set @pos, dialog(0,0,0, "채광술을 보유하고 있어야 입장이 가능합니다.");
		end;
	}
	else
	{
		set @pos, list2(@list, 0, 0, "안녕하세요. 어떻게 오셨나요?", "채광장 입장", "주의사항 듣기");
		if(@pos == 0) end;
		else if(@list == 1)
		{
			if(spellaethered(@sd, name2spellid("채광중")))
			{
				set @pos, dialog(0,0,0, "채광장에 입장하실 수 없습니다. 잠시후 다시와주세요.");
				end;
			}
			set @pos, dialog(0,0,1, "채광장으로 이동합니다.");
			if(@pos == 0) end;

			spellaether @sd, name2spellid("채광중"), 3600;
			warp @sd, 20001, rand(7,11), 18;
			end;
		}
		else if(@list == 2)
		{
		COS001:
			set @pos, dialog(0,0,1, "채광장에 입장하신 후 1시간동안 채광장에 계실수 있습니다.");
			if(@pos == 0) end;
		COS002:
			set @pos, dialog(0,1,1, "채광장에서 채광무기를 장착하지 않으시거나 채광맵을 이탈하실 경우 자동귀환 됩니다.");
			if(@pos == 0) end;
			else if(@pos == -1) goto COS001;
			else end;
			end;
		}
	}
}
