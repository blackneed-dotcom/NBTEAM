@주작영령공격	{
	set @id, objid();
	set @rand, rand(1, 10);
	set @rate, rand(1, 10);

	if($revival != $healhp) { set $healhp, $revival; set_mobhp @id, get_mobmaxhp(@id)/10*5; }

	if(rand(1, 10) <= 4)
	{
		callfunc "MONSTER_RANGE_ATTACK", objid(), 10, 6, -1, 107, 81, rand(2000000, 2500000), 1, "주작폭풍", 30, "-1";
		say @id, 2, get_mobname(@id) + ": 크르르르...";
	}
	if(rand(1, 10) <= 5)
	{
	callfunc "MONSTER_ATTACK", objid(), 10, 3, -1, 8, 8, rand(4000000, 5000000), 0, "헬파이어", -1;
	say @id, 2, get_mobname(@id) + ": 헬파이어..!";
	}
	if(rand(1, 10) <= 8)
	{
			set @target_id, get_mobtarget(@id);
			set @target_sd, id2sd(@target_id);
			if(@target_sd > 0)
			{
				if(!spellaethered(@target_sd, name2spellid("절망")))
				{
					if(get_equipitem_name(@target_sd, 6) == "시력회복의귀걸이" || get_equipitem_name(@target_sd, 7) == "시력회복의귀걸이" || spellaethered(@target_sd, name2spellid("정화의눈")) == 1) { end; }

					set_siya @target_sd, 1, 0;
					effect @target_id, 14;
					sound @target_id, 71;
					spellaether @target_sd, name2spellid("절망"), 25;
				}
			}
		}
	}
	if(rand(1, 10) <= 3)
	{
			set @type, istype(@target_id);
				
			for(set @i, 0; @i < 4; set @i, @i+1)
			{
				set @rand, rand(0, 3);
				switch(@rand)
				{
				case 0:
					set @rand_x, 0;
					set @rand_y, -1;
					set_side @id, 2;
					break;
				case 1:
					set @rand_x, 1;
					set @rand_y, 0;
					set_side @id, 3;
					break;
				case 2:
					set @rand_x, 0;
					set @rand_y, 1;
					set_side @id, 0;
					break;
				case 3:
					set @rand_x, -1;
					set @rand_y, 0;
					set_side @id, 1;
					break;
				}
				if(!canmove(@id, get_x(@target_id)+@rand_x, get_y(@target_id)+@rand_y) ) { goto LOOP; }
			}
LOOP:
			if(@i != 4) 
			{
				set_xy @id, get_x(@target_id)+@rand_x, get_y(@target_id)+@rand_y;
				updateobject @id;
				effect @id, 16;
				sound @id, 29;
			}
		}
		break;
	default:
		break;
	}
}

@죽음의손짓	{
	set @sd, mysd();
	set @id, myid();

	if($check_spell_type != 2)
	{
		for(set @i, 0; @i < 52; set @i, @i+1)
		{
			if(checkspell(name2spellid("죽음의손짓")) == @i) { delspell @i; }
		}
		callfunc "GMSendMail", "* 마법 배우기 악용자를 발견했습니다.", "죽음의손짓";
		end;
	}

	callfunc "SPELL_MOMENT";
	callfunc "MANA_DELAY", 7000, 0, @sd;
	//set_mp @sd, get_mp(@sd)-7000;

	set @target_id, script_target(@sd);
	if(@target_id > 0)
	{
		if(istype(@target_id) == 1)
		{
			set @target_sd, id2sd(@target_id);
			if(ismapoption(@target_sd, 6) && @target_sd != @sd && get_state(@target_sd) != 1)
			{
				if(spellaethered(@target_sd, name2spellid("죽음의손짓")) == 0)
				{
					effect @target_id, 146;
					sound @target_id, 8;

					set_ac @target_sd, get_ac(@target_sd)+5;

					spellaether @target_sd, name2spellid("죽음의손짓"), 6;
					message @target_sd, 3, get_name(@sd) + "님이 죽음의손짓을 시전하였습니다.";
				}
				else
				{
					message @sd, 3, "이미 걸려있습니다.";
					end;
				}
			}
			else
			{
				message @sd, 3, "걸리지 않습니다.";
				end;
			}
		}
		else if(istype(@target_id) == 2)
		{
			if(mobspellaethered(@target_id, name2spellid("몹죽음의손짓")) == 0)
			{
				effect @target_id, 146;
				sound @target_id, 8;

				set_mobac @target_id, get_mobac(@target_id)+5;

				mobspellaether @target_id, name2spellid("몹죽음의손짓"), 6;
			}
		}
		else
		{
			message @sd, 3, "걸리지 않습니다.";
		}

		action @id, 6, 30, 0;

		spelldelay @sd, name2spellid("죽음의손짓"), 40;
		message @sd, 3, "죽음의손짓을 시전하였습니다.";
	}
	else end;
	end;
}

		{

		}
