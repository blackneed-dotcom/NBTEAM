

@수호청룡공격	{
	set @id, objid();
	set @rand, rand(1, 10);
	set @rate, rand(1, 10);

	if(rand(1, 10) <= 4)
	{
		callfunc "MONSTER_RANGE_ATTACK", objid(), 10, 6, -1, 107, 185, rand(6000000, 6500000), 1, "청룡혼", 30, "-1";
		say @id, 2, get_mobname(@id) + ": 이곳이 어딘줄 아느냐..";
	}
	else
	{
		set @target_id, get_mobtarget(@id);
		set @target_sd, id2sd(@target_id);
		if(@target_sd > 0)
		{
			if(@rand <= 5 && spellaethered(@target_sd, name2spellid("혼마술")) == 0 && spellaethered(@target_sd, name2spellid("저주")) == 0)
			{
				effect @target_id, 39;
				sound @target_id, 79;
				set_ac @target_sd, get_ac(@target_sd)+65;
				spellaether @target_sd, name2spellid("혼마술"), 20;
				message @target_sd, 3, get_mobname(@id) + "이(가) 초혼마술을 가합니다";
				say @id, 1, get_mobname(@id) + ": 인간들이여..";
			}
		}
	}
	if(rand(1, 10) <= 7)
	{
		set @target_id, get_mobtarget(@id);
		set @target_sd, id2sd(@target_id);
		if(@target_sd > 0)
		{
			if(@rand <= 8 && spellaethered(@target_sd, name2spellid("혼마술")) == 0 && spellaethered(@target_sd, name2spellid("저주")) == 0)
			{
				effect @target_id, 13;
				sound @target_id, 69;
				set_ac @target_sd, get_ac(@target_sd)+50;
				spellaether @target_sd, name2spellid("저주"), 30;
				message @target_sd, 3, get_mobname(@id) + "이(가) 용의저주를 가합니다";;
				say @id, 2, get_mobname(@id) + ": 이전의 내가 아니거늘...";
			}
		}
	}
	if(rand(1, 10) <= 5)
	{
	callfunc "MONSTER_ATTACK", objid(), 10, 3, -1, 8, 118, rand(7000000, 8000000), 0, "백호의혼", -1;
	say @id, 2, get_mobname(@id) + ": 같잖구나...";
	}
	if(rand(1, 10) <= 8)
	{
			set @target_id, get_mobtarget(@id);
			set @target_sd, id2sd(@target_id);
			if(@target_sd > 0)
			{
				if(!spellaethered(@target_sd, name2spellid("절망")))
				{
					if(get_equipitem_name(@target_sd, 6) == "시력회복의귀걸이" || get_equipitem_name(@target_sd, 7) == "시력회복의귀걸이" || spellaethered(@target_sd, name2spellid("정화의눈")) == 1) { end; }

					set_siya @target_sd, 1, 0;
					effect @target_id, 14;
					sound @target_id, 71;
					spellaether @target_sd, name2spellid("절망"), 25;
				}
			}
		}
	}
	if(rand(1, 10) <= 3)
	{
	set @x1, get_x(@id)-13;
	set @x2, get_x(@id)+13;
	set @y1, get_y(@id)-13;
	set @y2, get_y(@id)+13;
		if (@x1 < 0) { set @x1, 0; }
		if (@x2 > mapwidth()-1) { set @x2, mapwidth()-1; }
		if (@y1 < 0) { set @y1, 0; }
		if (@y2 > mapheight()-1) { set @y2, mapheight()-1; }

		for (set @i, @x1; @i <= @x2; set @i, @i+1)
		{
			for (set @j, @y1; @j<=@y2; set @j, @j+1)
			{
				set @target_id, obj_getxy(@i, @j);
				if(@target_id > 0)
				{
					if(istype(@target_id) == 2)
					{
						if(get_mobhp(@target_id) <= 25000000)
						{
							effect @id, 63;
							effect @target_id, 63; //신령이펙트
							sound @id, 97; //신령효과음
							set_mobhp @target_id, get_mobhp(@target_id) + 200000000;
							showgage @target_id; //체력바 표시
						}
					}
				}
			}
		}
	}
}

}

LOOP:
