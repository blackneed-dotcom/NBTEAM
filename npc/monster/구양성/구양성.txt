@구양성입구입장	{
	callfunc "LViiii", 3, 12165, 15, 13, 18, 18, "구양성입구";

@구양성2단계입장	{
	set @sd, mysd();
	set @id, myid();

	if(#ninedungeon_a_level == 2)
	{
		if(party(@sd) != 0)
		{
			for(set @loop, 0; @loop < partymax(); set @loop, @loop+1)
			{
				set @target_sd, party_member(@sd, @loop);
				if(@target_sd > 0 && @target_sd != @sd)
				{
					if(get_mapid(@sd) == get_mapid(@target_sd))
					{
						set @x_plus, @x_plus+1;

						if(get_mapid(@sd) == name2mapid("구양성1(1)")) { warp @target_sd, name2mapid("구양성2(1)"), (28+@x_plus), 58; }
						else if(get_mapid(@sd) == name2mapid("구양성1(2)")) { warp @target_sd, name2mapid("구양성2(2)"), (28+@x_plus), 58; }
					}
				}
			}

			if(get_mapid(@sd) == name2mapid("구양성1(1)")) { warp @sd, name2mapid("구양성2(1)"), (28+(@x_plus+1)), 58; }
			else if(get_mapid(@sd) == name2mapid("구양성1(2)")) { warp @sd, name2mapid("구양성2(2)"), (28+(@x_plus+1)), 58; }
		}
		else
		{
			set @pos, dialog(0, 0, 0, "그룹이 없으면 다음 단계로 입장이 불가능합니다.");
			end;
		}
	}
}

@구양성클리어	{
	set @sd, mysd();
	set @id, myid();

	if(party(@sd) != 0)
	{
		if(get_joblevel(@sd) == 1) { set @exp, 3; }
		else if(get_joblevel(@sd) == 2) { set @exp, 5; }
		else if(get_joblevel(@sd) == 3) { set @exp, 20; }
		else if(get_joblevel(@sd) == 4) { set @exp, 50; }
		else if(get_joblevel(@sd) == 5) { set @exp, 500; }
		else if(get_joblevel(@sd) == 6) { set @exp, 700; }
		else { set @exp, 0; }

		if(@exp > 0)
		{
			set $saveexp_p, $saveexp_p+@exp;

			message @sd, 3, "구양성 클리어보상으로 십억경 " + @exp + "개 획득";
		}

		updatexy @sd;
		updatelookpc @sd, 0;
		updateallpc @sd;

		spellcast @sd, name2spellid("구양성딜레이");
	}
