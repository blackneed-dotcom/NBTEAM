해저경비병	{
	set @sd, mysd();
	set @id, myid();

	if($undersea_quest == 1) goto ing;
	else if($undersea_quest == 2) goto ee;
	else goto new;
	end;
new:
	set @pos, list2(@list,0, 0, "자네가 여기까지 어쩐 일이지?", "피폐된 동굴을 정화하는데 도움을 드리려고 왔습니다", "그냥..");
	if(@pos == 0) end;
	else if(@list == 1)
	{
		if(get_joblevel(@sd) < 4)
		{
			set @pos, dialog(0, 0, 1, "오호라.. 자네의 마음만은 깊이 사겠네.. 좀 더 강한 사람이 필요해.");
			end;
		}
		else
		{
			set $undersea_quest, 1;
			set @pos, dialog(1, 0, 1, "그럼 도움을 주게나, 이 밑에는 들쥐가 살고 있지.. 올래는 약했지만 기운을 받아 매우 강력해졌지.. 그래서 우리도 못잡고 쩔쩔 매고 있네. 자네가 저 들쥐들을 잡아서 기운을 11개 모아와주게나..");
			end;
		}
	}
	else end;
	end;
ing:
	if(countitem(name2itemid("해저의기운")) < 11)
	{
		set @pos, dialog(1, 0, 1, "그럼 도움을 주게나, 이 밑에는 들쥐가 살고 있지.. 올래는 약했지만 기운을 받아 매우 강력해졌지.. 그래서 우리도 못잡고 쩔쩔 매고 있네. 자네가 저 들쥐들을 잡아서 기운을 11개 모아와주게나..");
		end;
	}
	else
	{
		set $undersea_quest, 2;
		delitem @sd, name2itemid("해저의기운"), 11;
		set @pos, dialog(0, 0, 1, "고맙네.. 이 아래에는 들쥐보다 강력한 무시무시한 돌연변이들이 살고 있다는데.. 가보게나.");
		end;
	}
	end;
ee:
	set @pos, dialog(0, 0, 1, "이 아래에는 들쥐보다 강력한 무시무시한 돌연변이들이 살고 있다는데.. 가보게나.");
	end;
}

해저비석	{
	set @sd, mysd();
	set @id, myid();
	if($undersea_quest != 2)
	{
		set @pos, dialog(0, 0, 1, "퀘스트를 완료하지 않아 이용할 수 없습니다.");
		end;
	}

	set @pos, dialog(0, 0, 1, "개발중..");
	end;
}

해저낙랑	{
	set @sd, mysd();
	set @id, myid();

	if($undersea_quest != 2)
	{
		set @pos, dialog(0, 0, 1, "퀘스트를 완료하지 않아 입장할 수 없습니다.");
		end;
	}
	else
	{
		set @pos, list2(@list, 0, 0, "해저터널주둔지에 입장하시겠습니까? 입장하려면 각 직업원들이 1명씩 필요합니다. (최소 4인)", "준비가 되었습니다.", "아직 준비가 덜됬습니다.");
		if(@pos == 0) end;
		else if(@list == 1)
		{
			if(party(@sd) != 0)
			{
				if(party_member(@sd, 0) != get_name(@sd))
				{
					set @pos, dialog(0, 0, 1, "그룹장만 가능합니다. <여기서 그룹장은 가장 먼저 그룹맺은 회원이며 도중에 나갔으면 모두 그룹을 해제한 후에 다시 해야 합니다>");
					end;
				}
				for(set @loop, 0; @loop < partymax(); set @loop, @loop+1)
				{
					set @target_sd, party_member(@sd, @loop);
					if(@target_sd > 0)
					{
						if(get_joblevel(@target_sd) < 4)
						{
							set @pos, dialog(0, 0, 1, "4차 승급이 아닌 유저가 있습니다.");
							end;
						}

						if(get_uservar_i(@target_sd, "$undersea_quest") < 2)
						{
							set @pos, dialog(0, 0, 1, "퀘스트를 완료하지 않은 유저가 있습니다.\\n\\n아이디: " + get_name(@target_sd));
							end;
						}

						if(get_job(@target_sd) == 1) { set @junsa, @junsa + 1; }
						else if(get_job(@target_sd) == 2) { set @dojuk, @dojuk + 1; }
						else if(get_job(@target_sd) == 3) { set @jusul, @jusul + 1; }
						else if(get_job(@target_sd) == 4) { set @dosa, @dosa + 1; }
						set @man, @man + 1;
					}
				}

				if(@junsa > 0 && @dojuk > 0 && @jusul > 0 && @dosa > 0 && @man >= 4)
				{
					setarray @instance[1], #undersea_1in, #undersea_2in, #undersea_3in, #undersea_4in, #undersea_5in, #undersea_6in, #undersea_7in, #undersea_8in, #undersea_9in;
					for(set @loop, 1; @loop < 10; set @loop, @loop+1)
					{
						if(@instance[@loop] == 0)
						{
							if(@loop == 1) { set @map$, "해저터널입구주둔지(1)"; }
							else if(@loop == 2) { set @map$, "해저터널입구주둔지(2)"; }
							else if(@loop == 3) { set @map$, "해저터널입구주둔지(3)"; }
							else if(@loop == 4) { set @map$, "해저터널입구주둔지(4)"; }
							else if(@loop == 5) { set @map$, "해저터널입구주둔지(5)"; }
							else if(@loop == 6) { set @map$, "해저터널입구주둔지(6)"; }
							else if(@loop == 7) { set @map$, "해저터널입구주둔지(7)"; }
							else if(@loop == 8) { set @map$, "해저터널입구주둔지(8)"; }
							else if(@loop == 9) { set @map$, "해저터널입구주둔지(9)"; }
							message @sd, 3, "입장지를 정상적으로 읽었습니다.";
							message @sd, 3, name2mapid(@map$);
							message @sd, 3, @map$;
							goto next;
							end;
						}
					}
					next:
					if(@map$ == "해저터널입구주둔지(1)") { set #undersea_1in, 1; }
					else if(@map$ == "해저터널입구주둔지(2)") { set #undersea_2in, 1; }
					else if(@map$ == "해저터널입구주둔지(3)") { set #undersea_3in, 1; }
					else if(@map$ == "해저터널입구주둔지(4)") { set #undersea_4in, 1; }
					else if(@map$ == "해저터널입구주둔지(5)") { set #undersea_5in, 1; }
					else if(@map$ == "해저터널입구주둔지(6)") { set #undersea_6in, 1; }
					else if(@map$ == "해저터널입구주둔지(7)") { set #undersea_7in, 1; }
					else if(@map$ == "해저터널입구주둔지(8)") { set #undersea_8in, 1; }
					else if(@map$ == "해저터널입구주둔지(9)") { set #undersea_9in, 1; }
					deletearray @instance[0], getarraysize(@instance);

					for(set @loop, 0; @loop < partymax(); set @loop, @loop+1)
					{
						set @target_sd, party_member(@sd, @loop);
						if(@target_sd > 0)
						{
							warp @target_sd, name2mapid(@map$), 7,3;
						}
					}
					warp @sd, name2mapid(@map$), 7,3;

					if(@map$ == "해저터널입구주둔지(1)") { set #undersea_1ip, 30; }
					else if(@map$ == "해저터널입구주둔지(2)") { set #undersea_2ip, 30; }
					else if(@map$ == "해저터널입구주둔지(3)") { set #undersea_3ip, 30; }
					else if(@map$ == "해저터널입구주둔지(4)") { set #undersea_4ip, 30; }
					else if(@map$ == "해저터널입구주둔지(5)") { set #undersea_5ip, 30; }
					else if(@map$ == "해저터널입구주둔지(6)") { set #undersea_6ip, 30; }
					else if(@map$ == "해저터널입구주둔지(7)") { set #undersea_7ip, 30; }
					else if(@map$ == "해저터널입구주둔지(8)") { set #undersea_8ip, 30; }
					else if(@map$ == "해저터널입구주둔지(9)") { set #undersea_9ip, 30; }

					mobspawn name2mapid(@map$), 9, 5, 2, 1340;
					mobspawn name2mapid(@map$), 10, 5, 2, 1340;
					mobspawn name2mapid(@map$), 11, 5, 2, 1340;
					mobspawn name2mapid(@map$), 12, 5, 2, 1340;
					mobspawn name2mapid(@map$), 13, 5, 2, 1340;
					mobspawn name2mapid(@map$), 14, 5, 2, 1340;
					mobspawn name2mapid(@map$), 9, 8, 2, 1340;
					mobspawn name2mapid(@map$), 10, 8, 2, 1340;
					mobspawn name2mapid(@map$), 11, 8, 2, 1340;
					mobspawn name2mapid(@map$), 12, 8, 2, 1340;
					mobspawn name2mapid(@map$), 13, 8, 2, 1340;
					mobspawn name2mapid(@map$), 14, 8, 2, 1340;
					mobspawn name2mapid(@map$), 9, 11, 2, 1340;
					mobspawn name2mapid(@map$), 10, 11, 2, 1340;
					mobspawn name2mapid(@map$), 11, 11, 2, 1340;
					mobspawn name2mapid(@map$), 12, 11, 2, 1340;
					mobspawn name2mapid(@map$), 13, 11, 2, 1340;
					mobspawn name2mapid(@map$), 14, 11, 2, 1340;
					mobspawn name2mapid(@map$), 9, 14, 2, 1340;
					mobspawn name2mapid(@map$), 10, 14, 2, 1340;
					mobspawn name2mapid(@map$), 11, 14, 2, 1340;
					mobspawn name2mapid(@map$), 12, 14, 2, 1340;
					mobspawn name2mapid(@map$), 13, 14, 2, 1340;
					mobspawn name2mapid(@map$), 14, 14, 2, 1340;
					mobspawn name2mapid(@map$), 9, 17, 2, 1340;
					mobspawn name2mapid(@map$), 10, 17, 2, 1340;
					mobspawn name2mapid(@map$), 11, 17, 2, 1340;
					mobspawn name2mapid(@map$), 12, 17, 2, 1340;
					mobspawn name2mapid(@map$), 13, 17, 2, 1340;
					mobspawn name2mapid(@map$), 14, 17, 2, 1340;
					message @sd, 3, "몬스터 젠! (30마리를 모두 잡아야 함/사망 시 입구로 귀환)";
					message @sd, 3, "그룹원 : " + @man;
				}
				else
				{
					set @pos, dialog(0, 0, 1, "조건이 맞지 않습니다.");
					end;
				}
			}
			else
			{
				set @pos, dialog(0, 0, 1, "조건이 안맞다.");
				end;
			}
		}
		else end;
	}
}
