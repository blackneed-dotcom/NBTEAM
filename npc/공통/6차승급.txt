6차승급	{
	image 0, 60, 0;
	set @mysd, mysd();
	set @myid, myid();
	if(get_joblevel(@mysd) > 5){
		set @pos, dialog(0, 0, 0, "넌 이미 승급했지않느냐!!");
		end;
	}
	if(get_joblevel(@mysd) < 5){
		set @pos, dialog(0, 0, 0, "넌 5차가 아니로구만??????");
		end;
	}
	
POS00000:
	if(get_job(@mysd) == 1) { set @jobname$, "검사"; }
	else if(get_job(@mysd) == 2) { set @jobname$, "무영"; }
	else if(get_job(@mysd) == 3) { set @jobname$, "마제"; }
	else if(get_job(@mysd) == 4) { set @jobname$, "화타"; }

	set @pos, list2(@ret, 1, 0, "당신은 지금껏 수많은 수련을 거쳐 인간으로서 오르기\\n힘든 경지에 도달하셨고, 더욱이 세상을 종말로부터\\n보호하는 업적을 세우셨으니 " + @jobname$ + "으로 불리우시기에\\n부족함이 없습니다.\\n지금 바로 승급을 원하시나요?", "예", "아뇨");
	if(@pos == 0) { end; }
	else if(@ret == 1)
	{
		setarray @hp[0], 4000000, 3900000, 1800000, 1800000;// HP 승급기준
		setarray @mp[0], 400000, 600000, 2100000, 2200000;// MP 승급기준
		if(get_level(@mysd) >= 99 && get_purehp(@mysd) >= @hp[get_job(@mysd)-1] && get_puremp(@mysd) >= @mp[get_job(@mysd)-1]) // 승급기준인경우
		{
			if ( countitem(name2itemid("이무기의어금니")) == 0)
			{
				set @pos, dialog(0, 0, 0, "이무기의어금니가 없습니다.");
				end;
			}

			delitem @mysd, name2itemid("이무기의어금니"), 1;
			set_joblevel @mysd, get_joblevel(@mysd)+1;
			broadcast 4, get_name(@mysd) + "님께서 [" + get_jobname(@mysd) + "] 승급을 완료했습니다. 축하드립니다!";

			if(get_job(@mysd) == 1)
			{
				additem @mysd, name2itemid("사도의투구"),1;
			}
			else if(get_job(@mysd) == 2)
			{
				additem @mysd, name2itemid("사도의투구"),1;
			}
			else if(get_job(@mysd) == 3)
			{
				additem @mysd, name2itemid("사도의투구"),1;
			}
			else if(get_job(@mysd) == 4)
			{
				additem @mysd, name2itemid("사도의투구"),1;

			}
			set $fourclass, 8;
		}
		else
		{ // 승급기준이 아닌경우
			set @pos, dialog(0, 0, 0, "좀더 수련을 쌓고오세요.");
			end;
		}
	}
	end;
