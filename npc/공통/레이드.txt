@레이드시작	{
	set @sd, mysd();
	set @mapid, name2mapid("보스레이드");
	set @map_x, 8;
	set @map_y, 7;

	set @time, callfunc("Server_Nowtime");

	if(party(@sd) != 0)
	{
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @target_sd, party_member(@sd, @i);
			if(@target_sd > 0)
			{
				set @user_var, get_uservar_i(@target_sd, "$BOSS_timeset");
				if(@user_var + 300 > @time)
				{
					set @time_count5, @time_count5 + 1;
				}
				else
				{
					set @time_count5, 0;
				}
			}
		}
		if(@time_count5 > 0)
		{
			set @pos, dialog(0, 0, 0, "그룹원 중에 5분 이내에 보스레이드를 진행한 사람이 있습니다.");
			end;
		}
	}

// round check
	if(#BOSS > 0)
	{
		set @pos, dialog(0, 0, 0, "현재 레이드를 진행하고 있는 팀이 있습니다.\\n\\n잠시만 기다려주세요.");
		end;
	}
// party check
	if(! party(@sd))
	{
		set @pos, dialog(0, 0, 0, "그룹 없이는 레이드를 시작하실 수 없습니다.");
		end;
	}
// party boss check
	if(party_member(@sd, 0) != @sd)
	{
		set @pos, dialog(0, 0, 0, "그룹장만 레이드를 신청하실 수 있습니다.\\n\\n그룹장이 클릭하셔도 안된다면 그룹을 다시 한 번 만들어 보세요.");
		end;
	}
// party member check
	for(set @i, 0; @i < partymax(); set @i, @i+1)
	{
		set @target_sd, party_member(@sd, @i);
		if(@target_sd > 0)
		{
			if(get_level(@target_sd) < 99)
			{
				set @level_count, @level_count + 1;
			}
			if(get_mapid(@target_sd) != name2mapid("레이드대기실"))
			{
				set @mapcheck, @mapcheck + 1;
			}
			set @member_count, @member_count + 1;
		}
	}
	if(@member_count < 4 || @member_count > 5)
	{
		set @pos, dialog(0, 0, 0, "레이드의 적정인원은 4~5명 입니다.\\n\\n그룹인원 수를 확인해 주세요.");
		end;
	}

	if(@level_count > 0)
	{
		set @pos, dialog(0, 0, 0, "파티원 중 한명이라도 레벨이 99가 아닌 파티원이 있습니다.");
		end;
	}

	if(@mapcheck > 0)
	{
		set @pos, dialog(0, 0, 0, "파티원 모두가 대기실에 있어야 참여가 가능합니다.");
		end;
	}
	set @pos, list2(@list, 0, 0, "레이드에 참여하실 수 있습니다.\\n참여하시겠어요?", "지금 바로 입장하겠습니다.");
	if(@pos != 1) end;
	if(@list == 1)
	{
	if($BOSS_timeset < @time)
	{
		set $BOSS_timeset, @time+300;
	}
	else
	{
		set @pos, dialog(0, 0, 0, ($BOSS_timeset-@time) + "초 후에 입장이 가능합니다.");
		end;
	}
// party member sd # save in party_m array
		setarray @party_m$[0], "", "", "", "", "";
		setarray @party[0], 0, 0, 0, 0, 0;
		set @count, 0;
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @target_sd, party_member(@sd, @i);
			if(@target_sd > 0)
			{
				set @party_m$[@count], get_name(@target_sd);
				set @party[@count], @target_sd;
				set @count, @count + 1;
			}
		}
// save #
		set #BOSS_sd_boss$, @party_m$[0];
		set #BOSS_sd_1$, @party_m$[1];
		set #BOSS_sd_2$, @party_m$[2];
		set #BOSS_sd_3$, @party_m$[3];
		set #BOSS_sd_4$, @party_m$[4];

		broadcast 5, #BOSS_sd_boss$ + "님의 그룹이 레이드에 도전합니다!";

// warp party + BOSS check
		for(set @i, 0; @i < 5; set @i, @i+1)
		{
			if(@party[@i] > 0)
			{
				warp @party[@i], @mapid, @map_x + rand(1,3), @map_y + rand(1,3);
			}
		}
		set #BOSS, 1;
		set #BOSS_count, @member_count;
	}
}

		}
		for(set @i, 0; @i < 5; set @i, @i+1)

			{

				warp @party_m[@i], 15, 5, 5;
			}
