		end;

		end;
		}

정답도우미	{
	set @sd, mysd();
	set @id, myid();
	if(#quiz != 1 && #userquiz != 1){dialog 0,0,1, "퀴즈가 시작되지 않았습니다."; end;}
	if(spellaethered(@sd, name2spellid("퀴즈저주")) == 1) {dialog 1, 0, 1, "당신은 퀴즈저주가 걸려있어서 정답을 말할 수 없습니다."; end;}
	if(#quiz == 1){
	set @input, input1(@answer$, "정답을 입력해주세요.\\n틀릴 경우엔 10초간 정답을 말하실 수 없습니다.");
	if(@answer$ != #quiz_a$) {
	spellaether @sd, name2spellid("퀴즈저주"), 10;
	dialog 0, 0, 1, "정답이 아닙니다.\\n10초간 답을 입력할 수 없습니다.";
	action @id, 12, 30, 0;
	end;
	}
	if(@answer$ == #quiz_a$){
	if(#quiz != 1){dialog 0,0,1, "이미 정답자가 나왔습니다."; end;}
	set #quiz, 0;
	if(countitem(name2itemid("퀴즈상품권")) <= 0){
		additem @sd, name2itemid("아이템체크1"),1;
	sleep (100);
	if(countitem(name2itemid("아이템체크1")) <= 0) {dialog 0,0,1, "인벤토리 공간이 없습니다."; end;}
	sleep (100);
		delitem @sd, name2itemid("아이템체크1"),1;
	}
		say @id, 1, "아! 정답!";
		effect @id, 221;
		broadcast 4, "" + get_name(@sd) + " 님이 문제를 맞추셨습니다.";
		dialog 0, 0, 1, "정답입니다.\\n축하드립니다.";
		additem @sd, name2itemid("퀴즈상품권"),1;
		sleep 2500;
		broadcast 4, "정답은 " + #quiz_a$ + "입니다.";
		end;
		}
	}
	if(#userquiz == 1){
	set @input, input1(@answer$, "정답을 입력해주세요.\\n틀릴 경우엔 10초간 정답을 말하실 수 없습니다.");
	if(@answer$ != #userquiz_a$) {
	spellaether @sd, name2spellid("퀴즈저주"), 10;
	dialog 0, 0, 1, "정답이 아닙니다.\\n10초간 답을 입력할 수 없습니다.";
	action @id, 12, 30, 0;
	end;
	}
	if(@answer$ == #userquiz_a$){
	if(#userquiz != 1){dialog 0,0,1, "이미 정답자가 나왔습니다."; end;}
	set #userquiz, 0;
	additem @sd, name2itemid("아이템체크1"),1;
	sleep (100);
	if(countitem(name2itemid("아이템체크1")) <= 0) {dialog 0,0,1, "인벤토리 공간이 없습니다."; end;}
			sleep (100);

			say @id, 1, "아! 내가 맞혔다!";
			effect @id, 221;
			broadcast 4, "" + get_name(@sd) + " 님이 문제를 맞추셨습니다.";
			dialog 0, 0, 1, "정답입니다.\\n축하드립니다.";
			additem @sd, name2itemid(#userquiz_item$), #userquiz_itemnum;
			sleep 2500;
			broadcast 4, "정답은 " + #userquiz_a$ + "입니다.";
			end;
		}
	}
