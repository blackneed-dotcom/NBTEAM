선인	{
	set @sd, mysd();
	set @id, myid();
	image 0, 212, 0;
	if(get_joblevel(@sd) == 6)
	{
		if(get_job(@sd) == 1 || get_job(@sd) == 2) { set @hp, 6000000; set @mp, 0; }
		else if(get_job(@sd) == 3) { set @hp, 2000000; set @mp, 4000000; }
		else if(get_job(@sd) == 4) { set @hp, 2000000; set @mp, 4000000; }

		if(get_purehp(@sd) < @hp || get_puremp(@sd) < @mp) goto COS00008;

		if(!$sevenclass) goto COS00000;
		else if($sevenclass == 1) goto COS00001;
		else if($sevenclass == 2) goto COS00002;
		else if($sevenclass == 3) goto COS00003;
		else if($sevenclass == 4) goto COS00004;
		else if($sevenclass == 5) goto COS00005;
		else if($sevenclass == 6) goto COS00006;
		else goto COS00007;
COS00000:
		set @pos, dialog(1, 0, 1, "인간임에도 불구하고 이정도 힘을 가지다니..\\n\\n잘와주었다네, 이세계의 근간을 흔드는일이 \\n생겼다는걸 자네도 알고있는가?\\n\\n지금 조물주의 분신인\\n사태자와 악의 주인이\\n부활되어 세상을\\n파괴하고있다네!.");
		if(@pos == 0) end;
		set @pos, list2(@ret, 1, 1, "부디 그들을 물리쳐 주시게!", "제가 해보겠습니다.", "저는 아직 부족합니다..");
		if(@pos == 0) end;
		else if(@pos == -1) goto COS00000;
		else if(@ret == 1)
		{
			set $sevenclass, 1;
			additem @sd, name2itemid("사신수의심장"),1;
			addlegend @sd, 201, 7, 0, 16, "적태자 처치의 임무를 받다.";
			set @pos, dialog(0, 0, 1, "내가 준 사신수의심장이 있어야 적태자의방에 입장 가능하네.");
		}
		end;
COS00001:
		if(countitem(name2itemid("적태자의심장")) > 0)
		{
			set $sevenclass, 2;
			set @pos, dialog(1, 0, 1, "적태자... 교활한놈이지, 이제\\n시작이라네.\\n\\n흑태자를 잡아주시게!.\\n\\n흑태자도 가만히 있지않을걸세.");
			addlegend @sd, 201, 7, 0, 16, "흑태자 처치의 임무를 받다.";
			set @pos, dialog(0, 0, 1, "흑태자..교활한놈이지.. 조심하시게...");
		}
		end;
COS00002:
		if(countitem(name2itemid("흑태자의심장")) > 0)
		{
			set $sevenclass, 3;
			delitem @sd, name2itemid("적태자의심장"), 1;
			set @pos, dialog(1, 0, 1, "만만치 않았을걸세 고생햇네.\\n\\n이번엔 백태자를 상대해야 하네.\\n\\n부디 몸조심하시게.");
			addlegend @sd, 201, 7, 0, 16, "백태자 처치의 임무를 받다.";
			set @pos, dialog(0, 0, 1, "악의주인이 곧 도착한다더군..!...");
		}
		end;
COS00003:
		if(countitem(name2itemid("백태자의심장")) > 0)
		{
			set $sevenclass, 4;
			delitem @sd, name2itemid("흑태자의심장"), 1;
			set @pos, dialog(1, 0, 1, "수고했네..악의기운이 조금 누그러졋군,\\n\\n이번엔 황태자를 상대해야 하네.\\n\\n부탁하네.");
			addlegend @sd, 201, 7, 0, 16, "황태자 처치의 임무를 받다.";
			set @pos, dialog(0, 0, 1, "태자중 마지막 태자일세 조심하시게.....");
		}
		end;
COS00004:
		if(countitem(name2itemid("황태자의심장")) > 0)
		{
			set $sevenclass, 5;
			delitem @sd, name2itemid("백태자의심장"), 1;
			set @pos, dialog(1, 0, 1, "현무를 잘 물리쳐 주었네.\\n\\n하지만 끝이 아닐세! 한서린이무기가 부활햇네!\\n\\n이무기를 제거하고 증표를 가져오게!.");
			addlegend @sd, 201, 7, 0, 16, "한서린이무기 제거 임무를 받다.";
			set @pos, dialog(0, 0, 1, "악의주인이 아끼는 수하를 물리치게...");
		}
		end;
COS00005:
		if(countitem(name2itemid("이무기의어금니")) > 0)
		{
			set @pos, dialog(1, 0, 1, "이것으로 한시름 놓은건가?..\\n\\n이제 악의\\n기운이 사라지겠지...?!\\\n\\n아니, 이것은? 이것이 어떻게 된 일이란 말인가?\\n\\n어둠의 기운이 이토록 온 천지를 덮고 있다니....\\n어째서 이토록 증폭되고 있단 말인가!");
			if(@pos == 0) end;
			set @pos, list2(@ret, 1, 1, "결국 우리 인간은 신들의 예언대로 자멸하고 마는군.\\n\\n으흑흑!", "포기하지마십시오! 선인!", "제가 악의 주인을 봉인해오겠습니다!");
			if(@pos == 0) end;
			else if(@pos == -1) goto COS00006;
			else if(@ret == 1)
			{
				set $sevenclass, 6;
				delitem @sd, name2itemid("이무기의어금니"), 1;
				set @pos, dialog(1, 0, 1, "자네에게 이런 짐을 지우는 것이 못내\\n안쓰럽네만, 기회는 지금 뿐일세.\\n\\n어서 최후의전장으로 가서 주인을 물리치고\\n그의 심장을 구해오게!");
				addlegend @sd, 201, 7, 0, 16, "어둠의주인 처치의 임무를 받다.";
				set @pos, dialog(0, 0, 1, "어서 최후의 전장으로 가보게!");
			}
		}
		end;
COS00006:
		if(countitem(name2itemid("악의주인의심장")) > 0)
		{
			set $sevenclass, 7;
			dellegend @sd, 201;
			delitem @sd, name2itemid("황태자의심장"), 1;
			set @pos, dialog(1, 0, 1, "악의주인의심장을 들고 천상계의 천주마님을\\n찾아가게.\\n\\n이번 일로 자네는 다시 한번 세상을 살려냈네.\\n위대한 자네인만큼, 그에 걸맞는 새로운 지위를\\n인정받을 수 있을 것이네.");
		}
		end;
COS00007:
		set @pos, dialog(1, 0, 0, "자네는 아직 약하다네...");
		end;
	}
}

천신선인	{
	image 0, 130, 0;
	set @sd, mysd();
	set @id, myid();
	if($sevenclass == 7) { goto POS00000; }
	else { set @pos, dialog(1, 0, 1, "..."); end; }
POS00000:
	if(get_job(@sd) == 1) { set @jobname$, "검존"; }
	else if(get_job(@sd) == 2) { set @jobname$, "패황"; }
	else if(get_job(@sd) == 3) { set @jobname$, "마성"; }
	else if(get_job(@sd) == 4) { set @jobname$, "선인"; }

	set @pos, list2(@ret, 1, 0, "자네는 지금껏 수많은 수련을 거쳐 인간으로서 오르기\\n힘든 신의 경지에 도달하였고, 더욱이 세상을 종말로부터\\n구해내는 업적을 세웠으니 " + @jobname$ + "으로 불리우시기에\\n부족함이 없네.\\n지금 바로 승급을 원하시는가?", "예", "아뇨");
	if(@pos == 0) { end; }
	else if(@ret == 1)
	{
		setarray @hp[0], 6000000, 6000000, 2600000, 3000000;// HP 승급기준
		setarray @mp[0], 500000, 700000, 3500000, 3000000;// MP 승급기준
		
		if(get_level(@sd) >= 99 && get_purehp(@sd) >= @hp[get_job(@sd)-1] && get_puremp(@sd) >= @mp[get_job(@sd)-1]) // 승급기준인경우
		{
			if ( countitem(name2itemid("악의주인의심장")) == 0)
			{
				set @pos, dialog(0, 0, 0, "악의주인을 처치하지 않앗군!");
				end;
			}

			delitem @sd, name2itemid("악의주인의심장"), 1;
			set_joblevel @sd, get_joblevel(@sd)+1;
			broadcast 4, get_name(@sd) + " 님이 절대경지의 [" + get_jobname(@sd) + "] 승급을 완료했습니다. 축하드립니다!";

			if(get_job(@sd) == 1)
			{
				additem @sd, name2itemid("노란비서"),1;
			}
			else if(get_job(@sd) == 2)
			{
				additem @sd, name2itemid("노란비서"),1;
			}
			else if(get_job(@sd) == 3)
			{
				additem @sd, name2itemid("노란비서"),1;
			}
			else if(get_job(@sd) == 4)
			{
				additem @sd, name2itemid("노란비서"),1;
			}
			set $sevenclass, 8;
		}
		else
		{ // 승급기준이 아닌경우
			set @pos, dialog(0, 0, 0, "자네의 수련도로는 부족하네...");
			end;
		}
	}
}
