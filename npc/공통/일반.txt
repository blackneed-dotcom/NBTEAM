		if(@spell == 1)
		{
			if(checkspell(name2spellid("선필")) == 52)
			{
				addspell name2spellid("선필");
				set @pos, dialog(0, 0, 0, "선필 마법을 가르쳐 드렸습니다.");
				end;
			}
			else
			{
				set @pos, dialog(0, 0, 0, "이미 배우셨습니다.");
				end;
			}
		}
		else if(@spell == 2)
		{
			for(set @i, 0; @i < 52; set @i, @i+1)
			{
				if(checkspell(name2spellid("선필")) == @i)
				{
					delspell @i;
				}
			}
			set @pos, dialog(0, 0, 0, "선필 마법을 지웠습니다.");
			end;
		}
	}
	else if(@list == 3)
	{
			set @pos, input2(@name$, 0, 6, "선필 대련 승부를 요청할 상대방의 이름은 무엇입니까?", "상대방의 이름은", "입니다.");
			if(@pos == 0) end;

			set @target_id, name2id(@name$);
			if(validid(@target_id) == 1)
			{
				set @target_sd, id2sd(@target_id);

				if(spellaethered(@target_sd, name2spellid("선필요청")) == 1)
				{
					message @sd, 8, "Please wait delay.";
					end;
				}
				else if(get_level(@sd) < 99 || get_level(@target_sd) < 99)
				{
					set @pos, dialog(0, 0, 0, "신청자 또는 요청자가 레벨 99미만이라 선필 대련을 하실 수 없습니다.");
					end;
				}
				else if(@name$ == get_name(@sd))
				{
					set @pos, dialog(0, 0, 0, "본인의 이름으로 신청하실 수 없습니다.");
					end;
				}
					set #request, 1;
					set #appname$, get_name(@sd);

					spellcast @target_sd, name2spellid("선필요청");
				}
				else
				{
					set @pos, dialog(0, 0, 0, @name$+"님이 가까이에 계시지않아 요청할 수 없습니다.");
					end;
				}
			}
			else
			{
				set @pos, dialog(0, 0, 0, @name$+"님은 향수서버에 접속중이 아닙니다.");
				end;
			}
		}
		else
		{
			set @pos, dialog(0, 0, 0, ($first_strike-@time)+"초 후에 요청 가능합니다.");
			end;
		}
	}
}

@선필요청	{
	set @id, myid();
	set @sd, mysd();

	if(#request == 1)
	{
		set @pos, list2(@list, 0, 0, #appname$+"님이 당신에게 선필 승부 대련을 요청하고자합니다.\\n수락하시겠습니까?", "수락", "거절");
		if(@pos == 0 || @list == 2) set @ref, 1;

		else if(@list == 1)
		{
			if(#request != 0)
			{
				set #request, 0;
				set #agree, 1;
				set #agrname$, get_name(@sd);

				warp @sd, name2mapid("선필대련장"), 16, 5;
				spellaether @sd, name2spellid("선필마감"), 180;

				set @target_sd, name2sd(#appname$);
				warp @target_sd, name2mapid("선필대련장"), 3, 5;
				spellaether @target_sd, name2spellid("선필마감"), 180;

				set @objid, obj_getxy(7, 10);
				say @objid, 0, "선필대련심판: ["+#appname$+"]님과 ["+#agrname$+"]님의 일대일 선필 대련";
				say @objid, 0, "선필대련심판: 3분내로 승부가 결정나지 않으면 두 분 모두 무승부!";
				say @objid, 0, "선필대련심판: 그럼 이제 시작!!";
			}
			else
			{
				set @pos, dialog(0, 0, 0, "요청 받았던 시간이 초과되었습니다.");
				end;
			}
		}
		if(@ref == 1)
		{
			set @target_sd, name2sd(#appname$);
			runscript @target_sd, "@선필요청취소";
		}
	}
}

@선필요청취소	{
	set #request, 0;
	set #appname$, "";
	set #agrname$, "";

	if(mysd() > 0)
	{
		message mysd(), 5, "[정보]: 선필 요청이 종료되었습니다.";
	}
}



@선필	{
	set @id, myid();
	set @sd, mysd();
	set @target_id, obj_getfront();
	if(get_state(@sd) == 2 || get_state(@sd) == 5)
	{
		message @sd, 3, "투명 상태일땐 사용이 불가능합니다.";
		end;
	}
	if(@target_id > 0)
	{
		if(#agree == 1 && get_mapname(@sd) == "선필대련장" && istype(@target_id) == 1)
		{
			if(get_x(@id) >= 1 && get_x(@id) <= 18 && get_y(@id) >= 2 && get_y(@id) <= 8)
			{

				callfunc "MANA_DELAY", 60, 0, @sd;
				if(@target_id > 0) { message @sd, 3, "선필을 외웠습니다."; }
				callfunc "MAGICATTACK", @id, @sd, @damage, get_hp(@sd)/2, 60, "선필!!", 60, 35, 1;

				sleep 1000;

				set @target_sd, name2sd(#appname$);
				set @sd, name2sd(#agrname$);

				warp @sd, name2mapid("선필대련장"), 11, 11;
				runscript @sd, "선필딜레이";
				warp @target_sd, name2mapid("선필대련장"), 8, 11;
				runscript @target_sd, "선필딜레이";

				spelluncast @target_sd, name2spellid("선필마감");
				spelluncast @target_sd, name2spellid("선필요청");

				set @sd, mysd();
				set @objid, obj_getxy(7, 10);
				say @objid, 0, "선필대련심판: "+get_name(@sd)+"님 승리!!";

				spelluncast @sd, name2spellid("선필마감");
				spelluncast @sd, name2spellid("선필요청");

				set #appname$, "";
				set #refname$, "";
				set #agrname$, "";
				set #sp_wins, 0;
				set #request, 0;
				set #agree, 0;

				end;
			}
			else
			{
				message @sd, 3, "대련장에 올라가지 않아 사용할 수 없습니다.";
				end;
			}
		}
		else
		{
			message @sd, 3, "선필 대련 진행중이 아닙니다.";
			end;
		}
	}
}
