@진원관입구	{
	set @id, myid();
	set @sd, mysd();
	
	if(party(@sd) != 0)
	{
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @ts, party_member(@sd,@i);
			if(@ti > 0)
			{
				if(@i == 0)
				{
					if(checkitem(@ts,name2itemid("노란비서")) == 0)
					{
						dialog 0,0,1, "파티장은 노란비서를 가지고 있어야합니다.";
						end;
					}
				}
				set @job, get_job(@ts);
				set @count, @count+1;

				if(@job == 1 || @job == 2) { set @maxhp, 3500000; set @maxmp, 500000; }
				else if(@job == 3 || @job == 4) { set @maxhp, 1800000; set @maxmp, 2200000; }
				else	
				{
					dialog 0,0,1, "일행 중 평민이 있습니다.";
					end;
				}
				if(get_mapid(@sd) != get_mapid(@ts))
				{
					dialog 0,0,1, "일행이 모두 한곳에 있어야합니다.";
					end;
				}
				if(@count > 5)
				{
					dialog 0,0,1, "일행은 5명이하만 입장 가능합니다.";
					end;
				}
				if(get_purehp(@ts) < @maxhp || get_puremp(@ts) < @maxmp)
				{
					dialog 0,0,1, "일행 중 체력,마력 조건이 되지 않는 일행이 있습니다.";
					dialog 0,0,1, "[격수]: 체력 350만, 마력 50만 이상부터 입장하실 수 있습니다.";
					dialog 0,0,1, "[비격수]: 체력 180만, 마력 220만 이상부터 입장하실 수 있습니다.";
					end;
				}

			}
		}
		for(set @i, 0; @i < @count; set @i, @i+1)
		{

			set @ts, party_member(@sd,@i);
			if(@ts > 0)
			warp @ts, name2mapid("국내진원관1"), rand(9, 12), 14;
		}					
	}
	else
	{
		if(get_job(@sd) == 1 || get_job(@sd) == 2)
		{
			if(get_purehp(@sd) >= 3500000 && get_puremp(@sd) >=500000)
			{
				if(name2mapid("진원관대기실") > 0) { warp @sd, name2mapid("진원관대기실"), rand(9, 12), 14; end; }
			}
			else
			{
				message @sd, 5, "[알림]: 체력 350만, 마력 50만 이상부터 입장하실 수 있습니다.";
				warp @sd, get_mapid(@sd), get_x(@id)-2, get_y(@id);
				end;
			}
		}
		else if(get_job(@sd) == 3 || get_job(@sd) == 4)
		{
			if(get_purehp(@sd) >= 1800000 && get_puremp(@sd) >= 2200000)
			{
				if(name2mapid("진원관대기실") > 0) { warp @sd, name2mapid("진원관대기실"), rand(9, 12), 14; end; }
			}
			else
			{
				message @sd, 5, "[알림]: 체력 180만, 마력 220만 이상부터 입장하실 수 있습니다.";
				warp @sd, get_mapid(@sd), get_x(@id)-2, get_y(@id);
				end;
			}
		}
		else
		{
			warp @sd, get_mapid(@sd), get_x(@id)-2, get_y(@id);
			message @sd, 3, "감히 들어갈 수 없습니다.";		
			end;
		}
	}
}

			{
				set @target_sd, party_member(@sd, @i);
				if(@target_sd > 0)
				{
					if(get_mapid(@sd) == get_mapid(@target_sd))
					{
						if(@sd != @target_sd)
						{
							message @target_sd, 8, get_name(@sd) + "님에 의해 적태자의방으로 이동하였습니다.";

	else
	{
		delitem @sd, name2itemid("흑태자의심장"), 1;

		{
			set @pos, dialog(0, 0, 0, "백태자의방으로 이동하였습니다.");

			end;
		}
		else
		{
			for(set @i, 0; @i < partymax(); set @i, @i+1)
			{
				set @target_sd, party_member(@sd, @i);
				if(@target_sd > 0)
				{
					if(get_mapid(@sd) == get_mapid(@target_sd))
					{
						if(@sd != @target_sd)
						{
							message @target_sd, 8, get_name(@sd) + "님에 의해 백태자의방으로 이동하였습니다.";
							warp @target_sd, @warp_id, rand(4, 5), rand(4, 5);
						}
					}
				}
			}
			set @pos, dialog(0, 0, 0, "백태자의방으로 그룹원과 함께 이동하였습니다.");
			warp @sd, @warp_id, rand(4, 5), rand(4, 5);
			end;
		}
	}
