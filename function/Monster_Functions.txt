//=================몬스터드랍, 몬스터마법 부분=================
//callfunc "Monster_Drop_End", objid(), 확률변수, 확률, 아이템번호, 갯수;
FUNC_Monster_Drop_End	{
	set @id, getarg(0);
	set @rate, rand(1, getarg(1));
	set @rand, getarg(2);
	set @item, getarg(3);
	set @drop, getarg(4);

	if(@rand > @rate) { set @rand, @rate; }

	if(@item <= 0)
	{
		broadcast 5, " " + get_mobname(@id) + " 몬스터 드랍 오류!! 운영진에게 보고부탁드립니다.";
		end;
	}
	if(@rate <= @rand)
	{
		dropitem get_mobx(@id), get_moby(@id), @item, @drop;
		end;
	}
	return;
}

//원거리마법공격(0: 원거리, 1: 근접, 2: 격, 3: 필살), 확률변수?, 이펙트, 효과음, 데미지1, 데미지2, "격이름", 확률;
FUNC_MONSTER_ATTACKS	{
	set @id, objid();
	set @type, getarg(0);
	set @effect, getarg(2);
	set @sound, getarg(3);
	set @damage, rand(getarg(4), getarg(5));
	set @name$, getarg(6);
	set @target_id, get_mobtarget(@id);
	set @target_sd, id2sd(@target_id);
	set @rand1, rand(1, getarg(1));
	set @target_ac_check, 0;
	set @target_ac_1, 0;
	if(get_state(@target_sd) == 1){ end; }
	if(@rand1 > getarg(7)){ end; }
	if(mobspellaethered(@id, name2spellid("절망")) == 1) { mobspellaether @id, name2spellid("절망"), 1; }
	if(@target_id > 0)
	{
		effect @target_id, @effect;
		sound @target_id, @sound;
		action @id, 1, 30, @sound;
		set @target_ac, get_ac(@target_sd);
		if(@target_ac < -95)
		{
			set @damage_ac, @damage*100/(500+(95+@target_ac)*5)*100;
		}
		else
		{
			set @damage_ac, @damage/(100+@target_ac)*100;
		}
		damage @id, @target_id, @damage_ac, @type;		
	}
	return;
}

//callfunc "MONSTER_FRONT_ATTACK", 몬스터id값, 확률변수, 확률, 액션, 효과음, 이펙트, 대미지, 대미지타입, "마법이름", "몬스터할말";
FUNC_MONSTER_FRONT_ATTACK	{
	set @id, getarg(0);
	set @rand, rand(1, getarg(1));
	set @rate, getarg(2);
	set @action, getarg(3);
	set @sound, getarg(4);
	set @effect, getarg(5);
	set @damage, getarg(6);
	set @type, getarg(7);
	set @magic$, getarg(8);
	set @say$, getarg(9);

	if(mobspellaethered(@id, name2spellid("절망")) == 1) { mobspellaether @id, name2spellid("절망"), 1; }

	set @target_id, obj_getmobfront();
	if(@target_id > 0)
	{
		if(istype(@target_id) == 1)
		{
			set @target_sd, id2sd(@target_id);

			if(@action >= 0) { action @id, @action, 30, 0; }
			if(@sound >= 0) { sound @target_id, @sound; }
			if(@effect >= 0) { effect @target_id, @effect; }

			damage @id, @target_id, rand(150000, 250000), 2;
			message @target_sd, 3, get_mobname(@id) + "(이)가 " + @magic$ + "을(를) 가합니다.";

			if(@say$ != "-1") { say @id, 0, get_mobname(@id) + ": " + @say$; }
		}
	}
	return;
}

//callfunc "MONSTER_RANGE_ATTACK", 몬스터id값, 확률변수, 실패확률, 액션, 효과음, 이펙트, 대미지, 대미지타입, "마법이름", 범위, "몬스터할말";
FUNC_MONSTER_RANGE_ATTACK	{
	set @id, getarg(0);
	set @rand, rand(1, getarg(1));
	set @rate, getarg(2);
	set @action, getarg(3);
	set @sound, getarg(4);
	set @effect, getarg(5);
	set @damage, getarg(6);
	set @type, getarg(7);
	set @magic$, getarg(8);
	set @range, getarg(9);
	set @say$, getarg(10);

	if(mobspellaethered(@id, name2spellid("절망")) == 1) { mobspellaether @id, name2spellid("절망"), 1; }

	if(@rand <= @rate) { end; }

	set @x1, get_x(@id)-@range;
	set @x2, get_x(@id)+@range;
	set @y1, get_y(@id)-@range;
	set @y2, get_y(@id)+@range;

	if(@x1 < 0) { set @x1, 0; }
	if(@x2 > mapwidth()-1) { set @x2, mapwidth()-1; }
	if(@y1 < 0) { set @y1, 0; }
	if(@y2 > mapheight()-1) { set @y2, mapheight()-1; }

	for(set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for(set @j, @y1; @j <= @y2; set @j, @j+1)
		{
			set @target_id, obj_getxy(@i, @j);
			if(@target_id > 0)
			{
				if(istype(@target_id) == 1)
				{
					set @target_sd, id2sd(@target_id);

					if(@action >= 0) { action @id, @action, 30, 0; }
					if(@sound >= 0) { sound @target_id, @sound; }
					if(@effect >= 0) { effect @target_id, @effect; }

					damage @id, @target_id, @damage, @type;
					message @target_sd, 3, get_mobname(@id) + "(이)가 " + @magic$ + "을(를) 가합니다.";

					if(@say$ != "-1") { say @id, 0, get_mobname(@id) + ": " + @say$; }
				}
			}
		}
	}
	return;
}

FUNC_DROPITEM	{
	if(rand(1, getarg(1)) <= getarg(2))
	{
		if(getarg(3) > 0) { dropitem get_mobx(getarg(0)), get_moby(getarg(0)), getarg(3), getarg(4); }
		else { announce 5, get_mobname(getarg(0)) + "몹 드랍이 오류다!"; }
		end;
	}
	return;
}

FUNC_mob_constanting_spell	{
	set .@id, objid();

	//setting
	set .@percent, getarg(0);
	set .@action, getarg(1);
	set .@effect, getarg(2);
	set .@sound, getarg(3);
	set .@damage, getarg(4);
	set .@unbeatable, getarg(5);
	set .@magic$, getarg(6);
	set .@etc$, getarg(7);

	if(.@percent >= rand(1, 1000))
	{
		set .@target_id, get_mobtarget(.@id);
		if(istype(.@target_id) == 1)
		{
			action .@id, .@action, 30, 0;
			effect .@target_id, .@effect;
			sound .@id, .@sound;

			set .@target_sd, id2sd(.@target_id);
			if(get_state(.@target_sd) == 1) end;

			if(.@unbeatable == 1 && spellaethered(.@target_sd, name2spellid("금강불체")) == 1)
			{
				effect .@target_id, .@effect;
				message .@target_sd, 3, get_mobname(.@id) + "이(가) " + .@magic$ + .@etc$ + " 가합니다.";
				end;
			}
			callfunc "mob_damage_attack", .@target_id, .@target_sd, .@effect, .@damage;
			message .@target_sd, 3, get_mobname(.@id) + "이(가) " + .@magic$ + .@etc$ + " 가합니다.";
		}
	}
	return;
}
FUNC_mob_damage_attack	{
	set .@id, objid();

	set .@target_id, getarg(0);
	set .@target_sd, getarg(1);

	if(istype(.@target_id) != 1) end;

	set .@effect, getarg(2);
	set .@damage, getarg(3);



	//방어력
	set .@ac, get_ac(.@target_sd);
	if(.@ac > 99) { set .@ac, 99; }
	else if(.@ac < -95) { set .@ac, -95; }

	set .@damage, .@damage+(((.@damage/100)*.@ac)/10);

	//마법 방어력
	set .@spell_shield, get_protection(.@target_sd);
	if(.@spell_shield < 0) { set .@spell_shield, 0; }
	else if(.@spell_shield > 20) { set .@spell_shield, 20; }

	set .@decrease, ((.@damage/100)*4)*.@spell_shield;
	if(.@decrease < 0) { set .@decrease, 0; }

	set .@attack_shield, rand(1, 800)-.@spell_shield;
	if(.@attack_shield < .@spell_shield) { set .@value, 0; }

	set .@damage, .@damage-.@decrease;
	if(.@damage < 0) { set .@damage, 0; }

	set .@harm_hp, get_hp(.@target_sd)-.@damage;
	if(.@harm_hp > 0)
	{
		set_hp .@target_sd, .@harm_hp;

		if(istype(.@target_id) == 1)
			damage .@id, .@target_id, 1, 2;
	}
	else
	{
		set_hp .@target_sd, 1;

		if(istype(.@target_id) == 1)
			damage .@id, .@target_id, 1000000, 2;
	}
	return;
}
//callfunc "mob_atlarge_spell", percent(max:1000), action, effect, sound, dam, unbeatable, rangex, rangey, "spellname", "etc", "etc2";
FUNC_mob_atlarge_spell	{
	set .@id, objid();

	//setting
	set .@percent, getarg(0);
	set .@action, getarg(1);
	set .@effect, getarg(2);
	set .@sound, getarg(3);
	set .@damage, getarg(4);
	set .@unbeatable, getarg(5);
	set .@xrange, getarg(6);
	set .@yrange, getarg(7);
	set .@magic$, getarg(8);
	set .@etc$, getarg(9);
	set .@etc2$, getarg(10);

	if(.@percent >= rand(1, 1000)) end;

	set .@x1, get_x(.@id)-.@xrange;
	set .@x2, get_x(.@id)+.@xrange;
	set .@y1, get_y(.@id)-.@yrange;
	set .@y2, get_y(.@id)+.@yrange;

	if(.@x1 < 0) { set .@x1, 0; }
	if(.@x2 > mapwidth()-1) { set .@x2, mapwidth()-1; }
	if(.@y1 < 0) { set .@y1, 0; }
	if(.@y2 > mapheight()-1) { set .@y2, mapheight()-1; }

	action .@id, .@action, 30, 0;
	sound .@id, .@sound;

	for(set .@i, .@x1; .@i <= .@x2; set .@i, .@i+1)
	{
		for(set .@j, .@y1; .@j <= .@y2; set .@j, .@j+1)
		{
			set .@target_id, obj_getxy(.@i, .@j);
			if(.@target_id > 0)
			{
				if(istype(.@target_id) != 1) continue;

				set .@target_sd, id2sd(.@target_id);
				if(get_state(.@target_sd) == 1) continue;

				if(.@unbeatable == 1 && spellaethered(.@target_sd, name2spellid("금강불체")) == 1)
				{
					effect .@target_id, .@effect;

					message .@target_sd, 3, get_mobname(.@id) + "가 " + .@magic$ + .@etc$ + " 가합니다.";
					continue;
				}

				effect .@target_id, .@effect;

				callfunc "mob_damage_attack", .@target_id, .@target_sd, .@effect, .@damage;
				message .@target_sd, 3, get_mobname(.@id) + .@etc2$ + " " + .@magic$ + .@etc$ + " 가합니다.";
			}
		}
	}
	return;
}
