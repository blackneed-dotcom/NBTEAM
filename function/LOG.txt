FUNC_ITEMROUTELOG	{
	set .@sd, mysd();
	set .@id, myid();

	set .@year$, gettime(5,0);
	set .@month$, gettime(4,0);
	set .@day$, gettime(3,0);
	set .@hour$, gettime(2,0);
	set .@min$, gettime(1,0);
	set .@sec$, gettime(0,0,);

	if(gettime(5,0) < 10) set .@year$, "0"+gettime(5,0);
	if(gettime(4,0) < 10) set .@month$, "0"+gettime(4,0);
	if(gettime(3,0) < 10) set .@day$, "0"+gettime(3,0);
	if(gettime(2,0) < 10) set .@hour$, "0"+gettime(2,0);
	if(gettime(1,0) < 10) set .@min$, "0"+gettime(1,0);
	if(gettime(0,0) < 10) set .@sec$, "0"+gettime(0,0);

	if(mapid() < 10) set .@mapid$, "00000"+mapid();
	else if(mapid() < 100) set .@mapid$, "0000"+mapid();
	else if(mapid() < 1000) set .@mapid$, "000"+mapid();
	else if(mapid() < 10000) set .@mapid$, "00"+mapid();
	else if(mapid() < 100000) set .@mapid$, "0"+mapid();

	set .@x$, get_x(.@id);
	set .@y$, get_y(.@id);

	if(get_x(.@id) < 10) set .@x$, "00"+get_x(.@id);
	else if(get_x(.@id) < 100) set .@x$, "0"+get_x(.@id);

	if(get_y(.@id) < 10) set .@y$, "00"+get_y(.@id);
	else if(get_y(.@id) < 100) set .@y$, "0"+get_y(.@id);

	set .@mapinfo$, .@mapid$+"/"+.@x$+"/"+.@y$+"";
	set .@type$, getarg(0);

	//아이피 구해오기
	set .@ip$, "";
	for(set .@i, 0; .@i < 4; set .@i, .@i + 1)
	{
		set .@ip_class$, strtok(ip(.@sd), ".", .@i);

		if(atoi(.@ip_class$) < 10) set .@ip_class$, "00" + .@ip_class$;
		else if(atoi(.@ip_class$) < 100) set .@ip_class$, "0" + .@ip_class$;

		if(.@i == 3)
			set .@ip$, .@ip$ + .@ip_class$;
		else
			set .@ip$, .@ip$ + .@ip_class$ + ".";
	}

	if(.@type$ == "얻음")
	{
		if(get_lastget_success(.@sd) == 1)
		{
			set .@str$, "아이템이름: " + itemid2name(get_lastget_itemid(.@sd));
		}
	}
	else if(.@type$ == "떨굼")
	{
		set .@str$, "아이템이름: " + itemid2name(get_lastdrop_itemid(.@sd));
	}
	else
	{
		set .@str$, "";
	}

	set .@str$, .@str$ + " :: " + get_money(.@sd) + "원 " + get_itemlist(.@sd);

	write_log "log/itemroute/"+.@year$+"년_"+.@month$+"월_"+.@day$+"일_"+.@hour$+"시.txt", "["+.@hour$+"시 "+.@min$+"분 "+.@sec$+"초 - " + .@ip$ + " - " +.@mapinfo$+" - " + .@type$ + "] ■ "+get_name(.@sd)+": "+ .@str$ + " \\n";

	return;
}
