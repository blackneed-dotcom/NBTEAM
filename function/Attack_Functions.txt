//callfunc "calculate_damage", value, type(1:근접, 2:원거리, 3:부분범위, 4:전체범위);
FUNC_calculate_damage	{
	set @sd, mysd();
	set @value, getarg(0);

	if(getarg(1) == 1) { set @n, 97; set @k, 3; }
	else if(getarg(1) == 2) { set @n, 113; set @k, 7; }
	else if(getarg(1) == 3) { set @n, 137; set @k, 12; }
	else if(getarg(1) == 4) { set @n, 100; set @k, 0; }
	else { set @n, 75; set @k, 0; }

	if(@value >= 0 && @value < 7000)
	{
		set @calculate_min, @value-1;
		set @calculate_max, @value+1;
	}
	else
	{
		set @calculate_min, @value/10000*((90-@k)*@n);
		set @calculate_max, @value/10000*((110+@k)*@n);
	}

	if(@calculate_min > 300000000 || @calculate_max > 300000000) { message @sd, 5, "계산 오류!"; return; end; }
	if(@calculate_min > @calculate_max) { set @calculate_max, @calculate_min; }

	set @printf, rand(@calculate_min, @calculate_max);
	return @printf;
}

//callfunc "calculate_damage", value, type(1:근접, 2:원거리, 3:부분범위, 4:전체범위);
FUNC_calculate_damage	{
	set @sd, mysd();
	set @value, getarg(0);

	if(getarg(1) == 1) { set @n, 97; set @k, 3; }
	else if(getarg(1) == 2) { set @n, 113; set @k, 7; }
	else if(getarg(1) == 3) { set @n, 137; set @k, 12; }
	else if(getarg(1) == 4) { set @n, 100; set @k, 0; }
	else { set @n, 75; set @k, 0; }

	if(@value >= 0 && @value < 7000)
	{
		set @calculate_min, @value-1;
		set @calculate_max, @value+1;
	}
	else
	{
		set @calculate_min, @value/10000*((90-@k)*@n);
		set @calculate_max, @value/10000*((110+@k)*@n);
	}

	if(@calculate_min > 300000000 || @calculate_max > 300000000) { message @sd, 5, "계산 오류!"; return; end; }
	if(@calculate_min > @calculate_max) { set @calculate_max, @calculate_min; }

	set @printf, rand(@calculate_min, @calculate_max);
	return @printf;
}

//callfunc "calc_multiply", value;
FUNC_calc_multiply	{
	set @sd, mysd();
	set @id, myid();
	set @multiply, 100;

	if( !strcmp(get_equipitem_name(@sd, 3), "매의투구") )
	{
		effect @id, 130;
		set @multiply, @multiply/100*115;
	}

	if( !strcmp(get_equipitem_name(@sd, 6), "혁명의인장") || !strcmp(get_equipitem_name(@sd, 7), "혁명의인장") )
	{
		set @rand, rand(1, 100);
		if(@rand <= 25)
		{
			set @multiply, @multiply*2;
		}
	}
//버프리스트
	//전사
	if(get_job(@sd) == 1)
	{
		if(spellaethered(@sd, name2spellid("광폭태세")) == 1) { set @multiply, @multiply/2*3; }
		else if(spellaethered(@sd, name2spellid("진백호령")) == 1) { set @multiply, @multiply/2*3; }

		if(spellaethered(@sd, name2spellid("바람의춤")) == 1) { set @multiply, @multiply/2*3; }
		if(spellaethered(@sd, name2spellid("분노")) == 1) { set @multiply, @multiply/2*3; }
	}
	else if(get_job(@sd) == 2)
	{
		if(spellaethered(@sd, name2spellid("폭주")) == 1) { set @multiply, @multiply*2; }
		else if(spellaethered(@sd, name2spellid("맹독폭주")) == 1) { set @multiply, @multiply*2; }

		if(spellaethered(@sd, name2spellid("살인충동")) == 1) { set @multiply, @multiply/100*125; }
	}
	else if(get_job(@sd) == 3)
	{
		if(spellaethered(@sd, name2spellid("타락화")) == 1) { set @multiply, @multiply*2; }
	}

	//공통
	if(spellaethered(@sd, name2spellid("파력무참진")) == 1) { set @multiply, @multiply*2; }
	if(spellaethered(@sd, name2spellid("용기")) == 1) { set @multiply, @multiply/100*120; }
	if(spellaethered(@sd, name2spellid("막대귤사탕")) == 1) { set @multiply, @multiply/100*120; }

	//저주
	if(spellaethered(@sd, name2spellid("죽음의손짓")) == 1) { set @multiply, @multiply/100*10; }

//버프리스트끝
	set @calc, getarg(0)/100*@multiply;
	return @calc;
}
