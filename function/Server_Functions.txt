	{
		set @ti, node2id(@node);

		{
			if(istype(@ti) == 1)

				set @ts, id2sd(@ti);
				spellalluncast @ts;

		}
		set @node, get_objnextnode(@node);

FUNC_ALLIED	{
	set .@sd, mysd();
	set .@id, myid();
	set .@mapname$, get_mapname(.@sd);
	set .@ts, getarg(0);

	if(!strcmp(.@mapname$, "선필장") || !strcmp(.@mapname$, "쇄초장")) end; // 선필장,쇄초장시 기본마법 사용불가
	if(strpos(.@mapname$,"무한") != -1) return 0;
	if(ismapoption(.@sd,6))
	{
		if(clan_exist(.@sd) != 0) //PK에서 자신이 클랜일때
			if(isallied(.@sd,.@ts) == 1) return 1; //상대방이 같은편이면 리턴 1
		return 0; //클랜이 존재하지 않거나 같은편이 아니면 리턴 0
	}
	else if(ismapoption(.@sd,5))
	{
		if(clan_exist(.@sd) != 0) //사냥터에서 자신이 클랜일때
			if(isallied(.@sd,.@ts) == 1) return 1; //상대방이 같은편이면 리턴 1
			else return 0; //아니면 리턴 0
		else
			return 1; //클랜이 없을시 리턴 1
	}
	return 1;
}
