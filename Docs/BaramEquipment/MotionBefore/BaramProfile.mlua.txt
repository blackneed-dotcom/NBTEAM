    @ExecSpace("ClientOnly")
    method void OnBeginPlay()
        -- Wire page navigation and the original stat-box arrow hit areas.
        if not isvalid(self.Window) then log_error("[BaramProfile] missing UI binding") return end
        self.Window.Enable = false
        self._T.links = {}
        for _, name in ipairs({"Previous", "Next"}) do
            local button = self.Window:GetChildByName(name)
            local page = name == "Previous" and 0 or 1
            local handle = button:ConnectEvent(UITouchDownEvent, function(event) self:SetPage(page) end)
            table.insert(self._T.links, {entity = button, handle = handle})
        end
        -- ESC is owned by BaramInput, alongside S; do not close the pane in a second listener.
        for _, name in ipairs({"StatsUp", "StatsDown"}) do
            local button = self.Window:GetChildByName("Information"):GetChildByName(name)
            local direction = name == "StatsUp" and -1 or 1
            local handle = button:ConnectEvent(UITouchDownEvent, function(event) self:ScrollStats(direction) end)
            table.insert(self._T.links, {entity = button, handle = handle})
        end
        log("[BaramProfile] ready")
        for _, name in ipairs({"ExchangeDisabled", "GroupDisabled", "CombatToggle"}) do
            local button = self.Window:GetChildByName("Information"):GetChildByName(name)
            local action = name
            local handle = button:ConnectEvent(UITouchDownEvent, function(event) self:ProfileAction(action) end)
            table.insert(self._T.links, {entity = button, handle = handle})
        end
    end

    @ExecSpace("ClientOnly")
    method boolean IsOpen()
        return self.SelectedUser ~= ""
    end

    @ExecSpace("ClientOnly")
    method void Select(Entity target)
        -- Start each newly selected profile at the first stat row.
        local localPlayer = _UserService.LocalPlayer
        if not _BaramEntryFlow.InWorld or _BaramChatRelay:IsInputBlocked() then return end
        if not isvalid(target) or not isvalid(localPlayer) then return end
        if target.CurrentMap ~= localPlayer.CurrentMap or target.PlayerComponent == nil then return end
        local userId = target.PlayerComponent.UserId
        if userId == nil or userId == "" then return end
        if self.SelectedUser == userId then return end
        self:Close()
        self.SelectedUser = userId
        self._T.target = target
        self.Page = 0
        self.StatScroll = 0
        self.ShowCombatStats = false
        self:ScrollStats(0)
        self.Elapsed = 0
        _BaramInput:Close()
        local ownInfo = _EntityService:GetEntityByPath("/ui/BaramHUD/Frame/CharacterInfo")
        if isvalid(ownInfo) then ownInfo.Enable = false end
        self:Request(userId, self.Serial)
        log("[BaramProfile] selected " .. userId)
    end

    @ExecSpace("Server")
    method void Request(string userId, integer serial)
        -- Show per-instance equipment names and the same weapon bonus used by combat.
        local caller = senderUserId
        local viewer = _UserService:GetUserEntityByUserId(caller)
        -- A queued request may arrive after the requesting player disconnected.
        if not isvalid(viewer) then return end
        local target = _UserService:GetUserEntityByUserId(userId)
        local a = _BaramWorld:GetRecord(caller)
        local b = _BaramWorld:GetRecord(userId)
        if a == nil or b == nil or not a.active or not b.active or not isvalid(viewer) or not isvalid(target) then
            self:Receive(userId, serial, "", caller)
            return
        end
        if viewer.CurrentMap ~= target.CurrentMap then self:Receive(userId, serial, "", caller) return end
        ---@type BaramPlayerVisual
        local visual = target:GetComponent("script.BaramPlayerVisual")
        if visual == nil then self:Receive(userId, serial, "", caller) return end
        local equipped = b.state.items[tostring(b.state.equip)]
        local weaponData = equipped ~= nil and _BaramWorldData:Item(equipped.id) or nil
        -- Only fields drawn by winbaram 0x4D72A0. No private HP/currency/inventory.
        -- Clan/title/history remain empty; class uses the shared saved job metadata.
        local data = {name = target.PlayerComponent.Nickname, gender = visual.Gender, hair = visual.Hair, item = equipped ~= nil and equipped.id or 0, weaponName = equipped ~= nil and (equipped.name or weaponData.name) or "", exchange = b.allowExchange ~= false, group = b.allowGroup ~= false}
        data.jobId = b.state.jobId or 0
        for i = 1, 2 do
            local accessory = b.state.items[tostring(b.state["aux" .. tostring(i)] or 0)]
            data["aux" .. tostring(i)] = accessory ~= nil and accessory.id or 0
        end
        data.jobRank = b.state.jobRank or 0
        -- The same permanent attack bonus is added by BaramWorld.Attack.
        local bonuses = _BaramItemCodex:TotalStats(b.state)
        data.attack = (bonuses.attack or 0) + _BaramItemRules:WeaponAttack(b.state)
        data.defense = bonuses.defense or 0
        data.accuracy = bonuses.accuracy or 0
        data.evasion = bonuses.evasion or 0
        data.critical = bonuses.critical or 0
        self:Receive(userId, serial, _HttpService:JSONEncode(data), caller)
    end

    @ExecSpace("Client")
    method void Receive(string userId, integer serial, string raw)
        -- Refresh the selected character's profile and placeholder combat stats.
        if self.SelectedUser ~= userId or self.Serial ~= serial then return end
        if raw == "" or not self:TargetPresent() then self:Close() return end
        -- Refresh may register scripts before the updated UI tree is ready.
        -- Leave the signature untouched so the regular profile poll retries.
        if not isvalid(self.Window) then return end
        local information = self.Window:GetChildByName("Information")
        if not isvalid(information) then return end
        for _, name in ipairs({"Body", "Head", "HeldWeapon", "Auxiliary1", "Auxiliary2"}) do
            if not isvalid(information:GetChildByName(name)) then return end
        end
        if self._T.signature ~= raw then
            local data = _HttpService:JSONDecode(raw)
            self:Text("Name", data.name)
            self._T.profileData = data
            self:UpdateCombatStats(0, data.attack or 0, data.defense or 0, data.accuracy or 0, data.evasion or 0, data.critical or 0)
            self.ExchangeAllowed = data.exchange ~= false
            self.GroupAllowed = data.group ~= false
            self:RefreshList()
            self:RefreshButtons()
            self:Text("Class", _BaramJobs:GetName(data.jobId or 0, data.jobRank or 0))
            self:Text("Title", "")
            self:Text("Clan", "")
            self:Text("ClanTitle", "")
            local content = self.Window:GetChildByName("Information")
            local weaponArt = _BaramAppearance:GetWeapon(data.item, 6)
            local pose = weaponArt ~= "" and 18 or 6
            local body = content:GetChildByName("Body")
            -- Motion frames are 96x96 (foot 48,72); idle frames are 48x72 (foot 24,60).
            -- Both align to the preview's fixed foot position (96,172).
            body.UITransformComponent.RectSize = weaponArt ~= "" and Vector2(96, 96) or Vector2(48, 72)
            body.UITransformComponent.anchoredPosition = weaponArt ~= "" and Vector2(48, -100) or Vector2(72, -112)
            body.SpriteGUIRendererComponent.ImageRUID = DataRef(_BaramAppearance:GetBody(data.gender, pose))
            content:GetChildByName("Head").SpriteGUIRendererComponent.ImageRUID = DataRef(_BaramAppearance:GetHead(data.hair, 6))
            local weapon = content:GetChildByName("HeldWeapon")
            weapon.Enable = weaponArt ~= ""
            if weapon.Enable then weapon.SpriteGUIRendererComponent.ImageRUID = DataRef(weaponArt) end
            for i = 1, 2 do
                local icon = content:GetChildByName("Auxiliary" .. tostring(i))
                local art = _BaramInput.InventoryImages[tostring(data["aux" .. tostring(i)] or 0)]
                icon.Enable = art ~= nil
                if art ~= nil then icon.SpriteGUIRendererComponent.ImageRUID = DataRef(art.ruid) end
            end
            self._T.signature = raw
            log("[BaramProfile] data name=" .. data.name .. " equipment=" .. tostring(data.item))
        end
        self:SetPage(self.Page)
        self.Window.Enable = true
    end

    @ExecSpace("ClientOnly")
    method void ApplyJob(string userId, integer jobId, integer jobRank)
        -- Update only the selected public profile; its other fields stay untouched.
        if self.SelectedUser ~= userId or not self:TargetPresent() then return end
        self:Text("Class", _BaramJobs:GetName(jobId, jobRank))
        if self._T.profileData ~= nil then
            self._T.profileData.jobId = jobId
            self._T.profileData.jobRank = jobRank
        end
        self._T.signature = nil
    end

    @ExecSpace("ClientOnly")
    method void UpdateCombatStats(number combatPower, number attack, number defense, number accuracy, number evasion, number critical)
        -- Attack is supplied by the server; unimplemented stats remain zero.
        self._T.combatValues = {combatPower, attack, defense, accuracy, evasion, critical}
        if not self.ShowCombatStats then return end
        self:Text("WeaponText", "전투력 : " .. tostring(combatPower))
        self:Text("StatAttack", "공격력 : " .. tostring(attack))
        self:Text("StatDefense", "방어력 : " .. tostring(defense))
        self:Text("StatAccuracy", "명중 : " .. tostring(accuracy))
        self:Text("StatEvasion", "회피 : " .. tostring(evasion))
        self:Text("StatCritical", "치명타 : " .. tostring(critical))
        log("[BaramProfile] combat stats=" .. tostring(combatPower) .. "," .. tostring(attack) .. "," .. tostring(defense) .. "," .. tostring(accuracy) .. "," .. tostring(evasion) .. "," .. tostring(critical))
    end

    @ExecSpace("ClientOnly")
    method boolean TargetPresent()
        local target = self._T.target
        local viewer = _UserService.LocalPlayer
        return _BaramEntryFlow.InWorld and isvalid(target) and isvalid(viewer) and isvalid(target.CurrentMap) and target.CurrentMap == viewer.CurrentMap
    end

    @ExecSpace("ClientOnly")
    method void ScrollStats(integer direction)
        -- Four full-size bitmap rows fit; move in whole rows without fractional scaling.
        if not isvalid(self.Window) or self.SelectedUser == "" or self.Page ~= 0 then return end
        local content = self.Window:GetChildByName("Information")
        local count = self.ShowCombatStats and 6 or 7
        self.StatScroll = math.max(0, math.min(count - 4, self.StatScroll + direction))
        for index, name in ipairs({"WeaponText", "StatAttack", "StatDefense", "StatAccuracy", "StatEvasion", "StatCritical", "EquipmentExtra"}) do
            local row = content:GetChildByName(name)
            if isvalid(row) then
                local slot = index - 1 - self.StatScroll
                row.Enable = index <= count and slot >= 0 and slot < 4
                row.UITransformComponent.anchoredPosition = Vector2(22, -194 - slot * 14)
            end
        end
        log("[BaramProfile] stat scroll=" .. tostring(self.StatScroll) .. " rows=" .. tostring(self.StatScroll + 1) .. "-" .. tostring(self.StatScroll + 4))
    end

    @ExecSpace("ClientOnly")
    method void RefreshList()
        -- Default to real equipment; preserve original bitmap rows and arrow navigation.
        if self.ShowCombatStats then
            local v = self._T.combatValues or {0, 0, 0, 0, 0, 0}
            self:UpdateCombatStats(v[1], v[2], v[3], v[4], v[5], v[6])
        else
            local data = self._T.profileData or {}
            self:Text("WeaponText", "무기 : " .. (data.weaponName or ""))
            self:Text("StatAttack", "갑옷 : ")
            self:Text("StatDefense", "방패 : ")
            self:Text("StatAccuracy", "왼손 : ")
            self:Text("StatEvasion", "오른손 : ")
            self:Text("StatCritical", "보조 : ")
            self:Text("EquipmentExtra", "보조 : ")
        end
        self:ScrollStats(0)
    end

    @ExecSpace("ClientOnly")
    method void RefreshButtons()
        -- Use only LOOKBTN.EPF frames extracted from the provided original client.
        local info = self.Window:GetChildByName("Information")
        info:GetChildByName("ExchangeDisabled").SpriteGUIRendererComponent.ImageRUID = DataRef(self.ExchangeAllowed and "67272cdef96f48db84b57cb8a903e4e4" or "fd6663ce5da040e3b60faf1483e4da70")
        info:GetChildByName("GroupDisabled").SpriteGUIRendererComponent.ImageRUID = DataRef(self.GroupAllowed and "273e97ac44cd4f42bac56be80eb6b5c9" or "bafbc55602264393b8c45a2663290632")
    end

    @ExecSpace("ClientOnly")
    method void ProfileAction(string action)
        -- Separate list selection from server-owned social permission settings.
        if not self:IsOpen() or self.Page ~= 0 then return end
        if action == "CombatToggle" then
            self.ShowCombatStats = not self.ShowCombatStats
            self.StatScroll = 0
            self:RefreshList()
            log("[BaramProfile] combat view=" .. tostring(self.ShowCombatStats))
            return
        end
        local player = _UserService.LocalPlayer
        if not isvalid(player) then return end
        if self.SelectedUser ~= player.PlayerComponent.UserId then
            if action == "GroupDisabled" then _BaramParty:Request(self.SelectedUser) end
            if action == "ExchangeDisabled" then _BaramTrade:Request(self.SelectedUser) end
            return
        end
        if action == "ExchangeDisabled" then _BaramWorld:ToggleExchangeAllowed()
        elseif action == "GroupDisabled" then _BaramParty:ToggleAllowed() end
    end

    @ExecSpace("Server")
    method void SetGroupAllowed(boolean allowed)
        -- Only change the authenticated caller's group permission.
        _BaramParty:SetAllowedFor(senderUserId, allowed)
    end

    @ExecSpace("ClientOnly")
    method void OnUpdate(number delta)
        -- Open once per successful world entry, after the login input guard expires.
        if not _BaramEntryFlow.InWorld then
            self._T.entryOpened = false
        elseif not self._T.entryOpened and not _BaramChatRelay:IsInputBlocked() then
            local player = _UserService.LocalPlayer
            if isvalid(player) and player:GetComponent("script.BaramPlayerVisual") ~= nil then
                self:Select(player)
                self._T.entryOpened = self.SelectedUser ~= ""
                if self._T.entryOpened then log("[BaramProfile] opened own profile on login") end
            end
        end
        if self.SelectedUser == "" then return end
        if not self:TargetPresent() then self:Close() return end
        self.Elapsed = self.Elapsed + delta
        if self.Elapsed >= 0.5 then
            self.Elapsed = 0
            self:Request(self.SelectedUser, self.Serial)
        end
    end

    @ExecSpace("ClientOnly")
    method void SetPage(integer page)
        if not isvalid(self.Window) then return end
        self.Page = page == 1 and 1 or 0
        self.Window.SpriteGUIRendererComponent.ImageRUID = DataRef(self.Page == 0 and "2bdbd74b099240488bfd24774007c8eb" or "0ae88f3ef57c492ab841874911b35e5e")
        self.Window:GetChildByName("Information").Enable = self.Page == 0
        self.Window:GetChildByName("Previous").SpriteGUIRendererComponent.ImageRUID = DataRef(self.Page == 0 and "74806d9db63f4bc1a7dc8fdbc1c193a3" or "e67ebcd71ac847ca818b7cd64bee797e")
        self.Window:GetChildByName("Next").SpriteGUIRendererComponent.ImageRUID = DataRef(self.Page == 1 and "c2da3fcf480343debca0350d61f9926f" or "922c0e04346649118a7aeaa2f0575b54")
    end

    @ExecSpace("ClientOnly")
    method void Close()
        if self.SelectedUser ~= "" then log("[BaramProfile] closed " .. self.SelectedUser) end
        self.SelectedUser = ""
        self.Serial = self.Serial + 1
        self._T.target = nil
        self._T.signature = nil
        if isvalid(self.Window) then self.Window.Enable = false end
    end

    @ExecSpace("ClientOnly")
    method void Text(string fieldName, string value)
        -- Draw profile text with the original bitmap font; stat rows are left-aligned.
        local field = self.Window:GetChildByName("Information"):GetChildByName(fieldName)
        local leftAligned = fieldName == "WeaponText" or fieldName == "EquipmentExtra" or string.sub(fieldName, 1, 4) == "Stat"
        self:Draw(field, value, leftAligned)
    end

    @ExecSpace("ClientOnly")
    method void Draw(Entity field, string value, boolean leftAligned)
        -- Render shared nickname, speech and information fields with Galmuri11.
        if not isvalid(field) then return end
        local renderer = field.TextGUIRendererComponent
        if not isvalid(renderer) then
            renderer = field:AddComponent("TextGUIRendererComponent")
            renderer.FontColor = Color.white
            local first = field:GetChildByName("Glyph1")
            if isvalid(first) then renderer.FontColor = first:GetChildByName("Atlas").SpriteGUIRendererComponent.Color end
        end
        if self._T.textFields == nil then self._T.textFields = {} end
        if self._T.textFields[field.Id] ~= field then
            for i = 1, 32 do
                local glyph = field:GetChildByName("Glyph" .. tostring(i))
                if isvalid(glyph) then glyph.Enable = false end
            end
            self._T.textFields[field.Id] = field
        end
        renderer.Font = "Galmuri11"
        renderer.FontSize = 11
        renderer.BestFit = false
        renderer.IsRichText = false
        renderer.HorizontalAlignment = leftAligned and TextHorizontalAlignmentOption.Left or TextHorizontalAlignmentOption.Right
        renderer.VerticalAlignment = TextVerticalAlignmentOption.Top
        -- Original client rows are 12px high; Truncate hides the taller font line box.
        -- Limit horizontal text explicitly while allowing Galmuri11's line box to render.
        renderer.Overflow = TextOverflowMode.Overflow
        local available = field.UITransformComponent.RectSize.x
        if available > 0 and renderer:GetPreferredWidth(value) > available then
            local fitted = ""
            for _, cp in utf8.codes(value) do
                local nextText = fitted .. utf8.char(cp)
                if renderer:GetPreferredWidth(nextText) > available then break end
                fitted = nextText
            end
            value = fitted
        end
        renderer.Text = value
    end

    @ExecSpace("ClientOnly")
    method void OnEndPlay()
        self:Close()
        for _, link in ipairs(self._T.links or {}) do
            if isvalid(link.entity) then link.entity:DisconnectEvent(UITouchDownEvent, link.handle) end
        end
        if self._T.key ~= nil then _InputService:DisconnectEvent(KeyDownEvent, self._T.key) end
    end
