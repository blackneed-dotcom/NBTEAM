    @ExecSpace("ClientOnly")
    method void OnBeginPlay()
        -- Apply the original cursor and bind the shared input controls.
        -- 16x24 sprite: move its center so the upper-left tip is the click point.
        _InputService:SetCursor("892fa471519349fea1f091bdbde0bd80", Vector2(8, -12))
        log("[BaramCursor] applied original cursor 892fa471519349fea1f091bdbde0bd80")
        self.Mode = ""
        self.Pending = ""
        self.MovePending = false
        self.ReplyWait = 0
        self.MoveWait = 0
        self._T.inventory = _EntityService:GetEntityByPath("/ui/BaramInventory")
        self._T.inventoryLinks = {}
        self._T.inventoryRows = {}
        self:ResetInventoryPointer()
        self.InventoryPage = 1
        self.ActivePane = "inventory"
        self.InventoryIcons = false
        self.InventoryTipSeconds = 0
        local inventoryButton = _EntityService:GetEntityByPath("/ui/BaramHUD/Frame/InventoryButton")
        if isvalid(inventoryButton) then
            local link = inventoryButton:ConnectEvent(UITouchDownEvent, function(event)
                if event.TouchId ~= -1 and event.TouchId <= 0 then return end
                if not _BaramEntryFlow.InWorld or _BaramChatRelay.IsEditing then return end
                self:OpenInventory()
                log("[BaramInventoryUI] opened by HUD icon")
            end)
            table.insert(self._T.inventoryLinks, {entity = inventoryButton, link = link})
        end
        local skillButton = _EntityService:GetEntityByPath("/ui/BaramHUD/Frame/SkillButton")
        if isvalid(skillButton) then
            local link = skillButton:ConnectEvent(UITouchDownEvent, function(event)
                if event.TouchId ~= -1 and event.TouchId <= 0 then return end
                if not _BaramEntryFlow.InWorld or _BaramChatRelay.IsEditing then return end
                self:OpenSkills()
                log("[BaramSkillsUI] opened by HUD icon")
            end)
            table.insert(self._T.inventoryLinks, {entity = skillButton, link = link})
        end
        if isvalid(self._T.inventory) then
            self._T.inventory.Enable = false
            self._T.inventoryWindow = self._T.inventory:GetChildByName("Window", true)
            for i = 1, 15 do
                local row = self._T.inventoryWindow:GetChildByName("Row" .. tostring(i))
                self._T.inventoryRows[i] = row
                local link = row:ConnectEvent(UITouchDownEvent, function(event)
                    if event.TouchId == -1 or event.TouchId > 0 then self:InventoryPointerDown(0, event.TouchPoint)
                    elseif event.TouchId == -3 then self:InventoryPointerDown(1, event.TouchPoint) end
                end)
                table.insert(self._T.inventoryLinks, {entity = row, link = link})
                local upLink = row:ConnectEvent(UITouchUpEvent, function(event)
                    if event.TouchId == -1 or event.TouchId > 0 then self:InventoryPointerUp(0, event.TouchPoint)
                    elseif event.TouchId == -3 then self:InventoryPointerUp(1, event.TouchPoint) end
                end)
                table.insert(self._T.inventoryLinks, {entity = row, link = upLink, up = true})
            end
            for _, buttonName in ipairs({"Previous", "Next", "ListMode", "IconMode"}) do
                local actionName = buttonName
                local button = self._T.inventoryWindow:GetChildByName(buttonName)
                local link = button:ConnectEvent(UITouchDownEvent, function(event)
                    if event.TouchId ~= -1 and event.TouchId <= 0 then return end
                    if actionName == "Previous" then self:ChangeInventoryPage(-1)
                    elseif actionName == "Next" then self:ChangeInventoryPage(1)
                    else self:SetInventoryMode(actionName == "IconMode") end
                end)
                table.insert(self._T.inventoryLinks, {entity = button, link = link})
            end
        end
        self._T.window = _EntityService:GetEntityByPath("/ui/BaramInteraction/Frame/Window")
        if isvalid(self._T.window) then
            self._T.window.Enable = false
            self._T.close = self._T.window:GetChildByName("Close")
            self.CloseLink = self._T.close:ConnectEvent(UITouchDownEvent, function() self:Close() end)
        end
        self.KeyLink = _InputService:ConnectEvent(KeyDownEvent, function(event) self:HandleKey(event.key) end)
        self._T.inventoryKeyUp = _InputService:ConnectEvent(KeyUpEvent, function(event)
            if event.key == KeyboardKey.Mouse0 then self:InventoryPointerUp(0, _InputService:GetCursorPosition())
            elseif event.key == KeyboardKey.Mouse1 then self:InventoryPointerUp(1, _InputService:GetCursorPosition()) end
        end)
        log("[BaramKeys] original STR.RES key dispatcher ready")
        log("[BaramMove] original walk: 80ms x 4, frames 0/1/0/2; smooth display ready")
    end

    @ExecSpace("ClientOnly")
    method boolean Blocked()
        -- Target selection owns the arrows until Escape releases the lock.
        return (self.TargetLocked and not self.TargetPinned) or self.Mode ~= "" or self.Pending ~= "" or _BaramTrade.SessionId ~= ""
    end

    @ExecSpace("ClientOnly")
    method void Show(string title, string body, string hint)
        -- Generic dialogs replace the inventory presentation.
        if isvalid(self._T.inventory) then self._T.inventory.Enable = false end
        local window = self._T.window
        if not isvalid(window) then return end
        window.Enable = true
        window:GetChildByName("Title").TextGUIRendererComponent.Text = title
        window:GetChildByName("Body").TextGUIRendererComponent.Text = body
        window:GetChildByName("Hint").TextGUIRendererComponent.Text = hint
    end

    @ExecSpace("ClientOnly")
    method void Close()
        -- Release both dialog layers and the shared gameplay input guard.
        self:ResetInventoryPointer()
        self.InventoryTipSeconds = 0
        local wasBlocked = self:Blocked()
        _BaramNpcServices:Close()
        self.Mode = ""
        self.Pending = ""
        self.FirstSlot = 0
        self.InventoryTipSeconds = 0
        self:SetCommandPrompt("")
        if isvalid(self._T.window) then self._T.window.Enable = false end
        if isvalid(self._T.inventory) then self._T.inventory.Enable = false end
        log("[BaramInventoryUI] closed")
        if wasBlocked then _BaramChatRelay.InputBlockSeconds = 0.15 end
    end

        self._T.canvasMap = 14

    @ExecSpace("ClientOnly")
    method string InventoryText()
        local state = _BaramWorld.LocalState
        local lines = {}
        for i = 1, _BaramWorldData.InventoryCapacity do
            local item = state.items ~= nil and state.items[tostring(i)] or nil
            if item ~= nil then
                local data = _BaramWorldData:Item(item.id)
                local label = _BaramWorldData:InventoryLetter(i) .. ": " .. data.name .. " x" .. tostring(item.count)
                if state.equip == i then label = label .. " [무장]" end
                if data.durability > 0 then label = label .. " (" .. tostring(item.durability) .. ")" end
                table.insert(lines, label)
            end
        end
        if #lines == 0 then return "가진 물건이 없습니다." end
        return table.concat(lines, "\n")
    end

    @ExecSpace("ClientOnly")

        self:ResetInventoryPointer()

        if self.ActivePane ~= "inventory" then self.InventoryPage = 1 end
        self.ActivePane = "inventory"

        if isvalid(self._T.window) then self._T.window.Enable = false end
        self.Mode = ""
        self.Pending = ""
        self.FirstSlot = 0
        self:SetCommandPrompt("")
        self._T.inventory.Enable = true
        self:RefreshInventory()
        log("[BaramInventoryUI] opened")
    end

    @ExecSpace("ClientOnly")
    method void RefreshInventory()
        -- winbaram 0x450C90: scan 52 slots, skip empty entries, draw 15 per page.
        local window = self._T.inventoryWindow
        if not isvalid(window) then return end
        if self.ActivePane == "skills" then
            self:RefreshSkills()
            return
        end
        window.SpriteGUIRendererComponent.ImageRUID = DataRef("5825d1fe60ce4816886d8103fde0f972")
        for _, target in ipairs({window, window:GetChildByName("PreviousDisabled"), window:GetChildByName("NextDisabled")}) do
            target.PixelGUIRendererComponent.Enable = false
            target.SpriteGUIRendererComponent.Enable = true
        end
        window:GetChildByName("Previous").UITransformComponent.anchoredPosition = Vector2(59, -259)
        window:GetChildByName("Next").UITransformComponent.anchoredPosition = Vector2(112, -259)
        window:GetChildByName("PreviousDisabled").UITransformComponent.anchoredPosition = Vector2(59, -259)
        window:GetChildByName("NextDisabled").UITransformComponent.anchoredPosition = Vector2(112, -259)
        window:GetChildByName("PreviousDisabled").SpriteGUIRendererComponent.ImageRUID = DataRef("80bbab6caa444c07b96ef0b6ffd7a8b9")
        window:GetChildByName("NextDisabled").SpriteGUIRendererComponent.ImageRUID = DataRef("159c443f6a824eb8aa5cdf7a9ab5ffcb")
        for _, control in ipairs({"Previous", "Next", "ListMode", "IconMode"}) do
            window:GetChildByName(control).Enable = true
        end
        local state = _BaramWorld.LocalState
        local items = state.items or {}
        local slots = {}
        for i = 1, _BaramWorldData.InventoryCapacity do
            if items[tostring(i)] ~= nil then table.insert(slots, i) end
        end
        local signature = tostring(state.equip)
        for _, slot in ipairs(slots) do
            local item = items[tostring(slot)]
            signature = signature .. "|" .. tostring(slot) .. ":" .. tostring(item.id) .. ":" .. tostring(item.count) .. ":" .. tostring(item.durability) .. ":" .. tostring(item.name or "")
        end
        if self._T.inventorySignature ~= signature then self:ResetInventoryPointer() end
        self._T.inventorySignature = signature
        self._T.inventorySlots = slots
        self._T.inventoryPages = math.max(1, math.ceil(#slots / 15))
        self.InventoryPage = math.max(1, math.min(self.InventoryPage, self._T.inventoryPages))
        window:GetChildByName("Tooltip").Enable = false
        for i = 1, 15 do
            local slot = slots[(self.InventoryPage - 1) * 15 + i]
            local row = self._T.inventoryRows[i]
            row.Enable = slot ~= nil
            if slot ~= nil then
                local item = items[tostring(slot)]
                local data = _BaramWorldData:Item(item.id)
                local letter = row:GetChildByName("Letter")
                local label = row:GetChildByName("Label")
                local icon = row:GetChildByName("Icon")
                label.UITransformComponent.anchoredPosition = Vector2(30, 0)
                label.UITransformComponent.RectSize = Vector2(100, 13)
                label.TextGUIRendererComponent.FontColor = Color.FromHexCode("#EBDF8F")
                letter.Enable = not self.InventoryIcons
                label.Enable = not self.InventoryIcons
                letter.TextGUIRendererComponent.Text = _BaramWorldData:InventoryLetter(slot) .. ": "
                local text = item.name or (data ~= nil and data.name or tostring(item.id))
                if item.count > 1 then text = text .. " (" .. tostring(item.count) .. ")" end
                label.TextGUIRendererComponent.Text = text
                local image = self.InventoryImages[tostring(item.id)]
                icon.Enable = image ~= nil
                if self.InventoryIcons then
                    row.UITransformComponent.anchoredPosition = Vector2(40 + ((i - 1) % 3) * 45, -40 - math.floor((i - 1) / 3) * 41)
                    row.UITransformComponent.RectSize = Vector2(35, 35)
                else
                    row.UITransformComponent.anchoredPosition = Vector2(30, -46 - (i - 1) * 13)
                    row.UITransformComponent.RectSize = Vector2(130, 13)
                end
                if image ~= nil then
                    local scale = self.InventoryIcons and 1 or 0.5
                    local width = math.floor(image.width * scale)
                    local height = math.floor(image.height * scale)
                    icon.SpriteGUIRendererComponent.ImageRUID = DataRef(image.ruid)
                    icon.UITransformComponent.RectSize = Vector2(width, height)
                    -- Native centering is integer-truncated; list images are half size.
                    icon.UITransformComponent.anchoredPosition = Vector2(6 - math.floor(width / 2), -(self.InventoryIcons and 26 or 6) + math.floor(height / 2))
                end
            end
        end
        window:GetChildByName("PreviousDisabled").Enable = self.InventoryPage == 1
        window:GetChildByName("NextDisabled").Enable = self.InventoryPage == self._T.inventoryPages
        window:GetChildByName("ListSelected").Enable = not self.InventoryIcons
        window:GetChildByName("IconSelected").Enable = self.InventoryIcons
        if self.InventoryTipSeconds > 0 then
            local remaining = self.InventoryTipSeconds
            self.InventoryTipSeconds = 0
            for i = 1, 15 do
                local slot = slots[(self.InventoryPage - 1) * 15 + i]
                local item = slot ~= nil and items[tostring(slot)] or nil
                if slot == self._T.inventoryTipSlot and item ~= nil and item.id == self._T.inventoryTipItem then
                    self:InspectInventoryRow(i)
                    self.InventoryTipSeconds = remaining
                    break
                end
            end
        end
        self._T.inventoryState = state
        log("[BaramInventoryUI] page=" .. tostring(self.InventoryPage) .. "/" .. tostring(self._T.inventoryPages) .. " occupied=" .. tostring(#slots) .. " icons=" .. tostring(self.InventoryIcons))
    end

    @ExecSpace("ClientOnly")
    method void InspectInventoryRow(integer rowIndex)
        -- The original 0x1C click requests a description without closing the bag.
        if not self:InventoryPointerAllowed() then return end
        if not isvalid(self._T.inventory) or not self._T.inventory.Enable then return end
        local slot = (self._T.inventorySlots or {})[(self.InventoryPage - 1) * 15 + rowIndex]
        if slot == nil then return end
        local items = _BaramWorld.LocalState.items or {}
        if items[tostring(slot)] == nil then return end
        local item = items[tostring(slot)]
        local data = _BaramWorldData:Item(item.id)
        if data == nil then return end
        local lines = {item.name or data.name}
        if item.name ~= nil then table.insert(lines, "원래 이름: " .. data.name) end
        table.insert(lines, "가격 " .. tostring(data.price) .. "전 / 최대 " .. tostring(math.max(1, data.maxCount)))
        local requirements = {}
        for _, entry in ipairs({{"힘", data.strengthRequired}, {"민첩", data.dexterityRequired}, {"지력", data.intelligenceRequired}, {"승급", data.promotionRequired}}) do
            if (entry[2] or 0) > 0 then table.insert(requirements, entry[1] .. " " .. tostring(entry[2])) end
        end
        local jobs = {"전사", "도적", "주술사", "도사"}
        if (data.jobRequired or 0) > 0 then table.insert(requirements, jobs[data.jobRequired] or tostring(data.jobRequired)) end
        if (data.genderRequired or 0) > 0 then table.insert(requirements, data.genderRequired == 1 and "남성" or "여성") end
        if #requirements > 0 then table.insert(lines, "조건: " .. table.concat(requirements, " / ")) end
        table.insert(lines, data.dropAllowed == 1 and "버리기·교환 가능" or "버리기·교환 불가")
        local repair = _BaramItemRules:RepairFee(item)
        table.insert(lines, repair >= 0 and ("현재 수리비 " .. tostring(repair) .. "전") or "수리 불가")
        table.insert(lines, "보관료 " .. tostring(data.depositCost) .. "전 / 명명비 " .. tostring(data.renameCost) .. "전")
        if data.attack ~= nil then table.insert(lines, "공격력 " .. tostring(data.attack)) end
        if data.levelRequired > 0 then table.insert(lines, "레벨 제한 " .. tostring(data.levelRequired)) end
        if data.description ~= "-" and data.description ~= "" then table.insert(lines, data.description) end
        if data.tooltip ~= "-" then
            local description, replacements = string.gsub(data.tooltip, "\\n", "\n")
            table.insert(lines, description)
        end
        if data.durability > 0 then table.insert(lines, "내구도 " .. tostring(item.durability) .. " / " .. tostring(data.durability)) end
        local tooltip = self._T.inventoryWindow:GetChildByName("Tooltip")
        local textEntity = tooltip:GetChildByName("Text")
        local renderer = textEntity.TextGUIRendererComponent
        local content = table.concat(lines, "\n")
        renderer.Text = content
        -- Original client 0x441A70/0x441DD0: text-sized blue rectangle, 5px padding.
        local width = math.max(80, math.min(300, math.ceil(renderer:GetPreferredWidth(content)) + 10))
        local height = math.max(24, math.ceil(renderer:GetPreferredHeight(content, width - 10)) + 10)
        tooltip.UITransformComponent.RectSize = Vector2(width, height)
        tooltip:GetChildByName("Fill").UITransformComponent.RectSize = Vector2(width - 2, height - 2)
        textEntity.UITransformComponent.RectSize = Vector2(width - 10, height - 10)
        tooltip.UITransformComponent.anchoredPosition = Vector2(-width - 6, -math.min(math.max(0, 288 - height), -self._T.inventoryRows[rowIndex].UITransformComponent.anchoredPosition.y))
        tooltip.Enable = true
        self.InventoryTipSeconds = 10
        self._T.inventoryTipSlot = slot
        self._T.inventoryTipItem = item.id
        log("[BaramInventoryUI] inspect slot=" .. tostring(slot) .. " letter=" .. _BaramWorldData:InventoryLetter(slot))
    end

    @ExecSpace("ClientOnly")
    method string InventoryChoices()
        -- winbaram 0x4D9140: occupied runs of four or more use first-last notation.
        local items = _BaramWorld.LocalState.items or {}
        local parts = {}
        local first = 0
        for slot = 1, _BaramWorldData.InventoryCapacity + 1 do
            local occupied = slot <= _BaramWorldData.InventoryCapacity and items[tostring(slot)] ~= nil
            if occupied and first == 0 then first = slot end
            if not occupied and first > 0 then
                if slot - first > 3 then
                    table.insert(parts, _BaramWorldData:InventoryLetter(first) .. "-" .. _BaramWorldData:InventoryLetter(slot - 1))
                else
                    for index = first, slot - 1 do table.insert(parts, _BaramWorldData:InventoryLetter(index)) end
                end
                first = 0
            end
        end
        return table.concat(parts, "")
    end

    @ExecSpace("ClientOnly")
    method integer TargetDirection(KeyboardKey key)
        -- Match the screen axes: up, right, down, left.
        if key == KeyboardKey.UpArrow then return 0 end
        if key == KeyboardKey.RightArrow then return 1 end
        if key == KeyboardKey.DownArrow then return 2 end
        if key == KeyboardKey.LeftArrow then return 3 end

    end

    @ExecSpace("ClientOnly")
    method void MoveTarget(integer direction)
        -- The first key press responds immediately, before the held-key repeat delay.
        _BaramTargetBox:Step(direction)
        self._T.targetDirection = direction
        self._T.targetRepeat = 0.22
        self._T.targetJustPressed = true
    end

    @ExecSpace("ClientOnly")
    method void PollTarget(number delta)
        -- Repeat held arrows every 90 ms without depending on the OS repeat setting.
        if not self:SelectingTarget() then self._T.targetDirection = -1 return end
        local keys = {KeyboardKey.UpArrow, KeyboardKey.RightArrow, KeyboardKey.DownArrow, KeyboardKey.LeftArrow}
        local direction = self._T.targetDirection or -1
        -- Only KeyDown starts a press. Polling must never create a second first step.
        if direction < 0 then return end
        -- A slow frame's delta includes time before KeyDown; do not spend it on repeat.
        if self._T.targetJustPressed then self._T.targetJustPressed = false return end
        if not _InputService:IsKeyPressed(keys[direction + 1]) then self._T.targetDirection = -1 return end
        self._T.targetRepeat = math.max(0, (self._T.targetRepeat or 0) - delta)
        if self._T.targetRepeat <= 0 then
            _BaramTargetBox:Step(direction)
            self._T.targetRepeat = 0.09
        end
    end

    @ExecSpace("ClientOnly")
    method void SelectSlot(integer slot)
        -- Select locally, leaving the server responsible for the actual item mutation.
        local command = self.Pending
        local items = _BaramWorld.LocalState.items or {}
        if items[tostring(slot)] == nil and not (command == "swap" and self.FirstSlot > 0) then
            self:Notice("그 칸은 비어 있습니다.")
            return
        end
        if command == "swap" and self.FirstSlot == 0 then
            self.FirstSlot = slot
            self:SetCommandPrompt("자리 바꿀 아이템[" .. _BaramWorldData:InventoryLetter(slot) .. ",a-z ?] ")
            return
        end
        if command == "inspect" then
            local item = _BaramWorld.LocalState.items[tostring(slot)]
            if item == nil then self:Notice("그 칸은 비어 있습니다.") return end
            local data = _BaramWorldData:Item(item.id)
            self:EndCommand()
            self:ShowCommandChoices()
            for index, itemSlot in ipairs(self._T.inventorySlots or {}) do
                if itemSlot == slot then
                    self.InventoryPage = math.floor((index - 1) / 15) + 1
                    self:RefreshInventory()
                    self:InspectInventoryRow((index - 1) % 15 + 1)
                    break
                end
            end
            return
        end
        if command == "sell" then _BaramWorld:Shop("sell", slot)
        elseif command == "swap" then _BaramWorld:ItemAction(command, self.FirstSlot, slot)
        else _BaramWorld:ItemAction(command, slot, 0) end
        self:EndCommand()
    end

            local keys = {}
            local function add(getter)
                local ok, value = pcall(getter)
                if ok and value ~= nil then table.insert(keys, value) end
            end
            add(function() return KeyboardKey.Equals end)
            add(function() return KeyboardKey.Equal end)
            add(function() return KeyboardKey.Plus end)
            add(function() return KeyboardKey.KeypadPlus end)
            add(function() return KeyboardKey.CastFrom("=") end)
            add(function() return KeyboardKey.CastFrom("+") end)
            self._T.bookKeys = keys
        end
        for _, value in ipairs(self._T.bookKeys) do
            if key == value then return true end
        end
        return false
    end

    @ExecSpace("ClientOnly")
    method void ResetInventoryPointer()
        -- Cancel incomplete clicks when the pane, item or input context changes.
        self._T.inventoryPress = nil
        self._T.inventoryRightDown = false
        self._T.inventoryClickSeconds = 0
    end

    @ExecSpace("ClientOnly")
    method boolean InventoryPointerAllowed()
        -- All UI and keyboard mouse events share the same gameplay/modal guard.
        return self.ActivePane == "inventory" and isvalid(self._T.inventory)
            and self._T.inventory.Enable and not _BaramChatRelay:IsInputBlocked()
    end

    @ExecSpace("ClientOnly")
    method void InventoryPointerDown(integer button, Vector2 cursor)
        -- UI and KeyDown can report the same press; only the first owns it.
        if not self:InventoryPointerAllowed() then self:ResetInventoryPointer() return end
        if button == 1 then
            if self._T.inventoryRightDown then return end
            self._T.inventoryRightDown = true
            self._T.inventoryPress = nil
            self._T.inventoryClickSeconds = 0
            self:InspectInventoryAtPosition(cursor)
            return
        end
        if self._T.inventoryRightDown or self._T.inventoryPress ~= nil then return end
        local row = self:InventoryRowAtPosition(cursor)
        local slot = row > 0 and (self._T.inventorySlots or {})[(self.InventoryPage - 1) * 15 + row] or nil
        local item = slot ~= nil and (_BaramWorld.LocalState.items or {})[tostring(slot)] or nil
        if item == nil then self:ResetInventoryPointer() return end
        self._T.inventoryPress = {slot = slot, id = item.id, page = self.InventoryPage,
            icons = self.InventoryIcons, x = cursor.x, y = cursor.y}
    end

    @ExecSpace("ClientOnly")
    method void InventoryPointerUp(integer button, Vector2 cursor)
        -- Count a completed click once, and reject releases on another item or a drag.
        if button == 1 then self._T.inventoryRightDown = false return end
        local press = self._T.inventoryPress
        self._T.inventoryPress = nil
        if press == nil then return end
        if not self:InventoryPointerAllowed() then self:ResetInventoryPointer() return end
        local row = self:InventoryRowAtPosition(cursor)
        local slot = row > 0 and (self._T.inventorySlots or {})[(self.InventoryPage - 1) * 15 + row] or nil
        local item = slot ~= nil and (_BaramWorld.LocalState.items or {})[tostring(slot)] or nil
        local dx, dy = cursor.x - press.x, cursor.y - press.y
        if item == nil or slot ~= press.slot or item.id ~= press.id or self.InventoryPage ~= press.page
            or self.InventoryIcons ~= press.icons or dx * dx + dy * dy > 64 then
            self._T.inventoryClickSeconds = 0
            return
        end
        self:ClickInventoryAtPosition(cursor)
    end

    @ExecSpace("ClientOnly")
    method void InspectInventoryAtPosition(Vector2 cursor)
        -- A second completed right press on the same item closes its information.
        local row = self:InventoryRowAtPosition(cursor)
        if row <= 0 then return end
        local slot = (self._T.inventorySlots or {})[(self.InventoryPage - 1) * 15 + row]
        local item = slot ~= nil and (_BaramWorld.LocalState.items or {})[tostring(slot)] or nil
        if item == nil then return end
        if (self.InventoryTipSeconds or 0) > 0 and self._T.inventoryTipSlot == slot and self._T.inventoryTipItem == item.id then
            local tooltip = self._T.inventoryWindow:GetChildByName("Tooltip")
            if isvalid(tooltip) then tooltip.Enable = false end
            self.InventoryTipSeconds = 0
            self._T.inventoryTipSlot = nil
            self._T.inventoryTipItem = nil
            log("[BaramInventoryUI] tooltip toggled closed")
            return
        end
        self:InspectInventoryRow(row)
    end

    @ExecSpace("ClientOnly")
    method integer InventoryRowAtPosition(Vector2 cursor)
        -- Hit-test the entire rendered row, including the original UI frame scaling.
        if self.ActivePane ~= "inventory" or not isvalid(self._T.inventory) or not self._T.inventory.Enable then return 0 end
        for i = 1, 15 do
            local row = self._T.inventoryRows[i]
            if isvalid(row) and row.Enable then
                local transform = row.UITransformComponent
                local point = _UILogic:ScreenToLocalUIPosition(cursor, transform)
                -- ScreenToLocalUIPosition subtracts the pivot but retains ancestor UI scale.
                -- The original 640x480 Frame stretches by (3, 2.25); undo that for RectSize.
                local scaleX, scaleY = 1, 1
                local ancestor = row
                while isvalid(ancestor) do
                    local ancestorTransform = ancestor.UITransformComponent
                    if not isvalid(ancestorTransform) then break end
                    scaleX = scaleX * ancestorTransform.UIScale.x
                    scaleY = scaleY * ancestorTransform.UIScale.y
                    if ancestor == self._T.inventory then break end
                    ancestor = ancestor.Parent
                end
                if math.abs(scaleX) < 0.0001 or math.abs(scaleY) < 0.0001 then return 0 end
                point = Vector2(point.x / scaleX, point.y / scaleY)
                local size = transform.RectSize
                local pivot = transform.Pivot
                if point.x >= -size.x * pivot.x and point.x <= size.x * (1 - pivot.x) and point.y >= -size.y * pivot.y and point.y <= size.y * (1 - pivot.y) then
                    return i
                end
            end
        end
        return 0
    end

    @ExecSpace("ClientOnly")
    method void ClickInventoryAtPosition(Vector2 cursor)
        -- Activate only two completed clicks on the same slot, item and page.
        if not self:InventoryPointerAllowed() then self:ResetInventoryPointer() return end
        local row = self:InventoryRowAtPosition(cursor)
        local slot = row > 0 and (self._T.inventorySlots or {})[(self.InventoryPage - 1) * 15 + row] or nil
        local item = slot ~= nil and (_BaramWorld.LocalState.items or {})[tostring(slot)] or nil
        if item == nil then self._T.inventoryClickSeconds = 0 return end
        if (self._T.inventoryClickSeconds or 0) > 0 and self._T.inventoryClickSlot == slot and self._T.inventoryClickItem == item.id and self._T.inventoryClickPage == self.InventoryPage then
            self._T.inventoryClickSeconds = 0
            local data = _BaramWorldData:Item(item.id)
            if data == nil then return end
            local action = "use"
            if data.type == 0 then action = _BaramWorld.LocalState.equip == slot and "unequip" or "equip" end
            _BaramWorld:ItemAction(action, slot, 0)
            log("[BaramInventoryUI] double-click action=" .. action .. " slot=" .. tostring(slot))
        else
            self._T.inventoryClickSlot = slot
            self._T.inventoryClickItem = item.id
            self._T.inventoryClickPage = self.InventoryPage
            self._T.inventoryClickSeconds = 0.35
        end
    end

    @ExecSpace("ClientOnly")
    method void HandleKey(KeyboardKey key)
        -- F9 를 켜두면 누른 키 이름이 그대로 뜬다. 어떤 키가 안 잡히는지
        -- 짐작하지 말고 보라고 둔 것이다.
        if self.ShowTiming then self:Notice("키: " .. tostring(key)) end
        if key == KeyboardKey.Mouse1 then self:InventoryPointerDown(1, _InputService:GetCursorPosition()) return end
        if key == KeyboardKey.Mouse0 then self:InventoryPointerDown(0, _InputService:GetCursorPosition()) return end
        -- Route inventory keys only after the title and chat input guards.
        if _BaramTrade.SessionId ~= "" then _BaramTradeUI:Key(key) return end
        if not _BaramEntryFlow.InWorld or _BaramChatRelay.IsEditing then return end
        if self.Mode == "money" then _BaramMoney:Key(key) return end
        if self.Mode == "npc" then _BaramNpcServices:Key(key) return end
        local shift = _InputService:IsKeyPressed(KeyboardKey.LeftShift) or _InputService:IsKeyPressed(KeyboardKey.RightShift)
        local ctrl = _InputService:IsKeyPressed(KeyboardKey.LeftControl) or _InputService:IsKeyPressed(KeyboardKey.RightControl)
        local alt = _InputService:IsKeyPressed(KeyboardKey.LeftAlt) or _InputService:IsKeyPressed(KeyboardKey.RightAlt)
        if key == KeyboardKey.Escape then
            if self.TargetLocked or self.Pending == "target" then self:CancelTarget() return end
            if self.Mode == "command" or self.Mode == "quit" or self.Mode == "save" then self:EndCommand()
            else self:Close() _BaramProfile:Select(_UserService.LocalPlayer) end
            return
        end
        if self:SelectingTarget() then
            local direction = self:TargetDirection(key)
            if direction >= 0 then
                -- Polling may see this press before KeyDown in the same frame.
                if self._T.targetDirection ~= direction then self:MoveTarget(direction) end
                return
            end
        end
        if key == KeyboardKey.Tab and (self.TargetLocked or self.Pending == "target") and (self.Mode == "" or self.Pending == "target") then
            self:CycleTargetMode()
            return
        end
        if self.Mode == "help" and (key == KeyboardKey.LeftArrow or key == KeyboardKey.RightArrow) then self.HelpPage = 3 - self.HelpPage self:Help() return end
        if self.Mode == "quit" or self.Mode == "save" then
            if key == KeyboardKey.Y then
                local action = self.Mode
                self:EndCommand()
                if action == "quit" then _BaramWorld:LeaveToTitle() else _BaramWorld:Save() end
            elseif key == KeyboardKey.N then self:EndCommand() end
            return
        end
        if self.Mode == "shop" then
            if key == KeyboardKey.B then _BaramWorld:Shop("list", 0)
            elseif key == KeyboardKey.S then self:BeginItem("sell")
            elseif key == KeyboardKey.Alpha1 then _BaramWorld:Shop("buy", 1)
            elseif key == KeyboardKey.Alpha2 then _BaramWorld:Shop("buy", 2) end
            return
        end
        local slot = 0
        for i = 1, 26 do if key == KeyboardKey.CastFrom(string.upper(string.char(96 + i))) then slot = i break end end
        if slot > 0 and shift then slot = slot + 26 end
        if self.Pending ~= "" then
            if self.Pending == "drop" and (key == KeyboardKey.Backslash or (key == KeyboardKey.Slash and not shift) or key == KeyboardKey.Dollar or (key == KeyboardKey.Alpha4 and shift)) then
                _BaramMoney:BeginAmount()
                return
            end
            if self.Pending == "emotion" then
                if key == KeyboardKey.Question or (key == KeyboardKey.Slash and shift) then
                    self:Notice("a: 캬캬캬  b: 엉엉  c: 창피~  d: 윙크")
                    self:Notice("e: 하품  f: 쿨쿨  g: 으앗!  h: 우씨")
                    self:Notice("i: 메롱  j: 으쓱~  k: 픽~  l: 춤추기")
                    self:Notice("m: 안녕하세요?  n: 승리~~  o: 썰렁~~  p: 키스")
                else
                    local choice = slot > 26 and slot - 26 or slot
                    if choice >= 1 and choice <= 16 then
                        self:EndCommand()
                        _BaramWorld:ExpressEmotion(choice)
                    end
                end
                return
            end
            if self.Pending == "target" then
                -- Arrows are handled by the shared target navigation above.
                if key == KeyboardKey.Return or key == KeyboardKey.KeypadEnter or key == KeyboardKey.Space then
                    self:ConfirmTarget()
                end
                return
            end
            if self.Pending == "spell" then
                if key == KeyboardKey.Question or (key == KeyboardKey.Slash and shift) then
                    for index = 1, 52 do
                        local spell = self.SkillSlots[tostring(index)]
                        if spell ~= nil then
                            self:Notice(_BaramWorldData:InventoryLetter(index) .. ": " .. spell.name)
                        end
                    end
                    return
                end
                if slot > 0 then self:UseSpellSlot(slot) end
                return
            end
            if key == KeyboardKey.Question or (key == KeyboardKey.Slash and shift) then self:ShowCommandChoices() return end
            if self.Pending == "unequip" then
                if key == KeyboardKey.W and not shift then
                    local equipped = _BaramWorld.LocalState.equip or 0
                    if equipped > 0 then self:SelectSlot(equipped)
                    else self:Notice("벗을 장비가 없습니다.") self:EndCommand() end
                end
                return
            end
            if slot > 0 then self:SelectSlot(slot) end
            return
        end
        if key == KeyboardKey.F1 or key == KeyboardKey.Question or (key == KeyboardKey.Slash and shift) then self:Help() return end
        if alt then
            if key == KeyboardKey.S then self:BeginConfirmation("save")
            elseif key == KeyboardKey.X then self:BeginConfirmation("quit")
            elseif key == KeyboardKey.T then self.Transparent = not self.Transparent self:Notice(self.Transparent and "건물을 투명하게 표시합니다." or "건물 표시를 복원했습니다.") end
            return
        end
        if ctrl then
            if key == KeyboardKey.W then _BaramWorld:Users()
            elseif slot > 0 then _BaramWorld:ItemAction("use", slot, 0) end
            return
        end
        if key == KeyboardKey.F4 then self.MapScroll = not self.MapScroll self:Notice(self.MapScroll and "화면 따라가기 켬" or "화면 따라가기 끔") return end
        if key == KeyboardKey.F5 then self.HearWhispers = not self.HearWhispers _BaramChatChannels:SetListening(self.HearWhispers, self.HearShouts) self:Notice(self.HearWhispers and "귓속말듣기  : ON" or "귓속말듣기  : OFF") return end
        if key == KeyboardKey.F6 then self.HearShouts = not self.HearShouts _BaramChatChannels:SetListening(self.HearWhispers, self.HearShouts) self:Notice(self.HearShouts and "외치기듣기  : ON" or "외치기듣기  : OFF") return end
        if key == KeyboardKey.F7 then self:CanvasTest() return end
        if key == KeyboardKey.F8 then self:CanvasSwap() return end
        if key == KeyboardKey.F9 then self.ShowTiming = not self.ShowTiming self:Notice(self.ShowTiming and "응답시간 보기  : ON" or "응답시간 보기  : OFF") return end
        if key == KeyboardKey.F12 then self.FastWalk = not self.FastWalk self:Notice(self.FastWalk and "빠른 이동 켬" or "빠른 이동 끔") return end
        if self.Mode ~= "" then
            if key == KeyboardKey.I then self:Close() end
            return
        end
        if _BaramChatRelay.InputBlockSeconds > 0 then return end
        if not shift then
            local quick = self:QuickSpellSlot(key)
            if quick > 0 then self:UseSpellSlot(quick) return end
        end
        if key == KeyboardKey.Tab then self:CycleTargetMode() return end
        if isvalid(self._T.inventory) and self._T.inventory.Enable then
            if key == KeyboardKey.PageUp then self:ChangeInventoryPage(-1) return end
            if key == KeyboardKey.PageDown then self:ChangeInventoryPage(1) return end
        end
        if key == KeyboardKey.A or key == KeyboardKey.Space then
            -- 공격은 아래 OnUpdate 폴링에서 처리한다. 여기서 또 쏘면 첫 타가 두 번 나간다.
        elseif key == KeyboardKey.Colon or (key == KeyboardKey.Semicolon and shift) then
            self.Mode = "command"
            self.Pending = "emotion"
            self:SetCommandPrompt("표현할 감정 [a-p ? ] ")
        elseif key == KeyboardKey.Semicolon and not shift then _BaramWorld:Interact()
        elseif key == KeyboardKey.O then _BaramWorld:OpenDoor()
        elseif key == KeyboardKey.Comma or key == KeyboardKey.Less then _BaramWorld:Pickup(shift or key == KeyboardKey.Less)
        elseif key == KeyboardKey.I then self:OpenInventory()
        elseif key == KeyboardKey.K then self:OpenSkills()
        elseif key == KeyboardKey.S then
            _BaramProfile:Select(_UserService.LocalPlayer)
        elseif key == KeyboardKey.U then self:BeginItem("use")
        elseif key == KeyboardKey.E and not shift then self:BeginItem("eat")
        elseif key == KeyboardKey.E and shift then _BaramWorld:ToggleExchangeAllowed()
        elseif key == KeyboardKey.W then self:BeginItem(shift and "armor" or "equip")
        elseif key == KeyboardKey.T then
            if shift then
                self:BeginItem("unequip")
            else self:BeginItem("throw") end
        elseif key == KeyboardKey.D then self:BeginItem(shift and "dropall" or "drop")
        elseif key == KeyboardKey.C and not shift then self:BeginItem("swap")
        elseif key == KeyboardKey.X then self:BeginItem("inspect")
        elseif key == KeyboardKey.H then self:BeginItem(shift and "giveall" or "give")
        elseif key == KeyboardKey.G then
            if shift then _BaramParty:ToggleAllowed()
            else
                local root = _EntityService:GetEntityByPath("/ui/BaramHUD/ChatWindow")
                if isvalid(root) then
                    ---@type BaramChatHistory
                    local chat = root:GetComponent("script.BaramChatHistory")
                    if isvalid(chat) then chat:OpenGroupInput() end
                end
            end
        elseif key == KeyboardKey.Z and shift then
            -- 원본 도움말대로 갈랐다. + 는 "아는 주문"(창), Shift+Z 는 "주문 선택"(물음).
            self:BeginSpell()
        elseif self:IsSpellBookKey(key) or (key == KeyboardKey.C and shift) then
            self:OpenSkills()
        elseif key == KeyboardKey.Q then self:BeginConfirmation("quit")
        elseif key == KeyboardKey.Quote or key == KeyboardKey.DoubleQuote or key == KeyboardKey.Exclaim or (key == KeyboardKey.Alpha1 and shift) then
            local root = _EntityService:GetEntityByPath("/ui/BaramHUD/ChatWindow")
            ---@type BaramChatHistory
            local chat = root:GetComponent("script.BaramChatHistory")
            if key == KeyboardKey.Exclaim or key == KeyboardKey.Alpha1 then chat:OpenChannel("shout")
            elseif key == KeyboardKey.DoubleQuote or shift then chat:OpenChannel("whisper")
            else chat:OpenInput() end
        end
        local direction = -1
        if key == KeyboardKey.UpArrow then direction = 0 elseif key == KeyboardKey.RightArrow then direction = 1 elseif key == KeyboardKey.DownArrow then direction = 2 elseif key == KeyboardKey.LeftArrow then direction = 3 end
        if direction >= 0 then self:RequestMove(direction, shift) end
    end

    @ExecSpace("ClientOnly")
    method void RequestMove(integer direction, boolean turnOnly)
        if self.MovePending or self.MoveWait > 0 or _BaramChatRelay:IsInputBlocked() then return end
        local player = _UserService.LocalPlayer
        if not isvalid(player) then return end
        ---@type BaramPlayerVisual
        local visual = player:GetComponent("script.BaramPlayerVisual")
        if not isvalid(visual) then return end
        self.MovePending = true
        self.ReplyWait = 0
        self.PendingDirection = direction
        self._T.turnRequest = not turnOnly and visual.Facing ~= direction * 3 and self.TurnedTo ~= direction
        self._T.earlyPrediction = false
        self._T.moveSentAt = _UtilLogic.ElapsedSeconds
        if self.TraceMovement then
            visual._T.measureStart = self._T.moveSentAt
            log("[MoveTrace] request direction=" .. tostring(direction) .. " turn=" .. tostring(self._T.turnRequest))
        end
        -- 방향을 꺾으면 서버가 '돌기만' 하고 결과 2 를 준다(원작 0x4D89C0).
        -- 그때 바뀐 Facing 은 @Sync 라 한 박자 뒤에야 도착한다. 그동안 예측이
        -- 안 걸려서 꺾은 뒤 첫 걸음이 왕복 두 번을 서서 기다렸다.
        -- 서버가 이미 돌려놨다고 알려준 방향이면 Sync 를 기다리지 않고 예측한다.
        if self.FastWalk and not turnOnly
            and (visual.Facing == direction * 3 or self.TurnedTo == direction) then
            visual:PredictMove(direction)
        end
        _BaramWorld:Move(direction, turnOnly)
    end

    @ExecSpace("ClientOnly")
    method void MoveResult(integer result)
        if not self.MovePending then return end
        self.MovePending = false
        if self.TraceMovement then log("[MoveTrace] reply=" .. tostring(result) .. " elapsedMs=" .. tostring(math.floor((_UtilLogic.ElapsedSeconds - (self._T.moveSentAt or _UtilLogic.ElapsedSeconds)) * 1000))) end
        -- 한 칸이 0.32초(WalkFrameSeconds*4)인데 왕복이 그보다 길면 그만큼 걸음이 밀린다.
        -- 끊김이 지연 때문인지 아닌지 눈으로 보려고 남긴다.
        local budget = _BaramWorldData.WalkFrameSeconds * 4
        if self.ReplyWait > budget then
            local text = "걸음 응답 " .. tostring(math.floor(self.ReplyWait * 1000)) .. "ms / 한 칸 " .. tostring(math.floor(budget * 1000)) .. "ms"
            log("[BaramInput] " .. text)
            -- 화면에도 띄운다. 너무 자주 뜨면 채팅이 묻히니 1초에 한 번만.
            if self.ShowTiming and _UtilLogic.ElapsedSeconds >= (self._T.timingUntil or 0) then
                self._T.timingUntil = _UtilLogic.ElapsedSeconds + 1
                self:Notice(text)
            end
        end
        if result == 1 then
            self.TurnedTo = self.PendingDirection
            self.MoveWait = math.max(0, _BaramWorldData.WalkFrameSeconds * 4 - self.ReplyWait)
        else
            -- 2 = 방향만 바꿨다. 0 = 막혔다.
            -- 2 면 서버가 그 방향으로 돌려놓은 게 확실하니 다음 걸음은 바로 예측한다.
            self.TurnedTo = (result == 2) and self.PendingDirection or -1
            self.MoveWait = result == 2 and 0 or _BaramWorldData.InputPollSeconds
            local player = _UserService.LocalPlayer
            if isvalid(player) then
                ---@type BaramPlayerVisual
                local visual = player:GetComponent("script.BaramPlayerVisual")
                local keys = {KeyboardKey.UpArrow, KeyboardKey.RightArrow, KeyboardKey.DownArrow, KeyboardKey.LeftArrow}
                local continuing = result == 2 and self._T.earlyPrediction and _InputService:IsKeyPressed(keys[self.PendingDirection + 1])
                if isvalid(visual) and not continuing then visual:CancelPrediction() end
            end
        end
    end

    @ExecSpace("ClientOnly")
    method void OnUpdate(number delta)
        -- Target arrows stay responsive even while an earlier move awaits its reply.
        self._T.inventoryClickSeconds = math.max(0, (self._T.inventoryClickSeconds or 0) - delta)
        if not self:InventoryPointerAllowed() then self:ResetInventoryPointer() end
        self:PollTarget(delta)
        -- 접속하고 한 번만 주문 목록을 받아둔다. 안 받아두면 Shift+Z 물음에
        -- 고를 글자가 안 뜬다.
        if _BaramEntryFlow.InWorld and not self._T.spellsAsked then
            self._T.spellsAsked = true
            self:AskSpells()
        end
        -- Refresh only after a new server snapshot while the bag is visible.
        if isvalid(self._T.inventory) and self._T.inventory.Enable then
            if not _BaramEntryFlow.InWorld then self:Close()
            elseif self._T.inventoryState ~= _BaramWorld.LocalState then self:RefreshInventory() end
            if self.InventoryTipSeconds > 0 then
                self.InventoryTipSeconds = math.max(0, self.InventoryTipSeconds - delta)
                if self.InventoryTipSeconds == 0 then self._T.inventoryWindow:GetChildByName("Tooltip").Enable = false end
            end
        end
        -- 공격도 방향키처럼 눌린 상태를 매 틱 본다. 예전엔 키 down 이벤트에만 걸려 있어서
        -- 스페이스를 꾹 눌러도 한 번만 나갔다. 서버가 0.5초 쿨(r.nextAttack)을 들고 있으니
        -- 같은 간격으로 맞춰 보낸다. 걸음 응답을 기다리는 중에도 공격은 되게 위쪽에 둔다.
        self.AttackWait = math.max(0, self.AttackWait - delta)
        if self.Mode == "" and self.AttackWait <= 0 and not _BaramChatRelay:IsInputBlocked()
            and (_InputService:IsKeyPressed(KeyboardKey.Space) or _InputService:IsKeyPressed(KeyboardKey.A)) then
            self.AttackWait = 0.5
            _BaramWorld:Attack()
        end
        self.MoveWait = math.max(0, self.MoveWait - delta)
        if self.MovePending then
            self.ReplyWait = self.ReplyWait + delta
            -- Original 20ms held-key polling remains; only local presentation is speculative.
            -- The turn RPC and all server collision/cooldown checks remain authoritative.
            if self.FastWalk and self._T.turnRequest and not self._T.earlyPrediction
                and self.ReplyWait >= _BaramWorldData.InputPollSeconds and not _BaramChatRelay:IsInputBlocked()
                and not _InputService:IsKeyPressed(KeyboardKey.LeftShift) and not _InputService:IsKeyPressed(KeyboardKey.RightShift) then
                local keys = {KeyboardKey.UpArrow, KeyboardKey.RightArrow, KeyboardKey.DownArrow, KeyboardKey.LeftArrow}
                if _InputService:IsKeyPressed(keys[self.PendingDirection + 1]) then
                    local player = _UserService.LocalPlayer
                    if isvalid(player) then
                        local visual = player:GetComponent("script.BaramPlayerVisual")
                        if isvalid(visual) then visual:PredictMove(self.PendingDirection) self._T.earlyPrediction = true end
                    end
                end
            end
            if self.ReplyWait > 2 then self:MoveResult(0) end
            return
        end
        if self.MoveWait > 0 or _BaramChatRelay:IsInputBlocked() then return end
        local direction = -1
        if _InputService:IsKeyPressed(KeyboardKey.UpArrow) then direction = 0
        elseif _InputService:IsKeyPressed(KeyboardKey.RightArrow) then direction = 1
        elseif _InputService:IsKeyPressed(KeyboardKey.DownArrow) then direction = 2
        elseif _InputService:IsKeyPressed(KeyboardKey.LeftArrow) then direction = 3 end
        if direction >= 0 then
            local turn = _InputService:IsKeyPressed(KeyboardKey.LeftShift) or _InputService:IsKeyPressed(KeyboardKey.RightShift)
            self:RequestMove(direction, turn)
        end
    end

    @ExecSpace("ClientOnly")
    method void OnEndPlay()
        -- Restore the cursor and disconnect controls across repeated play sessions.
        _InputService:ResetCursor()
        for _, entry in ipairs(self._T.inventoryLinks or {}) do
            if isvalid(entry.entity) then
                if entry.up then entry.entity:DisconnectEvent(UITouchUpEvent, entry.link)
                else entry.entity:DisconnectEvent(UITouchDownEvent, entry.link) end
            end
        end
        if isvalid(self._T.inventory) then self._T.inventory.Enable = false end
        if self.KeyLink ~= nil then _InputService:DisconnectEvent(KeyDownEvent, self.KeyLink) end
        if self._T.inventoryKeyUp ~= nil then _InputService:DisconnectEvent(KeyUpEvent, self._T.inventoryKeyUp) end
        if isvalid(self._T.close) and self.CloseLink ~= nil then self._T.close:DisconnectEvent(UITouchDownEvent, self.CloseLink) end
    end
