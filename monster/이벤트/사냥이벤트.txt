@고래공격	{
	set @id, objid();
	if(#event_whale != 2)
	{
		callfunc "MONSTER_ATTACK", @id, 10, 8, -1, 0, 0, rand(250, 500), 0, "고래공격", "-1";
	}
	if(#event_whale == 1)
	{
		if(get_mobhp(@id) >= #whalehp)
		{
			set_mobhp @id, #whalehp;
		}
		else
		{
			set #whalehp, get_mobhp(@id);
		}   
	}
	if(#whalemob == 1)
	{
		deletemob @id, mapid();
	} 
}

@고래	{
	set @id, objid();

	if(#whalemob == 0)
	{
		
		set @map, mapid();
		set @x1, 0;
		set @x2, 18;
		set @y1, 10;
		set @y2, 49;
		set #whalemob, 1;
		set #event_whale, 2;

		if (@x1 < 0) { set @x1, 0; }
		if (@x2 > mapwidth() -1) { set @x2, mapwidth() -1; }
		if (@y1 < 0) { set @y1, 0; }
		if (@y2 > mapheight()-1) { set @y2, mapheight() -1; }

		for (set @i, @x1; @i <= @x2; set @i, @i+1)
		{
			for (set @j, @y1; @j <= @y2; set @j, @j+1)
			{
				set @target, obj_getmxy(@map, @i, @j); 
				if(@target > 0 && @target != @id && istype(@target) == 1)
				{
					set @rnd, rand(1,100);
					if(@rnd <= 80) { set @item$, "선물상자"; }
					else if(@rnd <= 10) { set @item$, "고급선물상자"; }
					else if(@rnd <= 10) { set @item$, "고래상자"; }

					dropitem @i, @j, name2itemid(@item$), 1;
				}
			}
		}
		timecount_all 2, 0;
		broadcast 5, "[이벤트] 고래이벤트가 종료되었습니다."; 
	}
	end;	
}

@달이동	{
	set @id, objid();

	set $moon_move, $moon_move + 1;
	if($moon_move < 2) end;
	set $moon_move, 0;

	set @x, get_mobx(@id);

	for(set @i, 0; @i < 15; set @i, @i+1)
	{
		set @target_id, obj_getxy(@x, @i);
		if(@target_id > 0)
		{
			if(istype(@target_id) == 1)
			{
				set @target_sd, id2sd(@target_id);
				if(get_equipitem_name(@target_sd, 1) == "남자설빔" || get_equipitem_name(@target_sd, 1) == "여자설빔")
				{
					spellcast @target_sd, name2spellid("개비획득");
				}
			}
		}
	}

	move @id, 1;
}
