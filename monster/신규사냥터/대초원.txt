@초원진인	{
	set @id, objid();
	set @target_id, get_mobtarget(@id);
	set @rand, rand(1, 10);
	if(@rand <= 2)
	{
	set @x1, get_x(@id)-3;
	set @x2, get_x(@id)+3;
	set @y1, get_y(@id)-3;
	set @y2, get_y(@id)+3;
		if (@x1 < 0) { set @x1, 0; }
		if (@x2 > mapwidth()-1) { set @x2, mapwidth()-1; }
		if (@y1 < 0) { set @y1, 0; }
		if (@y2 > mapheight()-1) { set @y2, mapheight()-1; }

		for (set @i, @x1; @i <= @x2; set @i, @i+1)
		{
			for (set @j, @y1; @j<=@y2; set @j, @j+1)
			{
				set @target_id, obj_getxy(@i, @j);
				if(@target_id > 0)
				{
					if(istype(@target_id) == 1)
					{
						set @target_sd, id2sd(@target_id);
						if(spellaethered(@target_sd, name2spellid("파력무참진")) == 0)
						{
							effect @id, 115;
							effect @target_id, 107;
							action @id, 6, 30, 0;
							sound @id, 108;
							spellaether @target_sd, name2spellid("파력무참진"), 4;
						}
					}
					else if(istype(@target_id) == 2)
					{
						if(mobspellaethered(@target_id, name2spellid("파력무참진")) == 0)
						{
							effect @id, 115;
							effect @target_id, 107;
							action @id, 6, 30, 0;
							sound @id, 108;
							mobspellaether @target_id, name2spellid("파력무참진"), 4;
						}
					}
				}
			}
		}
	}
}

@초원도사	{
	set @id, objid();
	set @rand, rand(1, 20);
	if(@rand <= 1)
	{
	set @x1, get_x(@id)-13;
	set @x2, get_x(@id)+13;
	set @y1, get_y(@id)-13;
	set @y2, get_y(@id)+13;
		if (@x1 < 0) { set @x1, 0; }
		if (@x2 > mapwidth()-1) { set @x2, mapwidth()-1; }
		if (@y1 < 0) { set @y1, 0; }
		if (@y2 > mapheight()-1) { set @y2, mapheight()-1; }

		for (set @i, @x1; @i <= @x2; set @i, @i+1)
		{
			for (set @j, @y1; @j<=@y2; set @j, @j+1)
			{
				set @target_id, obj_getxy(@i, @j);
				if(@target_id > 0)
				{
					if(istype(@target_id) == 2)
					{
						if(get_mobhp(@target_id) <= 2500000)
						{
							effect @id, 63;
							effect @target_id, 63; //신령이펙트
							sound @id, 97; //신령효과음
							set_mobhp @target_id, get_mobhp(@target_id) + 1000000;
							showgage @target_id; //체력바 표시
						}
					}
				}
			}
		}
	}
	else if(@rand <= 1)
	{
		set @target_id, get_mobtarget(@id);
		set @target_sd, id2sd(@target_id);
		if(@target_sd > 0)
		{
			if(@rate <= 1 && spellaethered(@target_sd, name2spellid("혼마술")) == 0 && spellaethered(@target_sd, name2spellid("저주")) == 0)
			{
				effect @target_id, 39;
				sound @target_id, 79;
				set_ac @target_sd, get_ac(@target_sd)+50;
				spellaether @target_sd, name2spellid("혼마술"), 60;
				message @target_sd, 3, get_mobname(@id) + "이(가) 혼마술을 가합니다";
			}
		}
	}
	else if(@rand <= 3)
	{
	set @me_x, get_x(@id);
	set @me_y, get_y(@id);
	set @x1, @me_x-9;
	set @x2, @me_x+9;
	set @y1, @me_y-9;
	set @y2, @me_y+9;

	if(@x1 < 0) { set @x1, 0; }
	if(@x2 > mapwidth() -1) { set @x2, mapwidth() -1; }
	if(@y1 < 0) { set @y1, 0; }
	if(@y2 > mapheight()-1) { set @y2, mapheight() -1; }

	for(set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for(set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target, obj_getxy(@i, @j);
			
			if( validid(@target) && @target != @id)
			{
				set @chk_x, callfunc("maxtomincalc", @me_x, @i);
				set @chk_y, callfunc("maxtomincalc", @me_y, @j);
				
				if( @chk_x >= @chk_y )
				{
					set @ret, @chk_x;
				} 
				else 
				{
					set @ret, @chk_y;
				}

				if(istype(@target) == 1)
				{
					set @target_sd, id2sd(@target);
					if( spellaethered(@target_sd, "호체주술") )
					{
						sound @target, 96;
						effect @target, 98;
						message @sd, 3, "마법 보호!!";
						end;
					}

					if(istype(@target) == 1)
					{
						if(spellaethered(@target_sd, name2spellid("귀염추혼소")) ||
						   spellaethered(@target_sd, name2spellid("저주")) ||
						   spellaethered(@target_sd, name2spellid("염화")) ||
						   spellaethered(@target_sd, name2spellid("혼마술"))) 
						{
							continue;
						}

						effect @target, 39;
						set_ac @target_sd, get_ac(@target_sd) + 50;
						spellaether @target_sd, name2spellid("귀염추혼소"), 30;
					}
				}
			}
		}
	}
	if(rand(1, 10) == 2)
	{
			set @target_id, get_mobtarget(@id);
			set @target_sd, id2sd(@target_id);
			if(@target_sd > 0)
			{
				if(!spellaethered(@target_sd, name2spellid("절망")))
				{
					if(get_equipitem_name(@target_sd, 6) == "시력회복의귀걸이" || get_equipitem_name(@target_sd, 7) == "시력회복의귀걸이" || spellaethered(@target_sd, name2spellid("정화의눈")) == 1) { end; }

					set_siya @target_sd, 1, 0;
					effect @target_id, 14;
					sound @target_id, 71;
					spellaether @target_sd, name2spellid("절망"), 30;
				}
			}
		}
	}
}

@혈마동전사공격	{
	set @id, myid();
	set @side, get_side(@id);
	set @rand, rand(1, 2);
	if(@rand == 1)
	{
	for(set @i, 1; @i < 5; set @i, @i+1)
	{
		switch(@side)
		{
		case 0:
			set @x, get_x(@id);
			set @y, get_y(@id)-@i;
			break;
		case 1:
			set @x, get_x(@id)+@i;
			set @y, get_y(@id);
			break;
		case 2:
			set @x, get_x(@id);
			set @y, get_y(@id)+@i;
			break;
		case 3:
			set @x, get_x(@id)-@i;
			set @y, get_y(@id);
			break;
		}

		set @target_id, obj_getxy(@x, @y);
		if((@x >= mapwidth() || @x < 0) || (@y >= mapheight() || @y < 0 )) { break; }
		else if(@i != 4 && canmove(@id, @x, @y)) { break; };
		
		if(@target_id > 0)
		{
			if(@i != 4) { break; }
			else
			{
				if(istype(@target_id) != 3)
				{
					switch(@side)
					{
					case 0:
						set_y @target_id, get_y(@id)-1;
						set_side @target_id, 2; set @side, 2;
						break;
					case 1:
						set_x @target_id, get_x(@id)+1;
						set_side @target_id, 3; set @side, 3;
						break;
					case 2:
						set_y @target_id, get_y(@id)+1;
						set_side @target_id, 0; set @side, 0;
						break;
					case 3:
						set_x @target_id, get_x(@id)-1;
						set_side @target_id, 1; set @side, 1;
						break;
					}
					if(istype(@target_id) == 1)
					{
						set @target_sd, id2sd(@target_id);
//						updatexy @target_sd;
//						updatelookpc @target_sd, 0;
						updateallpc @target_sd;
						say @id, 2, get_mobname(@id) + ": 초혼비무~~!~!";
					}
					else
					{
						move @target_id, @side;
					}
					callfunc "cho_MONSTER_RANGE_ATTACK", objid(), 10, 2, -1, 99, 105, rand(150000, 200000), 0, "초혼비무~~!~!", 16, "-1";
				}
			}
		}
	}
}
	else if(@rand == 2)
