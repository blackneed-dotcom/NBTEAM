@건룡공격	{
	set @id, objid();

	if($revival != $healhp) { set $healhp, $revival; set_mobhp @id, get_mobmaxhp(@id)/10*3; }

	if(mobspellaethered(@id, name2spellid("절망")) == 1) { mobspellaether @id, name2spellid("절망"), 5; }

	if(rand(1, 10) <= 5) end;

	set @x1, 0;
	set @x2, mapwidth()-1;
	set @y1, 0;
	set @y2, mapheight()-1;

	for(set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for(set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target_id, obj_getxy(@i, @j);
			if(@target_id > 0)
			{
				if(istype(@target_id) == 1)
				{
					set @target_sd, id2sd(@target_id);

					if(get_state(@target_sd) == 1) { continue; }

					sound @target_id, 506;
					effect @target_id, 141;

					damage @id, @target_id, rand(1000000, 1200000), 2;
					message @target_sd, 3, "건룡이 용의분노를 가합니다.";
				}
			}
		}
	}
}
