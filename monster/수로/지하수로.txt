@거륵주독	{
	set @id, objid();
	set @target_id, get_mobtarget(@id);
	if(rand(1, 10) <= 1)
	{
		if(@target_id > 0)
		{
			if(istype(@target_id) == 1)
			{
				set @target_sd, id2sd(@target_id);
				say @id, 0, get_mobname(@id) + ": 그르르..";
				if(spellaethered(@target_sd, name2spellid("거륵주독")) == 0 && spellaethered(@target_sd, name2spellid("록란주")) == 0)
				{
					spellaether @target_sd, name2spellid("거륵주독"), 5;
					message @target_sd, 3, get_mobname(@id) + "이 거륵주독을 가합니다.";
				}
			}
		}
	}

}

@거륵주독지속	{
	set @sd, mysd();
	set @id, myid();

	if(spellaethered(@sd, name2spellid("거륵주독")) == 0) { spelluncast @sd, name2spellid("거륵주독"); end; }

	set @hp, get_hp(@sd)-(get_maxhp(@sd)/100*3);
	if(@hp <= 0) { set_hp @sd, 1; }
	else { set_hp @sd, @hp; }

	showgage @id;
	effect @id, 84;
}
