		if(istype(@target_id) == 2)
		{
			if(@rand <= 100)
			{
				if(mobspellaethered(@target_id, name2spellid("몬스터중독")) == 1 || mobspellaethered(@target_id, name2spellid("노도성황")) == 1) { message @sd, 3, "이미 걸려있습니다."; end; }

				effect @target_id, 84;

				message @sd, 3, "중독을 외웠습니다.";
			}
		}
	}
	if(istype(@target_id) == 1)
	{
		set @target_sd, id2sd(@target_id);
		if(spellaethered(@target_sd, name2spellid("중독")) == 0 && spellaethered(@target_sd, name2spellid("노도성황")) == 0)
		{
			if(ismapoption(@sd, 6))
			{
				spellaether @target_sd, name2spellid("노도성황"), 30;
				if(@target_id != @id) { message @sd, 3, get_name(@sd) + "님이 귀문진산격을 가합니다."; }
			}
			else
			{
				message @sd, 3, "걸리지 않습니다.";
				end;
			}
		}
		else
		{
			message @sd, 3, "이미 걸려있습니다.";
			end;
		}
	}

	{
		for (set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target, obj_getxy(@i, @j);
			if (@target > 0 && @target != @id)

				if (istype(@target) == 1)
				{
					set @target_sd, id2sd(@target);
