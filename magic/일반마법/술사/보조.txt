@마기지체지속	{
	set @sd, mysd();
	set @id, myid();
	set @x, get_x(@id);
	set @y, get_y(@id);

	setarray @objid[0], obj_getxy(@x-1,@y), obj_getxy(@x+1,@y), obj_getxy(@x,@y-1), obj_getxy(@x,@y+1);

	for(set @i, 0; @i < 4; set @i, @i+1)
	{
		if(validid(@objid[@i]) == 1)
		{
			if(istype(@objid[@i]) == 1)
			{ // 사람
				set @target_id, @objid[@i];
				set @target_sd, id2sd(@target_id);
				if(istype(@target_id) == 1 && ismapoption(@sd,6) == 1)
				{
					if(spellaethered(@target_sd, name2spellid("마기지체저주")) == 1)
					{
						end;
					}
					if(spellaethered(@target_sd, name2spellid("호체주술")) == 1) { message @sd, 3, "마기지체로 부터 보호되어 있습니다."; end; }
					spellaether @target_sd, name2spellid("마기지체저주"), 60;
					effect @target_id, 108;
					set_ac @target_sd, get_ac(@target_sd)+50;
					if(@target_id != @id)
					{
						message @target_sd, 3, get_name(@sd) + "님이 마기지체를 걸었습니다.";
					}
				}
			}
			else if(istype(@objid[@i]) == 2)
			{ // 몹
				set @target_id, @objid[@i];
				if(!mobspellaethered(@target_id, name2spellid("몹마기지체저주")))

@만파지독	{
	set @sd, mysd();
	set @id, myid();
	set @pk, ismapoption(@sd, 6);
	
	callfunc "MANA_DELAY", 600, 0, @sd;
	spelldelay @sd, name2spellid("만파지독"), 60;

	set @x1, get_x(@id)-8;
	set @x2, get_x(@id)+8;
	set @y1, get_y(@id)-8;
	set @y2, get_y(@id)+8;
	
	if (@x1 < 0) { set @x1, 0; }
	if (@x2 > get_mapwidth(@sd) -1) { set @x2, get_mapwidth(@sd) -1; }
	if (@y1 < 0) { set @y1, 0; }
	if (@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd) -1; }
	
	for (set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for (set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target_id, obj_getxy(@i, @j);
			if (@target_id > 0 && @target_id != @id)
			{
				if(istype(@target_id) == 1)
				{
					set @target_sd, id2sd(@target_id);
					if(@target_id == @id || !@pk) { continue; }
					else if(@pk) { continue; }
					if(spellaethered(@target_sd, name2spellid("만파지독")) == 1) { continue; }
					else if(spellaethered(@target_sd, name2spellid("마비")) == 1) { continue; }
				}
				else if(istype(@target_id) == 2)
				{
					if(mobspellaethered(@target_id, name2spellid("만파지독")) == 1) { continue; }
					else if(mobspellaethered(@target_id, name2spellid("마비")) == 1) { continue; }
				}
				else { continue; }
				
				callfunc "BEGINFUNC", 1, 70, 0, 0, 6, @target_id;
				if(istype(@target_id) == 1)
				{
					spellaether @target_sd, name2spellid("만파지독"), 9;
					set_siya @target_sd, 2, 0;
					if(@target_id != @id)
					{
						message @target_sd, 3, get_name(@sd) + "님이 만파지독를 걸었습니다.";
					}
				}
				else if(istype(@target_id) == 2)
				{
					mobspellaether @target_id, name2spellid("만파지독"), 9;
					set_siya @target_id, 2, 1;
				}
			}
		}
	}
	set_mp @sd, get_mp(@sd)-600;
	message @sd, 3, "만파지독를 외웠습니다.";
}

	set @sd, mysd();

@중독	{
	set @sd, mysd();
	set @id, myid();
	set @pk, ismapoption(@sd, 6);
	set @target_id, script_target(@sd);
	
	callfunc "MANA_DELAY", 30, 0, @sd;
	set_mp @sd, get_mp(@sd)-30;

	if(istype(@target_id) == 1)
	{
		set @target_sd, id2sd(@target_id);
		if(@target_sd == @sd || !@pk) { message @sd, 3, "걸리지 않습니다"; end; }
		if(spellaethered(@target_sd, name2spellid("노도성황")) == 1) { message @sd, 3, "이미 걸려있습니다."; end; }
		else if(spellaethered(@target_sd, name2spellid("중독")) == 1) { message @sd, 3, "이미 걸려있습니다."; end; }

		spellaether @target_sd, name2spellid("중독"), 30;
	}
	else if(istype(@target_id) == 2)
	{
		if(mobspellaethered(@target_id, name2spellid("노도성황")) == 1) { message @sd, 3, "이미 걸려있습니다."; end; }
		else if(mobspellaethered(@target_id, name2spellid("몬스터중독")) == 1) { message @sd, 3, "이미 걸려있습니다."; end; }

		if(1000 <= get_mobid(@target_id) && 1300 >= get_mobid(@target_id)) { message @sd, 3, "마법 보호!!"; end; }

		mobspellaether @target_id, name2spellid("몬스터중독"), 30;
	}
	else
	{
		message @sd, 3, "걸리지 않습니다";
		end;
	}

	message @sd, 3, "중독을 외웠습니다.";
	callfunc "BEGINFUNC", 14, 68, 0, 0, 6, @target_id;

	//attachtimer @sd, @id, mapid(), "@중독마법", 1000;
}

@중독지속	{
	set @sd, mysd();
	set @id, myid();

	if(get_equipitem(@sd, 6) == name2itemid("해독의귀걸이") || get_equipitem(@sd, 7) == name2itemid("해독의귀걸이"))
	{
		spelluncast @sd, name2spellid("중독");
	}
	else
	{
		effect @id, 14;
		if(get_maxhp(@sd) <= 100000)
		{
			set @check, get_hp(@sd)-((get_hp(@sd)*3)/100);
		}
		else
		{
			set @check, get_hp(@sd)-(get_hp(@sd)/500);
		}
		if(@check > 100) { set_hp @sd, @check; }
		else { set_hp @sd, get_hp(@sd); }

		showhitgage @id;
		}
	}
}
