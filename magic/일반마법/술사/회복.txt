@환기	{
	set @sd, mysd();
	set @id, myid();

	callfunc "MANA_DELAY", 300, name2spellid("환기"), @sd;
	set_mp @sd, get_mp(@sd)-300;

	spellaether @sd, name2spellid("환기"), 20;

	message @sd, 3, "환기를 외웠습니다.";
}

@환기지속	{
	set @id, myid();
	set @sd, mysd();

	if(spellaethered(@sd, name2spellid("환기")) == 0){ spelluncast @sd, name2spellid("환기"); end; }

	callfunc "BEGINFUNC", 70, 2, 0, 0, 6, 0;

	if(get_maxhp(@sd) < (get_hp(@sd) + 30000)) { set_hp @sd, get_maxhp(@sd); }
	else { set_hp @sd, get_hp(@sd) + 30000; }
}

@환기해제	{
	set @sd, mysd();
	set @id, myid();

	spelldelay @sd, name2spellid("환기"), 120;

	message @sd, 3, "환기 해제.";
}
