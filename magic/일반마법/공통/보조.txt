@자동무장	{
	set @sd, mysd();
	set @id, myid();

	if(spellaethered(@sd, name2spellid("무장")) == 0 && spellaethered(@sd, name2spellid("자동무장")) == 0)
	{
		if(get_state(@sd) == 1)
		{
			message @sd, 3, "귀신은 할 수 없습니다.";
			end;
		}

		action @id, 6, 30, 0;
		effect @id, 110;
		sound @id, 8;

		set_ac @sd, get_ac(@sd)-10;
		spellaether @sd, name2spellid("자동무장"), 60;
		message @sd, 3, "자동무장을 외웠습니다.";
	}
	else
	{
		message @sd, 3, "이미 걸려있습니다.";
		end;
	}
}
