@보상지급	{
	set @sd, mysd();
	set @id, myid();
item:
	set @time, @time +1;
	if(@time >= 3600)
	{
		for(set @i, 0; @i < 1001; set @i, @i +1)
		{
			if(countitem(name2itemid(추억윤도조각)) < @i)
			{
				break;
			}
		}
		for(set @r, 0; @r < 1001; set @r, @r +1)
		{
			if(countitem(name2itemid(십억경)) < @r)
			{
				break;
			}
		}
		set @time, 0;
		set @hour, @hour +1;
		additem @sd, name2itemid("추억윤도조각"), 1;
		for(set @a, 0; @a < 1001; set @a, @a +1)
		{
			if(countitem(name2itemid(추억윤도조각)) < @a)
			{
				break;
			}
		}
		if(@i != @a)
		{
			set @hit, @a - @i;
			for(set @loop, 0; @loop < @hit; set @loop, @loop+1)
			{
				delitem @sd, name2itemid("추억윤도조각"), 1;
			}
		}
		if(get_level(@sd) >= 99) 
		{ 
			additem @sd, name2itemid("십억경"), 1;
			for(set @b, 0; @b < 1001; set @b, @b +1)
			{
				if(countitem(name2itemid(추억윤도조각)) < @b)
				{
					break;
				}
			}
			if(@r != @b)
			{
				set @hit2, @b - @r;
				for(set @loop, 0; @loop < @hit2; set @loop, @loop+1)
				{
					delitem @sd, name2itemid("추억윤도조각"), 1;
				}
			} 
		}
		message @sd, 0, "접속 후, "+@hour+"시간이 경과하여 접속보상이 지급되었습니다.";
	}
	sleep 1000;
	goto item;
}

@십억경자동변환지속	{
	set @sd, mysd();
	set @id, myid();
	
	updatestatus @sd, 2; //상태창 업데이트
	//message @sd, 3, "적용중..";
	set @need_money, 150;
	set @my_money, get_money(@sd);
	set @exp, get_exp(@sd);// 현재 갖고있는 경험치

	if(countitem(name2itemid("십억경")) == 1000) // 십억경 갯수로 2차 체크
	{
		message @sd, 3, "십억경 한도 초과.";
		end;
	}
	// 돈복사가 우려되므로 아래에서 금전 체크 후 처리
	if(@exp < 0 && @exp < (-1147483648)){ // 중간 음수처리
		if(@my_money < @need_money) {message @sd, 3, "금전이 부족합니다."; end;}
		else {set_money @sd, @my_money - @need_money;}
		set_exp @sd, @exp + 2147483648 - 1000000000;
		additem @sd, name2itemid("십억경"), 1;
		updatestatus @sd, 2; //상태창 업데이트
	}
	else if(@exp > 0 && @exp >= 1000000000){ // 양수 처리
		if(@my_money < @need_money) {message @sd, 3, "금전이 부족합니다."; end;}
		else {set_money @sd, @my_money - @need_money;}
		set_exp @sd, @exp - 1000000000;
		additem @sd, name2itemid("십억경"), 1;
		updatestatus @sd, 2; //상태창 업데이트
	}
	else if(@exp < 0 && @exp >= (-1147483648)){ // 나머지 음수 처리
		if(@my_money < @need_money) {message @sd, 3, "금전이 부족합니다."; end;}
		else {set_money @sd, @my_money - @need_money;}
		set_exp @sd, @exp - 1000000000;
		additem @sd, name2itemid("십억경"), 1;
		updatestatus @sd, 2; //상태창 업데이트
	}
}

@출두	{
	set @sd, mysd();
	set @id, myid();

	callfunc "MANA_DELAY", 30, 0, @sd;
	set_mp @sd, get_mp(@sd)-30;

	set @target_id, name2id(script_input(@sd));
	if(@target_id > 0 && istype(@target_id) == 1)
	{
		set @target_sd, name2sd(script_input(@sd));
		if(@target_sd == @sd) { message @sd, 3, "나 자신을 출두할 수 없습니다."; end; }
		if(get_level(@sd) > get_level(@target_sd) && get_nation(@sd) == get_nation(@target_sd) && get_mapid(@sd) == get_mapid(@target_sd) && ismapoption(@sd, 5) != 1 && get_will(@sd) > get_will(@target_sd))
		{
			effect @id, 16;
			sound @id, 29;
			for(set @i, 0; @i < 4; set @i, @i+1)
			{
				set @rand, rand(0, 3);
				switch(@rand)
				{
				case 0:
					set @rand_x, 0;
					set @rand_y, -1;
					set_side @id, 2;
					break;
				case 1:
					set @rand_x, 1;
					set @rand_y, 0;
					set_side @id, 3;
					break;
				case 2:
					set @rand_x, 0;
					set @rand_y, 1;
					set_side @id, 0;
					break;
				case 3:
					set @rand_x, -1;
					set @rand_y, 0;
					set_side @id, 1;
					break;
				}
				if( !canmove(@id, get_x(@target_id)+@rand_x, get_y(@target_id)+@rand_y) ) { goto LOOP; }
			}
LOOP:
			callfunc "BEGINFUNC", 0, 0, name2spellid("출두"), 30, 0, @target_id;
			if(@i != 4) { warp @sd, get_mapid(@target_sd), get_x(@target_id)+@rand_x, get_y(@target_id)+@rand_y; }

			message @sd, 3, "출두를 외웠습니다.";
		}
		else
		{
			message @sd, 3, "마력이 미치지 않습니다.";
		}
	} 
	else
	{
		message @sd, 3, script_input(@sd) + "님은 현재 바람의나라에 없습니다.";
	}
}

	}

	{
		if(spellaethered(@sd, name2spellid("정답:O")) == 1) spelluncast @sd, name2spellid("정답:O");
		if(spellaethered(@sd, name2spellid("정답:X")) == 1) spelluncast @sd, name2spellid("정답:X");
		if(spellaethered(@sd, name2spellid("OX퀴즈장")) == 1) spelluncast @sd, name2spellid("OX퀴즈장");
		set #ox_event_count, #ox_event_count-1;		
	}
	action @id, 6, 30, 0;
	sound @id, 33;

	set @warp_mapid, @mapid[rand(0, getarraysize(@mapid)-1)];
	set @warp_x, @x[rand(0, getarraysize(@x)-1)];
	set @warp_y, @y[rand(0, getarraysize(@y)-1)];

	deletearray @mapid[0], getarraysize(@mapid);
	deletearray @x[0], getarraysize(@x);
	deletearray @y[0], getarraysize(@y);

	warp @sd, @warp_mapid, @warp_x, @warp_y;

	message @sd, 3, "귀환을 외웠습니다.";
}
