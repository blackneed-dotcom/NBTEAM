//@초심자노리개착용	{
 set @sd, mysd(); spelluncast @sd, name2spellid("전혈");
 set_ac @sd, get_ac(@sd)-1;
 set $ii, 1;
	if(get_equipitem(@sd, 6) == name2itemid("초심자노리개") && get_equipitem(@sd, 7) == name2itemid("초심자노리개"))
	{end;}


 while($ii==1)
{
	sleep 2000;
	if(get_equipitem(@sd, 6) != name2itemid("초심자노리개") && get_equipitem(@sd, 7) != name2itemid("초심자노리개"))
	{
		set $i, 0;
	}
	if(get_equipitem(@sd, 6) == name2itemid("초심자노리개") || get_equipitem(@sd, 7) == name2itemid("초심자노리개"))
	{
		set $i, 1;
	}
	if(get_equipitem(@sd, 6) == name2itemid("초심자노리개") && get_equipitem(@sd, 7) == name2itemid("초심자노리개"))
	{
		set $i, 2;
	}
	switch($i)
	{
		case 0: 
			set $ii, 0;
			break;
		case 1: 
			if(get_hp(@sd)+(get_maxhp(@sd)/10) > get_maxhp(@sd))
			{
				set_hp @sd, get_maxhp(@sd);
				break;
			}
			else 
			{
			set_hp @sd, get_hp(@sd)+(get_maxhp(@sd)/10);
			break;
			}
		case 2: 
			if(get_hp(@sd)+((get_maxhp(@sd)/10)*2) > get_maxhp(@sd))
			{
				set_hp @sd, get_maxhp(@sd);
				break;
			}
			else 
			{
				set_hp @sd, get_hp(@sd)+((get_maxhp(@sd)/10)*2);
				break;
			}
	}
}}
