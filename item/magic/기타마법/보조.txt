@승급보상	{
	set @sd, mysd();
	set @id, myid();
	set @joblv, get_joblevel(@sd);
	set @gender, get_gender(@sd);

	switch(get_job(@sd))
	{
	case 1:
		if(@joblv == 3)
		{
			if(@gender == 1)
			{
				additem @sd, name2itemid("검황의심장"), 1;
				additem @sd, name2itemid("검황의영혼"), 1;
			}
			else
			{
				additem @sd, name2itemid("검황의영혼"), 1;
				additem @sd, name2itemid("검황의심장"), 1;
			}
		}
		else if(@joblv == 4)
		{
			additem @sd, name2itemid("검성기검"), 1;
		}
		else if(@joblv == 5)
		{
			additem @sd, name2itemid("승리의증표"), 1;
			additem @sd, name2itemid("용마제육검"), 1;
		}
		break;
	case 2:
		if(@joblv == 3)
		{
			if(@gender == 1)
			{
				additem @sd, name2itemid("귀검의심장"), 1;
				additem @sd, name2itemid("귀검의영혼"), 1;
			}
			else
			{
				additem @sd, name2itemid("귀검의영혼"), 1;
				additem @sd, name2itemid("귀검의심장"), 1;
			}
		}
		else if(@joblv == 4)
		{
			additem @sd, name2itemid("태성태도"), 1;
		}
		else if(@joblv == 5)
		{
			additem @sd, name2itemid("승리의증표"), 1;
			additem @sd, name2itemid("용천제육검"), 1;
		}
		break;
	case 3:
		if(@joblv == 3)
		{
			if(@gender == 1)
			{
				additem @sd, name2itemid("현인의심장"), 1;
				additem @sd, name2itemid("현인의영혼"), 1;
			}
			else
			{
				additem @sd, name2itemid("현인의영혼"), 1;
				additem @sd, name2itemid("현인의심장"), 1;
			}
		}
		else if(@joblv == 4)
		{
			additem @sd, name2itemid("현자금봉"), 1;
		}
		else if(@joblv == 5)
		{
			additem @sd, name2itemid("승리의증표"), 1;
			additem @sd, name2itemid("용랑제육봉"), 1;
		}
		break;
	case 4:
		if(@joblv == 3)
		{
			if(@gender == 1)
			{
				additem @sd, name2itemid("진인의심장"), 1;
				additem @sd, name2itemid("진인의영혼"), 1;
			}
			else
			{
				additem @sd, name2itemid("진인의영혼"), 1;
				additem @sd, name2itemid("진인의심장"), 1;
			}
		}
		else if(@joblv == 4)
		{
			additem @sd, name2itemid("진선역봉"), 1;
			additem @sd, name2itemid("여신의방패"), 1;
		}
		else if(@joblv == 5)
		{
			additem @sd, name2itemid("승리의증표"), 1;
			additem @sd, name2itemid("용겸제육봉"), 1;
		}
		break;
	}
}
