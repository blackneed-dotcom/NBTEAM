@만공	{
	set @sd, mysd();
	set @id, myid();
	spelldelay @sd, name2spellid("만공"), 180;

	set @grace, get_grace(@sd);
	if(get_level(@sd) >= 99 && @grace < 90) { set @bonus_rate, 10; }
	else if(@grace >= 1 && @grace < 30) { set @bonus_rate, 1; }
	else if(@grace >= 30 && @grace < 50) { set @bonus_rate, 2; }
	else if(@grace >= 50 && @grace < 80) { set @bonus_rate, 3; }
	else if(@grace >= 80 && @grace < 100) { set @bonus_rate, 4; }
	else if(@grace >= 100 && @grace < 125) { set @bonus_rate, 5; }
	else if(@grace >= 125 && @grace < 150) { set @bonus_rate, 6; }
	else if(@grace >= 150 && @grace < 170) { set @bonus_rate, 7; }
	else if(@grace >= 170 && @grace <= 200) { set @bonus_rate, 8; }
	else if(@grace >= 200 && @grace <= 220) { set @bonus_rate, 9; }
	else { set @bonus_rate, 10; }
	set @rate, rand(1,2);
	set @success, 2+@bonus_rate;
	if(@rate <= @success)
	{
		set @joblv, get_joblevel(@sd);
		set @maxhp, get_maxhp(@sd);
		if(@joblv == 0) { set @success_hp, 100; }
		else if(@joblv == 1) { set @success_hp, @maxhp/100; }
		else if(@joblv == 2) { set @success_hp, @maxhp/100*2; }
		else if(@joblv == 3) { set @success_hp, @maxhp/100*3; }
		else if(@joblv == 4) { set @success_hp, @maxhp/100*4; }
		else if(@joblv == 5) { set @success_hp, @maxhp/100*5; }
		else { set @success_hp, @maxhp/100*5; }

		effect @id, 11;
		sound @id, 65;

		if(spellaethered(@sd, name2spellid("마력의장벽")) == 1)
		{
			spelldelay @sd, name2spellid("공력증강"), 3;
			set_hp @sd, get_hp(@sd)/100*75;
		}
		else
		{
			if(get_hp(@sd) < (get_mp(@sd)/100*10)) { set_hp @sd, @success_hp; }
			//else if(get_hp(@sd)-(get_mp(@sd)/100*40) <= 100) { set_hp @sd, @success_hp; }
			else { set_hp @sd, get_hp(@sd)-((get_mp(@sd)/100)*10); } 
		}

		if(get_job(@sd) == 3)
		{
			set $protect_hp, get_hp(@sd);
		}

		set_mp @sd, get_maxmp(@sd);
		message @sd, 3, "만공을 외웠습니다..";
	}	
	else
	{
		message @sd, 3, "정신집중에 실패하였습니다.";
	}
}

@공력증강	{
	set @sd, mysd();
	set @id, myid();

	callfunc "MANA_DELAY", 30, 0, @sd;

	set @grace, get_grace(@sd);
	if(get_level(@sd) >= 99 && @grace < 90) { set @bonus_rate, 50; }
	else if(@grace >= 1 && @grace < 30) { set @bonus_rate, 50; }
	else if(@grace >= 30 && @grace < 50) { set @bonus_rate, 50; }
	else if(@grace >= 50 && @grace < 80) { set @bonus_rate, 50; }
	else if(@grace >= 80 && @grace < 100) { set @bonus_rate, 50; }
	else if(@grace >= 100 && @grace < 125) { set @bonus_rate, 50; }
	else if(@grace >= 125 && @grace < 150) { set @bonus_rate, 50; }
	else if(@grace >= 150 && @grace < 170) { set @bonus_rate, 50; }
	else if(@grace >= 170 && @grace <= 200) { set @bonus_rate, 50; }
	else if(@grace >= 200 && @grace <= 220) { set @bonus_rate, 50; }
	else { set @bonus_rate, 50; }

	set @rate, rand(1, 17);
	set @success, 5+@bonus_rate;
	if(@rate <= @success)
	{
		set @hp, get_hp(@sd);
		set @joblv, get_joblevel(@sd);
		set @maxhp, get_maxhp(@sd);
		if(@joblv == 0) { set @success_hp, 100; }
		else if(@joblv == 1) { set @success_hp, @maxhp/100; }
		else if(@joblv == 2) { set @success_hp, @maxhp/100*2; }
		else if(@joblv == 3) { set @success_hp, @maxhp/100*3; }
		else if(@joblv == 4) { set @success_hp, @maxhp/100*4; }
		else if(@joblv == 5) { set @success_hp, @maxhp/100*5; }
		else { set @success_hp, @maxhp/100*5; }

		effect @id, 11;
		sound @id, 65;

		set_mp @sd, get_maxmp(@sd);

		if(spellaethered(@sd, name2spellid("마력의장벽")) == 1)
		{
			set_hp @sd, @hp/100*75;
			spelldelay @sd, name2spellid("공력증강"), 0;
		}
		else
		{
			if(@hp < (get_mp(@sd)/100*30)) { set_hp @sd, @success_hp; }
			else { set_hp @sd, @hp-((get_mp(@sd)/100)*30); } 
		}

		if(get_job(@sd) == 3)
		{
			set $protect_hp, get_hp(@sd);
		}

		message @sd, 3, "공력증강을 외웠습니다.";
	}	
	else
	{
		message @sd, 3, "정신집중에 실패하였습니다.";
		set_mp @sd, get_mp(@sd) - 30;
	}
}

@귀염추혼소	{
	set @sd, mysd();
	set @id, myid();

	if(get_mp(@sd) < 600)
	{
		message @sd, 3, "마력이 부족합니다.";
		end;
	}

	set @me_x, get_x(@id);
	set @me_y, get_y(@id);
	set @x1, @me_x-9;
	set @x2, @me_x+9;
	set @y1, @me_y-9;
	set @y2, @me_y+9;

	if(@x1 < 0) { set @x1, 0; }
	if(@x2 > mapwidth() -1) { set @x2, mapwidth() -1; }
	if(@y1 < 0) { set @y1, 0; }
	if(@y2 > mapheight()-1) { set @y2, mapheight() -1; }

	for(set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for(set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target, obj_getxy(@i, @j);
			
			if( validid(@target) && @target != @id)
			{
				set @chk_x, callfunc("maxtomincalc", @me_x, @i);
				set @chk_y, callfunc("maxtomincalc", @me_y, @j);
				
				if( @chk_x >= @chk_y )
				{
					set @ret, @chk_x;
				} 
				else 
				{
					set @ret, @chk_y;
				}

				if(istype(@target) == 1)
				{
					set @target_sd, id2sd(@target);
					if( spellaethered(@target_sd, "호체주술") )
					{
						sound @target, 96;
						effect @target, 98;
						message @sd, 3, "마법 보호!!";
						end;
					}

					if(ismapoption(@sd, 6) == 1)
					{
						if(spellaethered(@target_sd, name2spellid("귀염추혼소")) ||
						   spellaethered(@target_sd, name2spellid("저주")) ||
						   spellaethered(@target_sd, name2spellid("혼마술"))) 
						{
							continue;
						}

						effect @target, 39;
						set_ac @target_sd, get_ac(@target_sd) + 50;
						spellaether @target_sd, name2spellid("귀염추혼소"), 20;
						message @target_sd, 3, get_name(@sd) + "님께서 귀염추혼소를 가했습니다.";
					}
				}
				else if(istype(@target) == 2)
				{
					if(mobspellaethered(@target, name2spellid("저주")) ||
					   mobspellaethered(@target, name2spellid("혼마술")) ||
					   mobspellaethered(@target, name2spellid("귀염추혼소")))
					{
						continue;
					}

					effect @target, 39;
					set_mobac @target, get_mobac(@target) + 50;
					mobspellaether @target, name2spellid("귀염추혼소"), 425;
				}
			}
		}
	}

	action @id, 6, 30, 79;
	if(get_equipitem_name(@sd, 0) == "천랑곤봉'환") { set @delay, 25; }
	else if(get_equipitem_name(@sd, 0) == "대모마령'산") { set @delay, 20; }
	else { set @delay, 20; }
	spelldelay @sd, name2spellid("귀염추혼소"), @delay;
	set_mp @sd, get_mp(@sd) - 600;
}

	set @id, myid();

	set_ac @sd, get_ac(@sd) - 50;
	message @sd, 3, "귀염추혼소 해제.";
}
