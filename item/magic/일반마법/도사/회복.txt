@백호의희원'첨	{
	set @sd, mysd();
	set @weapon$, get_equipitem_name(@sd, 0);
	
	if(party(@sd) != 0)
	{
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @target_sd, party_member(@sd, @i);
			if(@target_sd > 0)
			{
				if(get_mapid(@sd) == get_mapid(@target_sd))
				{
					set @member_count, @member_count+1;
				}
			}
		}
	}
	else { set @member_count, 1; }

	if( !strcmp(@weapon$, "용겸제삼봉") ) { set @delay, 1 + get_mp(@sd)/(16000*@member_count); }
	else if( !strcmp(@weapon$, "진선장") ) { set @delay, 1 + get_mp(@sd)/(18000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제사봉") ) { set @delay, 1 + get_mp(@sd)/(19000*@member_count); }
	else if( !strcmp(@weapon$, "귀문진산장'진") ) { set @delay, 1 + get_mp(@sd)/(19000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제오봉") ) { set @delay, 1 + get_mp(@sd)/(23000*@member_count); }
	else if( !strcmp(@weapon$, "대모홍접선'혈") ) { set @delay, 1 + get_mp(@sd)/(23000*@member_count); }
	else if( !strcmp(@weapon$, "진'적염곤봉") ) { set @delay, 1 + get_mp(@sd)/(23000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제육봉") ) { set @delay, 1 + get_mp(@sd)/(25000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제칠봉") ) { set @delay, 1 + get_mp(@sd)/(55000*@member_count); }
	else if( !strcmp(@weapon$, "[손상]용겸제칠봉") ) { set @delay, 1 + get_mp(@sd)/(55000*@member_count); }
	else if( !strcmp(@weapon$, "[이벤]용겸제칠봉") ) { set @delay, 1 + get_mp(@sd)/(35000*@member_count); }
	else if( !strcmp(@weapon$, "기간용겸제칠봉") ) { set @delay, 1 + get_mp(@sd)/(55000*@member_count); }
	else if( !strcmp(@weapon$, "귀문진산장'태") ) { set @delay, 1 + get_mp(@sd)/(35000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제팔봉") ) { set @delay, 1 + get_mp(@sd)/(85000*@member_count); }
	else if( !strcmp(@weapon$, "용겸제구봉") ) { set @delay, 1 + get_mp(@sd)/(130000*@member_count); }
	else { set @delay, 1 + get_mp(@sd)/(15000*@member_count); 
}
	if(@delay > 0) { spelldelay @sd, name2spellid("백호의희원'첨"), @delay; }
	else { spelldelay @sd, name2spellid("백호의희원'첨"), 1; }

	callfunc "CHOMHEALMAGIC", get_mp(@sd)/2, get_mp(@sd)*2, 109, 95, "백호의희원'첨";
}

@신령의기원'첨	{
	set @sd, mysd();
	set @hp, get_mp(@sd)/12;
	set @mp, get_mp(@sd)/13;
	
	if(party(@sd) != 0)
	{
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @target_sd, party_member(@sd, @i);
			if(@target_sd > 0) { set @member_count, @member_count+1; }
		}
	}
	else { set @member_count, 1; }

	set @delay, get_mp(@sd)/(30000*@member_count);

	if(@hp > 100000) { set @hp, 150000; }
	if(@mp > 50000) { set @mp, 50000; }

	if(@delay > 15) { spelldelay @sd, name2spellid("신령의기원'첨"), 15; }
	else if(@delay > 0) { spelldelay @sd, name2spellid("신령의기원'첨"), @delay; }
	else { spelldelay @sd, name2spellid("신령의기원'첨"), 1; }
	callfunc "CHOMHEALMAGIC", @mp, @hp, 63, 97, "신령의기원'첨";
}
