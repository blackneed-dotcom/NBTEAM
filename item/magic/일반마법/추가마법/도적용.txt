@극'파천검무	{
	set @id, myid();
	set @sd, mysd();
	set @move_y, 0;
	set @move_x, 0;
	set @save_hp, get_hp(@sd);
	set @damage, get_hp(@sd);
	if(get_mp(@sd) > 20000) { set @damage, get_hp(@sd) + 13000000; }

	callfunc "MANA_DELAY", 0, name2spellid("극'파천검무"), @sd;

	if($damageup) { set @damage, @damage*2; }

	spelldelay @sd, name2spellid("극'파천검무"), 30;

	set_hp @sd, @save_hp/2;

	//좌표체크
	switch(get_side(@id))
	{
	case 0:
		set @target, obj_getxy(get_x(@id), get_y(@id)-1);

		callfunc "BATTLE_DOOR_CHECK", @target;

		if(validid(@target) == 0) { goto failed; }
		for(set @loop, 1;@loop <= 4;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id), get_y(@id)-@loop);
			if(validid(@target) != 0)
			{
				set @move_y, @move_y-1;
			}
			else { set @move_y, @move_y-1; goto next; }
		}
		set @move_y, @move_y-1;	
		break;
	case 1:
		set @target, obj_getxy(get_x(@id)+1, get_y(@id));

		callfunc "BATTLE_DOOR_CHECK", @target;

		if(validid(@target) == 0) { goto failed; }
		for(set @loop, 1;@loop <= 4;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id)+@loop, get_y(@id));
			if(validid(@target) != 0)
			{
				set @move_x, @move_x+1;
			}
			else { set @move_x, @move_x+1; goto next; }		
		}
		set @move_x, @move_x+1;
		break;
	case 2:
		set @target, obj_getxy(get_x(@id), get_y(@id)+1);

		callfunc "BATTLE_DOOR_CHECK", @target;

		if(validid(@target) == 0) { goto failed; }
		for(set @loop, 1;@loop <= 4;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id), get_y(@id)+@loop);
			if(validid(@target) != 0)
			{
				set @move_y, @move_y+1;
			}
			else { set @move_y, @move_y+1; goto next; }	
		}
		set @move_y, @move_y+1;
		break;
	case 3:
		set @target, obj_getxy(get_x(@id)-1, get_y(@id));

		callfunc "BATTLE_DOOR_CHECK", @target;

		if(validid(@target) == 0) { goto failed; }
		for(set @loop, 1;@loop <= 4;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id)-@loop, get_y(@id));
			if(validid(@target) != 0)
			{
				set @move_x, @move_x-1;
			}
			else { set @move_x, @move_x-1; goto next; }		
		}
		set @move_x, @move_x-1;
		break;
	}
	next:
	if(canmove(@id, get_x(@id)+@move_x, get_y(@id)+@move_y) == 1) { goto failed; }
	if(get_x(@id)+@movex < 0 || get_x(@id)+@movex > get_mapwidth(@sd)-1) { goto failed; }
	if(get_y(@id)+@movey < 0 || get_y(@id)+@movey > get_mapheight(@sd)-1) { goto failed; }
	set_xy @id, get_x(@id)+@move_x, get_y(@id)+@move_y;
	if(@move_x < 0) set @move_x, @move_x*-1;
	if(@move_y < 0) set @move_y, @move_y*-1;
	updatexy @sd;
	updatelookpc @sd, 0;
	updateallpc @sd;
	sound @id, 501;
	//이펙트 먼저 처리하고 움직이면 뒤로 묻히기 때문에.. 재처리
	//위쪽 switch 에도 처리를 하면되나 마지막 좌표 체킹처리를 위해 어쩔수 없이 아래에 둠.
	switch(get_side(@id))
	{
	case 0:
		set_side @id, 2;
		for(set @loop, 1;@loop <= @move_y-1;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id), get_y(@id)+@loop);
			effect @target, 167;
			damage @id, @target, @damage, 0; 
		}
		break;
	case 1:
		set_side @id, 3;
		for(set @loop, 1;@loop <= @move_x+1;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id)-@loop, get_y(@id));
			effect @target, 167;
			damage @id, @target, @damage, 0; 
		}
		break;
	case 2:
		set_side @id, 0;
		for(set @loop, 1;@loop <= @move_y+1;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id), get_y(@id)-@loop);
			effect @target, 167;
			damage @id, @target, @damage, 0; 
		}
		break;
	case 3:
		set_side @id, 1;
		for(set @loop, 1;@loop <= @move_x-1;set @loop, @loop+1)
		{
			set @target, obj_getxy(get_x(@id)+@loop, get_y(@id));
			effect @target, 167;
			damage @id, @target, @damage, 0; 
		}
		break;
	}

	updatexy @sd;
	updatelookpc @sd, 0;
	updateallpc @sd;
	action @id, 1, 15, 0;
	message @sd, 3, "극'파천검무를 외웠습니다.";

failed:
	set_hp @sd, @save_hp;
	message @sd, 3, "극'파천검무 실패!";

	set @loopx1, 0;
	set @loopx2, 0;
	set @loopy1, 0;
	set @loopy2, 0;
	
	set @mob_count, 0;
	set @loop_count, 0;
	set @damage, 0;
	
	if($finalattack) { set @damage, get_hp(@sd)*0; }
	if($damageup) { set @damage, @damage*0; }

	switch(get_side(@id))
	{
	case 0:
		if(get_y(@id)-1 < 0) { goto failed1; }		
		for(set @loop, 1; @loop <= 3; set @loop, @loop+1)
		{
			if(get_y(@id)-@loop < 0) { goto next1; }
			for(set @loop2, @loopx1; @loop2 <= @loopx2; set @loop2, @loop2+1)
			{
				if(get_x(@id)+@loop2 < 0 || get_x(@id)+@loop2 > get_mapwidth(@sd)-1) { goto next0_11; }
				set @target, obj_getxy(get_x(@id)+@loop2, get_y(@id)-@loop);
				if(validid(@target) != 0)
				{
					set @mob_count, @mob_count+1;
					effect @target, 167;
					damage @id, @target, @damage, 0;
				}
				next0_11:
			}
			set @loopx1, @loopx1-1;
			set @loopx2, @loopx2+1;
		}
		break;
		
	case 1:
		if(get_x(@id)+1 > get_mapwidth(@sd)-1) { goto failed1; }
		for(set @loop, 1; @loop <= 3; set @loop, @loop+1)
		{
			if(get_x(@id)+@loop > get_mapwidth(@sd)-1) { goto next1; }
			for(set @loop2, @loopy1; @loop2 <= @loopy2; set @loop2, @loop2+1)
			{
				if(get_y(@id)+@loop2 < 0 || get_y(@id)+@loop2 > get_mapheight(@sd)-1) { goto next1_11; }
				set @target, obj_getxy(get_x(@id)+@loop, get_y(@id)+@loop2);
				if(validid(@target) != 0)
				{
					set @mob_count, @mob_count+1;
					effect @target, 167;
					damage @id, @target, 0, 0;
				}
				next1_11:
			}
			set @loopy1, @loopy1-1;
			set @loopy2, @loopy2+1;
		}
		break;
		
	case 2:
		if(get_y(@id)+1 > get_mapheight(@sd)-1) { goto failed1; }
		for(set @loop, 1; @loop <= 3; set @loop, @loop+1)
		{
			if(get_y(@id)+@loop > get_mapheight(@sd)-1) { goto next1; }
			for(set @loop2, @loopx1; @loop2 <= @loopx2; set @loop2, @loop2+1)
			{
				if(get_x(@id)+@loop2 < 0 || get_x(@id)+@loop2 > get_mapwidth(@sd)-1) { goto next2_11; }
				set @target, obj_getxy(get_x(@id)+@loop2, get_y(@id)+@loop);
				if(validid(@target) != 0)
				{
					set @mob_count, @mob_count+1;
					effect @target, 167;
					damage @id, @target, 0, 0;
				}
				next2_11:
			}
			set @loopx1, @loopx1-1;
			set @loopx2, @loopx2+1;
		}
		break;
		
	case 3:
		if(get_x(@id)-1 < 0) { goto failed1; }
		for(set @loop, 1; @loop <= 3; set @loop, @loop+1)
		{
			if(get_x(@id)-@loop < 0) { goto next1; }
			for(set @loop2, @loopy1; @loop2 <= @loopy2; set @loop2, @loop2+1)
			{
				if(get_y(@id)+@loop2 < 0 || get_y(@id)+@loop2 > get_mapheight(@sd)-1) { goto next3_11; }
				set @target, obj_getxy(get_x(@id)-@loop, get_y(@id)+@loop2);
				if(validid(@target) != 0)
				{
					set @mob_count, @mob_count+1;
					effect @target, 167;
					damage @id, @target, 0, 0;
				}
				next3_11:
			}
			set @loopy1, @loopy1-1;
			set @loopy2, @loopy2+1;
		}
		break;
	}

	next1:
	
	if(@mob_count == 0) { goto failed1; }
	else { goto success; }	

	failed1:
	message @sd, 3, "극'파천검무 실패!";
	end;
	
	success:
	set_hp @sd, get_hp(@sd)/2;
	set_mp @sd, get_mp(@sd)-50;
	action @id, 1, 30, 87;
}

@혈독주입지속	{
	set @sd, mysd();
	set @id, myid();

	if(spellaethered(@sd, name2spellid("혈독주입")) == 0) { end; }

	set @value, callfunc("calculate_damage", (get_hp(@sd)+(get_mp(@sd)/100*125))/4, 4);
	set @damage, callfunc("calc_multiply", @value);

	set @hp, get_hp(@sd)-(get_maxhp(@sd)/100*5);
	if(@hp <= 0) { set_hp @sd, 1; }
	else { set_hp @sd, @hp; }

	set @x1, get_x(@id)-2;
	set @x2, get_x(@id)+2;
	set @y1, get_y(@id)-2;
	set @y2, get_y(@id)+2;
	
	if(@x1 < 0) { set @x1, 0; }
	if(@x2 > get_mapwidth(@sd)-1) { set @x2, get_mapwidth(@sd)-1; }
	if(@y1 < 0) { set @y1, 0; }
	if(@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd)-1; }
	
	for(set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for(set @j, @y1; @j <= @y2; set @j, @j+1)
		{
			set @target_id, obj_getxy(@i, @j);
			if(@target_id > 0 && @target_id != @id)
			{
				if(istype(@target_id) == 1)
				{
					set @target_sd, id2sd(@target_id);
					if(ismapoption(@target_sd, 6) && get_state(@target_sd) != 1)
					{
						set @hp, get_hp(@target_sd)-(get_maxhp(@target_sd)/100*2);
						if(@hp <= 0) { set_hp @target_sd, 1; }
						else { set_hp @target_sd, @hp; }

						showhitgage @target_id;

						message @target_sd, 3, get_name(@sd) + "님께서 혈독주입을 가합니다.";
					}
					effect @target_id, 139;
				}
				else if(istype(@target_id) == 2)
				{
					if(get_mobid(@target_id) == 1012) || (get_mobid(@target_id) == 1013) || (get_mobid(@target_id) == 1162) || (get_mobid(@target_id) == 1242) || (get_mobid(@target_id) == 1243) || (get_mobid(@target_id) == 1244) || (get_mobid(@target_id) == 1245) || (get_mobid(@target_id) == 1254) { message @sd, 3, "마법 보호!!"; end; }
					
					if(rand(1, 10) <= 10)
					{
						mobspellaether @target_id, name2spellid("몹독"), 5;
					}
					effect @target_id, 139;

					callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
				}
			}
		}
	}
}

@끌어오기	{
	set @sd, mysd();
	set @id, myid();

	callfunc "SPELL_MOMENT", 1;
	callfunc "MANA_DELAY", 300, 0, @sd;

	set @chk, 0;
	set @me_x, get_x(@id);
	set @me_y, get_y(@id);
	set @side, get_side(@id);
	set @damage, get_hp(@sd)*3/2;
	if(spellaethered(@sd, name2spellid("파력무참진")) == 1) { set @damage, @damage*2; }
	if($runaway) { set @damage, @damage*2; }

	action @id, 1, 30, 0;
	sound @id, 42;

	set_hp @sd, get_hp(@sd)/2;
	//set_mp @sd, get_mp(@sd)-300;

	spelldelay @sd, name2spellid("끌어오기"), 60;
	message @sd, 3, "끌어오기를 외웠습니다.";

	//위치 확인하고 배열함수 주기
	switch(@side)
	{

	case 0:
		setarray @range_x[0], @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  ;
		setarray @range_y[0], @me_y-1, @me_y-2, @me_y-3, @me_y-4, @me_y-5, @me_y-6, @me_y-6, @me_y-8, @me_y-9, @me_y-10;
		break;

	case 1:
		setarray @range_x[0], @me_x+1, @me_x+2, @me_x+3, @me_x+4, @me_x+5, @me_x+6, @me_x+7, @me_x+8, @me_x+9, @me_x+10;
		setarray @range_y[0], @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  ;
		break;

	case 2:
		setarray @range_x[0], @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  ;
		setarray @range_y[0], @me_y+1, @me_y+2, @me_y+3, @me_y+4, @me_y+5, @me_y+6, @me_y+7, @me_y+8, @me_y+9, @me_y+10;
		break;

	case 3:
		setarray @range_x[0], @me_x-1, @me_x-2, @me_x-3, @me_x-4, @me_x-5, @me_x-6, @me_x-7, @me_x-8, @me_x-9, @me_x-10;
		setarray @range_y[0], @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  ;
		break;

	}

	for(set @loop, 0; @loop < 10; set @loop, @loop+1)
	{
		//발동 좌표 체크
		if(@range_x[@loop] >= 0 && @range_x[@loop] <= mapwidth(@sd) && @range_y[@loop] >= 0 && @range_y[@loop] <= mapheight(@sd))
		{
			//타겟 아이디 검색
			set @target_id, obj_getxy(@range_x[@loop], @range_y[@loop]);

			//타겟 아이디가 있고, 내가 아닐 경우
			if(@target_id > 0 && @target_id != @id)
			{
				if(istype(@target_id) == 1)
				{
					set @target_sd, id2sd(@target_id);
					effect @target_id, 103;
					damage @id, @target_id, @damage + (@damage*(@loop*3/10)), 0;
					set_hp @sd, get_hp(@sd) + @damage/3;
					if(get_hp(@sd) > get_maxhp(@sd)) { set_hp @sd, get_maxhp(@sd); }
					message @sd, 3, @damage/3 + "의 체력을 흡수!!";
					goto next;
					end;
				}
				else if(istype(@target_id) == 2)
				{
					effect @target_id, 103;
					callfunc "ATTACKDAMAGE", @target_id,  @damage + (@damage*(@loop*3/10)), 0;
					set_hp @sd, get_hp(@sd) + @damage/3;
					if(get_hp(@sd) > get_maxhp(@sd)) { set_hp @sd, get_maxhp(@sd); }
					message @sd, 3, @damage/3 + "의 체력을 흡수!!";
					goto next;
					end;
				}
			}
			else
			{
				if(@loop == 10)
				{
					deletearray @range_x[0], getarraysize(@range_x);
					deletearray @range_y[0], getarraysize(@range_y);
					end;
				}
			}
		}
		else
		{
			if(@loop == 10)
			{
				deletearray @range_x[0], getarraysize(@range_x);
				deletearray @range_y[0], getarraysize(@range_y);
				end;
			}
		}
	}
	end;

next:
	deletearray @range_x[0], getarraysize(@range_x);
	deletearray @range_y[0], getarraysize(@range_y);

	//데미지도 줬으니, 데미지를 줘도 사람/몹이 살아있으면 끌어오장!

	if(@target_id > 0)
	{
		if(istype(@target_id) == 1)
		{
			if(get_gmlevel(id2sd(@target_id)) >= 1) { end; }
		}

		//끌어오기 전에 자신의 case에 따라 확인!!
		if((@me_x >= mapwidth() || @me_x < 0) || (@me_y >= mapheight() || @me_y < 0 )) { message @sd, 3, "상대방을 끌어오진 못했습니다."; end; }
		if(@side == 0 && canmove(@id, @me_x, @me_y-1)) { message @sd, 3, "상대방을 끌어오진 못했습니다."; end; }
		else if(@side == 1 && canmove(@id, @me_x+1, @me_y)) { message @sd, 3, "상대방을 끌어오진 못했습니다."; end; }
		else if(@side == 2 && canmove(@id, @me_x, @me_y+1)) { message @sd, 3, "상대방을 끌어오진 못했습니다."; end; }
		else if(@side == 3 && canmove(@id, @me_x-1, @me_y)) { message @sd, 3, "상대방을 끌어오진 못했습니다."; end; }

		//저걸 체크했으면 이제 끌어오자!! 오류가 절대로 일어날리가 없다.
		switch(@side)
		{
		case 0:
			set_y @target_id, get_y(@id)-1;
			set_side @target_id, 2; set @side, 2;
			break;
		case 1:
			set_x @target_id, get_x(@id)+1;
			set_side @target_id, 3; set @side, 3;
			break;
		case 2:
			set_y @target_id, get_y(@id)+1;
			set_side @target_id, 0; set @side, 0;
			break;
		case 3:
			set_x @target_id, get_x(@id)-1;
			set_side @target_id, 1; set @side, 1;
			break;
		}
		updatexy @sd;
		updatelookpc @sd, 0;
		updateallpc @sd;
	}
}
