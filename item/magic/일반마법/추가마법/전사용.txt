@돌진	{
	set @sd, mysd();
	set @id, myid();

	callfunc "MANA_DELAY", 180, 0, @sd;

	set @me_x, get_x(@id);
	set @me_y, get_y(@id);

	switch(get_side(@id))
	{
	case 0:
		set @move, 4;
		set @range, 4;
		if(canmove(@id, get_x(@id), get_y(@id)-4) == 1) { set @move, 3; set @range, 3; }
		if(canmove(@id, get_x(@id), get_y(@id)-3) == 1) { set @move, 2; set @range, 2; }
		if(canmove(@id, get_x(@id), get_y(@id)-2) == 1) { set @move, 1; set @range, 1; }
		if(canmove(@id, get_x(@id), get_y(@id)-1) == 1) { set @move, 0; set @range, 0; }
		break;
	case 1:
		set @move, 4;
		set @range, 4;
		if(canmove(@id, get_x(@id)+4, get_y(@id)) == 1) { set @move, 3; set @range, 3; }
		if(canmove(@id, get_x(@id)+3, get_y(@id)) == 1) { set @move, 2; set @range, 2; }
		if(canmove(@id, get_x(@id)+2, get_y(@id)) == 1) { set @move, 1; set @range, 1; }
		if(canmove(@id, get_x(@id)+1, get_y(@id)) == 1) { set @move, 0; set @range, 0; }
		break;
	case 2:
		set @move, 4;
		set @range, 4;
		if(canmove(@id, get_x(@id), get_y(@id)+4) == 1) { set @move, 3; set @range, 3; }
		if(canmove(@id, get_x(@id), get_y(@id)+3) == 1) { set @move, 2; set @range, 2; }
		if(canmove(@id, get_x(@id), get_y(@id)+2) == 1) { set @move, 1; set @range, 1; }
		if(canmove(@id, get_x(@id), get_y(@id)+1) == 1) { set @move, 0; set @range, 0; }
		break;
	case 3:
		set @move, 4;
		set @range, 4;
		if(canmove(@id, get_x(@id)-4, get_y(@id)) == 1) { set @move, 3; set @range, 3; }
		if(canmove(@id, get_x(@id)-3, get_y(@id)) == 1) { set @move, 2; set @range, 2; }
		if(canmove(@id, get_x(@id)-2, get_y(@id)) == 1) { set @move, 1; set @range, 1; }
		if(canmove(@id, get_x(@id)-1, get_y(@id)) == 1) { set @move, 0; set @range, 0; }
		break;
	}

	if(@range <= 0)
	{
		message @sd, 3, "돌진 실패!";
		spelldelay @sd, name2spellid("돌진"), 20;
		spellaether @sd, name2spellid("돌진"), 20;
		end;
	}

	switch(@range)
	{
	case 1:
		setarray @range_x[0], @me_x-1, @me_x+1, @me_x  , @me_x;
		setarray @range_y[0], @me_y  , @me_y  , @me_y-1, @me_y+1;
		set @damage, get_hp(@sd)*3;
		set @loop_count, 4;
		break;
	case 2:
		setarray @range_x[0], @me_x-2, @me_x-1, @me_x+1, @me_x+2, @me_x  , @me_x  , @me_x  , @me_x;
		setarray @range_y[0], @me_y  , @me_y  , @me_y  , @me_y  , @me_y-2, @me_y-1, @me_y+1, @me_y+2;
		set @damage, get_hp(@sd)*3;
		set @loop_count, 8;
		break;
	case 3:
		setarray @range_x[0], @me_x-3, @me_x-2, @me_x-1, @me_x+1, @me_x+2, @me_x+3, @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x;
		setarray @range_y[0], @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y-3, @me_y-2, @me_y-1, @me_y+1, @me_y+2, @me_y+3;
		set @damage, get_hp(@sd)*3;
		set @loop_count, 12;
		break;
	case 4:
		setarray @range_x[0], @me_x-4, @me_x-3, @me_x-2, @me_x-1, @me_x+1, @me_x+2, @me_x+3, @me_x+4, @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x  , @me_x;
		setarray @range_y[0], @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y  , @me_y-4, @me_y-3, @me_y-2, @me_y-1, @me_y+1, @me_y+2, @me_y+3, @me_y+4;
		set @damage, get_hp(@sd)*3;
		set @loop_count, 16;
		break;
	}

	if(get_side(@id) == 0) { set_y @id, get_y(@id)-@move; }
	else if(get_side(@id) == 1) { set_x @id, get_x(@id)+@move; }
	else if(get_side(@id) == 2) { set_y @id, get_y(@id)+@move; }
	else { set_x @id, get_x(@id)-@move; }

	if($damageup) { set @damage, @damage*2; }
	if($finalattack) { set @damage, @damage*2; }


	for(set @loop, 0; @loop < @loop_count; set @loop, @loop+1)
	{
		set @target_id, obj_getxy(@range_x[@loop], @range_y[@loop]);
		if(@target_id > 0 && @target_id != @id)
		{
			if(istype(@target_id) == 1)
			{
				set @target_sd, id2sd(@target_id);
				message @target_sd, 3, get_name(@sd) + "님이 돌진을 가합니다.";
				effect @target_id, 193;
				damage @id, @target_id, @damage, 0;
			}
			else if(istype(@target_id) == 2)
			{
				effect @target_id, 193;
				damage @id, @target_id, @damage, 0;
			}
		}
	}
	updatexy @sd;
	updatelookpc @sd, 0;
	updateallpc @sd;

	set_hp @sd, get_hp(@sd)/2;
	set_mp @sd, get_mp(@sd)-180;

	message @sd, 3, "돌진을 외웠습니다.";
	action @id, 1, 30, 0;
	sound @id, 8;
	spelldelay @sd, name2spellid("돌진"), 30;
	spellaether @sd, name2spellid("돌진"), 30;

	deletearray @range_x[0], getarraysize(@range_x);
	deletearray @range_y[0], getarraysize(@range_y);
}

				{
					effect @target_id, 88;

		}

@살신반착	{
	set @sd, mysd();
	set @id, myid();
	set @target_id, script_target(@sd);
	set @mapid, get_mapid(@sd);
	
	if(@id == @target_id) { message @sd, 3, "자신에게는 사용할 수 없습니다."; end; }
	if(istype(@target_id) != 1) { message @sd, 3, "걸리지 않습니다."; end;} // 몬스터는 불가능
	if(!ismapoption(@sd, 6) && !ismapoption(@sd, 5)) // 사냥터나 pk지역만 허용
	{
		message @sd, 3, "사용할 수 없습니다.";
		end;
	}
	if(party(@sd) != 0)
	{
		for(set @i, 0; @i < partymax(); set @i, @i+1)
		{
			set @target_sd, party_member(@sd, @i);
			if(id2sd(@target_id) == @target_sd)
			{
				set @check, 1;
			}

		}
		if(@check > 0)
		{
			set @x, get_x(@id);
			set @y, get_y(@id);
			set @target_x, get_x(@target_id);
			set @target_y, get_y(@target_id);

			warp @sd, @mapid, @target_x, @target_y;
			warp id2sd(@target_id), @mapid, @x, @y;

			set @id, obj_getxy(@x,@y); // 이동 후 좌표값으로 오브젝트를 새로 불러옴
			set @target_id, obj_getxy(@target_x,@target_y);

			sound @id, 36;
			effect @id, 3;
			sound @target_id, 36;
			effect @target_id, 3;
			
			message @sd, 3, "살신반착을 외웠습니다.";
			end;
		}
		else { message @sd, 3, "그룹원이 아닙니다."; end;}
	}
	else
	{
		message @sd, 3, "가입된 그룹이 없습니다.";
		end;
	}
}

				}
			}

			for(set @loop, @x1; @loop <= @x2; set @loop, @loop+1)
			{
				set @target_id, obj_getxy(@loop, @me_y-1);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}

				set @target_id, obj_getxy(@loop, @me_y+1);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}
			}

			set_x @id, @me_x+@range;

			updatexy @sd;
			updatelookpc @sd, 0;
			updateallpc @sd;
		}
		else
		{
			goto failed;
			end;
		}

		break;
	case 2:
		for(set @range, 6; @range > 0; set @range, @range-1)
		{
			set @target_id, obj_getxy(@me_x, @me_y+@range);
			if(@target_id <= 0)
			{
				if(!canmove(@id, @me_x, @me_y+@range))
				{
					break;
				}
			}
		}

		if(@range > 0)
		{
			set @x1, @me_x-1;
			set @x2, @me_x+1;
			set @y1, @me_y+1;
			set @y2, @me_y+(@range-1);

			if(@x1 < 0) { set @x1, 0; }
			if(@x2 > get_mapwidth(@sd)-1) { set @x2, get_mapwidth(@sd)-1; }
			if(@y1 < 0) { set @y1, 0; }
			if(@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd)-1; }

			for(set @i, @x1; @i <= @x2; set @i, @i+1)
			{
				for(set @j, @y1; @j <= @y2; set @j, @j+1)
				{
					set @target_id, obj_getxy(@i, @j);
					if(@target_id > 0 && @target_id != @id)
					{
						if(istype(@target_id) == 1)
						{
							set @target_sd, id2sd(@target_id);
							if(ismapoption(@target_sd, 6) && get_state(@target_sd) != 1)
							{
								message @target_sd, 3, get_name(@sd) + "님이 그림자의맹습을 가합니다.";
							}

							effect @target_id, 20;
							effect @target_id, 33;
							//sound @target_id, 96;

							callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
						}
						else if(istype(@target_id) == 2)
						{
							effect @target_id, 20;
							effect @target_id, 33;
							//sound @target_id, 96;

							callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
						}
					}
				}
			}

			for(set @loop, @y1; @loop <= @y2; set @loop, @loop+1)
			{
				set @target_id, obj_getxy(@me_x-1, @loop);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}

				set @target_id, obj_getxy(@me_x+1, @loop);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}
			}

			set_y @id, @me_y+@range;

			updatexy @sd;
			updatelookpc @sd, 0;
			updateallpc @sd;
		}
		else
		{
			goto failed;
			end;
		}

		break;
	case 3:
		for(set @range, 6; @range > 0; set @range, @range-1)
		{
			set @target_id, obj_getxy(@me_x-@range, @me_y);
			if(@target_id <= 0)
			{
				if(!canmove(@id, @me_x-@range, @me_y))
				{
					break;
				}
			}
		}

		if(@range > 0)
		{
			set @x1, @me_x-(@range-1);
			set @x2, @me_x-1;
			set @y1, @me_y-1;
			set @y2, @me_y+1;

			if(@x1 < 0) { set @x1, 0; }
			if(@x2 > get_mapwidth(@sd)-1) { set @x2, get_mapwidth(@sd)-1; }
			if(@y1 < 0) { set @y1, 0; }
			if(@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd)-1; }

			for(set @i, @x1; @i <= @x2; set @i, @i+1)
			{
				for(set @j, @y1; @j <= @y2; set @j, @j+1)
				{
					set @target_id, obj_getxy(@i, @j);
					if(@target_id > 0 && @target_id != @id)
					{
						if(istype(@target_id) == 1)
						{
							set @target_sd, id2sd(@target_id);
							if(ismapoption(@target_sd, 6) && get_state(@target_sd) != 1)
							{
								message @target_sd, 3, get_name(@sd) + "님이 그림자의맹습을 가합니다.";
							}

							effect @target_id, 20;
							effect @target_id, 33;
							//sound @target_id, 96;

							callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
						}
						else if(istype(@target_id) == 2)
						{
							effect @target_id, 20;
							effect @target_id, 33;
							//sound @target_id, 96;

							callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
						}
					}
				}
			}

			for(set @loop, @x1; @loop <= @x2; set @loop, @loop+1)
			{
				set @target_id, obj_getxy(@loop, @me_y-1);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}

				set @target_id, obj_getxy(@loop, @me_y+1);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 2)
					{
						set_mobtarget @target_id, 0;
						mobspellaether @target_id, name2spellid("몹타겟삭제"), 3;
					}
				}
			}

			set_x @id, @me_x-@range;

			updatexy @sd;
			updatelookpc @sd, 0;
			updateallpc @sd;
		}
		else
		{
			goto failed;
			end;
		}

		break;
	}

	if(#Siege_Mapname$ != "")
	{
		if(get_mapname(@sd) == #Siege_Mapname$ + "의심장")
		{
			if(get_x(@id) == 20 && get_y(@id) == 10)
				runscript @sd, "@신물헌신";
		}
	}

	action @id, 1, 30, 0;
	sound @id, 96;

	set_hp @sd, get_hp(@sd)/2;

	spelldelay @sd, name2spellid("그림자의맹습"), 135;

	message @sd, 3, "그림자의맹습을 시전하였습니다.";
	end;
failed:
	set_hp @sd, get_hp(@sd)/2;

	spelldelay @sd, name2spellid("그림자의맹습"), 135;
	message @sd, 3, "그림자의맹습 실패!";
}
