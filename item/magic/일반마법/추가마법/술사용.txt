@수폭지술	{
	set @sd, mysd();
	set @id, myid();
	set @damage, get_mp(@sd)*9/2;
	if($damageup) { set @damage, @damage*9/2; }
	
	callfunc "MANA_DELAY", 600, 0, @sd;
	
	set @x1, get_x(@id)-8;
	set @x2, get_x(@id)+8;
	set @y1, get_y(@id)-7;
	set @y2, get_y(@id)+7;
	
	if (@x1 < 0) { set @x1, 0; }
	if (@x2 > get_mapwidth(@sd) -1) { set @x2, get_mapwidth(@sd) -1; }
	if (@y1 < 0) { set @y1, 0; }
	if (@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd) -1; }
	
	for (set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for (set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target, obj_getxy(@i, @j);
			if (@target > 0 && @target != @id)
			{
				if (istype(@target) == 1)
				{
					set @target_sd, id2sd(@target);
					damage @id, @target, @damage, 0;
					effect @target, 182;
					if(ismapoption(@sd, 6)) message @target_sd, 3, get_name(@sd) + "님께서 수폭지술을 가합니다.";
				}
				else if (istype(@target) == 2)
				{
					effect @target, 182;
					damage @id, @target, @damage, 0;
				}
			}
		}
	}
	action @id, 6, 30, 0;
	sound @id, 107;
	set_mp @sd, 0;
	spellaether @sd, name2spellid("수폭지술"), 240;
	spelldelay @sd, name2spellid("수폭지술"), 240;

@시폭염화지속	{

	set @id, myid();
	set @me_x, $bomb_x;
	set @me_y, $bomb_y;
	set @me_side, $bomb_side;

	if(spellaethered(@sd, name2spellid("시폭염화")) == 0) end;

	if(spellaetherdelay(@sd, name2spellid("시폭염화")) == 1)
	{
		spelluncast @sd, name2spellid("시폭염화");

		if(get_mapid(@sd) != $bomb_mapid) end;

		set @value, callfunc("calculate_damage", $bomb_damage, 3);
		set @damage, callfunc("calc_multiply", @value);
		set @save_damage, @damage;

		switch(@me_side)
		{
		case 0:
			setarray @x[0], @me_x-1, @me_x  , @me_x+1, @me_x-1, @me_x  , @me_x+1, @me_x-1, @me_x  , @me_x+1;
			setarray @y[0], @me_y+2, @me_y+2, @me_y+2, @me_y+3, @me_y+3, @me_y+3, @me_y+4, @me_y+4, @me_y+4;
			break;
		case 1:
			setarray @x[0], @me_x-4, @me_x-3, @me_x-2, @me_x-4, @me_x-3, @me_x-2, @me_x-4, @me_x-3, @me_x-2;
			setarray @y[0], @me_y-1, @me_y-1, @me_y-1, @me_y  , @me_y  , @me_y  , @me_y+1, @me_y+1, @me_y+1;
			break;
		case 2:
			setarray @x[0], @me_x-1, @me_x  , @me_x+1, @me_x-1, @me_x  , @me_x+1, @me_x-1, @me_x  , @me_x+1;
			setarray @y[0], @me_y-2, @me_y-2, @me_y-2, @me_y-3, @me_y-3, @me_y-3, @me_y-4, @me_y-4, @me_y-4;
			break;
		case 3:
			setarray @x[0], @me_x+4, @me_x+3, @me_x+2, @me_x+4, @me_x+3, @me_x+2, @me_x+4, @me_x+3, @me_x+2;
			setarray @y[0], @me_y-1, @me_y-1, @me_y-1, @me_y  , @me_y  , @me_y  , @me_y+1, @me_y+1, @me_y+1;
			break;
		}

		for(set @loop, 0; @loop < 9; set @loop, @loop+1)
		{
			if(@x[@loop] >= 0 && @x[@loop] < get_mapwidth(@sd) && @y[@loop] >= 0 && @y[@loop] < get_mapheight(@sd))
			{
				set @target_id, obj_getxy(@x[@loop], @y[@loop]);
				if(@target_id > 0 && @target_id != @id)
				{
					if(istype(@target_id) == 1)
					{
						set @target_sd, id2sd(@target_id);
						if(ismapoption(@target_sd, 6) && get_state(@target_sd) != 1)
						{
							message @sd, 3, get_name(@sd) + "님의 폭탄에 맞았습니다.";
						}

						effect @target_id, 19;
						sound @target_id, 700;

						callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
					}
					else if(istype(@target_id) == 2)
					{
						effect @target_id, 19;
						sound @target_id, 700;

						callfunc "ATTACKDAMAGE", @target_id,  @damage, 0;
					}
				}
			}
		}

		deletearray @x[0], getarraysize(@x);
		deletearray @y[0], getarraysize(@y);
	}
}

		end;
	}

	set @user_x, get_x(@id);
	set @user_y, get_y(@id);
	
	set @target_x, get_x(@target_id);
	set @target_y, get_y(@target_id);

	set @gap_x, @user_x - @target_x; // 내 위치와 거리 계산
	set @gap_y, @user_y - @target_y;

	if(@gap_x < 0){ set @gap_x, @gap_x * (-1); } // 거리 차가 음수일경우 양수로 변경
	if(@gap_y < 0){ set @gap_y, @gap_y * (-1); }

	if(@gap_x > 8 || @gap_y > 8)
	{
		message @sd, 3, "마력이 미치지 않습니다.";
		end;
	}
	/* 거리 조절 끝*/

	set @x, get_x(@target_id);
	set @y, get_y(@target_id);

	setarray @target_x[0], @x+1, @x  , @x-1, @x;
	setarray @target_y[0], @y  , @y+1, @y  , @y-1;
	
	for(set @loop, 0; @loop < getarraysize(@target_x); set @loop, @loop+1) // 주변에 몹이 있는지 체크

		set @target_mid, obj_getxy(@target_x[@loop], @target_y[@loop]);

			if(istype(@target_mid) == 2) { set @count, @count + 1; } // 주변 "몹" 체크용
		}
	}
	if(@count == 0){ set @time, 1; } // 주변에 몹 없으면 0초
	
	if(istype(@target_id) == 1)
	{
		message @sd, 3, "걸리지 않습니다.";
		end;
	}
	else if(istype(@target_id) == 2)
	{
		sound @target_id, 59;

	}

	deletearray @target_x[0], getarraysize(@target_x);
	deletearray @target_y[0], getarraysize(@target_y);

}

			set @target_mid, obj_getxy(@target_x[@loop], @target_y[@loop]);

					}
				}
