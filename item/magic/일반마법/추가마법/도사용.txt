@자동기원1	{
	set @sd, mysd();
	set @id, myid();
	set_spellinterval @sd, get_spellinterval(@sd)-200;
    
	set @pk, ismapoption(@sd, 6);
	if(@pk)
	{
		message @sd, 3, "사용할수없는곳입니다.";
		end;
	}
	if(get_state(@sd) == 3 || get_state(@sd) == 1)
	{
		message @sd, 3, "사용이 불가능합니다.";
		end;
	}
	if(ismapoption(@sd,48))
	{
		message @sd, 3, "공성맵에서는 사용이 불가능합니다.";
		end;
	}
	if($heal_name$ == "백호의희원")
	{
		message @sd, 3, "백호의희원은 자동기원을 사용할 수 없습니다.";
		end;
	}

	set @ts, name2sd($heal_target_name$);
	if(@ts > 0)
	{
		set @x1, get_x(@id);
		set @y1, get_y(@id);
		set @x2, get_x(name2id($heal_target_name$));
		set @y2, get_y(name2id($heal_target_name$));

		if(@x1 > @x2) set @cha, @x1-@x2;
		else set @cha, @x2-@x1;
		if(@y1 > @y2) set @cha2, @y1-@y2;
		else set @cha2, @y2-@y1;

		if(@cha < 7 && @cha2 < 7)
		{
			if(@id != name2id($heal_target_name$))
			{	
				callfunc "TARGETHEALMAGIC", $mp_delay, $hp_target, $heal_effect, $heal_sound, $heal_name$, 1;
				end;
			}
		}
	}
	else
	{
		message @sd, 3, "타겟이범위내에없습니다.";
		end;
	}
}

@전체부활	{
	set @sd, mysd();
	set @id, myid();

	callfunc "NOREVIVAL";

	if(get_mp(@sd) < 600)
	{
		message @sd, 3, "마력이 부족합니다.";
		end;
	}

	set @x1, get_x(@id) - 8;
	set @x2, get_x(@id) + 8;
	set @y1, get_y(@id) - 8;
	set @y2, get_y(@id) + 8;

	if(@x1 < 0) { set @x1, 0; }
	if(@x2 > get_mapwidth(@sd) - 1) { set @x2, get_mapwidth(@sd) - 1; }
	if(@y1 < 0) { set @y1, 0; }
	if(@y2 > get_mapheight(@sd)- 1) { set @y2, get_mapheight(@sd) - 1; }

	for(set @i, @x1; @i <= @x2; set @i, @i + 1)
	{
		for(set @j, @y1; @j <= @y2; set @j, @j + 1)
		{
			set @target, obj_getxy(@i, @j);

			if(@target > 0)
			{
				if(istype(@target) == 2)
				{
					continue;
				}
				else if(istype(@target) == 1)
				{
					set @temp, 1;

					set @target_sd, id2sd(@target);

					if(get_state(@target_sd) == 1)
					{
						effect @target, 11;

						set_state @target_sd, 0;
					}

					if(get_hp(@target_sd) < 30)
					{
						set_hp @target_sd, 30;
					}

					if(get_equipitem(@target_sd, 6) == name2itemid("보무의목걸이") || get_equipitem(@target_sd, 7) == name2itemid("보무의목걸이"))
					{
						spellcast @target_sd, name2spellid("자동보호");
						spellcast @target_sd, name2spellid("자동무장");
					}

					if(@target != @id)
					{
						message @target_sd, 3, get_name(@sd) + "님이 부활을 외워주셨습니다.";
					}

					updatelookpc @target_sd, 1;
				}
			}
		}
	}

	if(@temp)
	{
		action @id, 6, 30, 0;
		sound @id, 8;

		set_mp @sd, get_mp(@sd) - 600;
		message @sd, 3, "전체부활을 외웠습니다.";
		spelldelay @sd, name2spellid("전체부활"), 40;
	}
}
