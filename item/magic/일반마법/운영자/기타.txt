@순보	{
	set @sd, mysd();
	set @id, myid();
	if(get_mp(@sd) < 5)
	{
		message @sd, 3, "마력이 부족합니다.";
		end;
	}
	switch(get_side(@id))
	{
	case 0:
		set @move, 3;
		if(canmove(@id, get_x(@id), get_y(@id)-3) == 1) { set @move, 2; }
		if(canmove(@id, get_x(@id), get_y(@id)-2) == 1) { set @move, 1; }
		if(canmove(@id, get_x(@id), get_y(@id)-1) == 1) { set @move, 0; }
		set_y @id, get_y(@id)-@move;
		break;
	case 1:
		set @move, 3;
		if(canmove(@id, get_x(@id)+3, get_y(@id)) == 1) { set @move, 2; }
		if(canmove(@id, get_x(@id)+2, get_y(@id)) == 1) { set @move, 1; }
		if(canmove(@id, get_x(@id)+1, get_y(@id)) == 1) { set @move, 0; }
		set_x @id, get_x(@id)+@move;
		break;
	case 2:
		set @move, 3;
		if(canmove(@id, get_x(@id), get_y(@id)+3) == 1) { set @move, 2; }
		if(canmove(@id, get_x(@id), get_y(@id)+2) == 1) { set @move, 1; }
		if(canmove(@id, get_x(@id), get_y(@id)+1) == 1) { set @move, 0; }
		set_y @id, get_y(@id)+@move;
		break;
	case 3:
		set @move, 3;
		if(canmove(@id, get_x(@id)-3, get_y(@id)) == 1) { set @move, 2; }
		if(canmove(@id, get_x(@id)-2, get_y(@id)) == 1) { set @move, 1; }
		if(canmove(@id, get_x(@id)-1, get_y(@id)) == 1) { set @move, 0; }
		set_x @id, get_x(@id)-@move;
		break;
	}
	updatexy @sd;
	updatelookpc @sd, 0;
	updateallpc @sd;
	action @id, 1, 30, 0;
	effect @id, 15;
	sound @id, 8;
	set_mp @sd, get_mp(@sd)-5;
	message @sd, 3, "순보를 외웠습니다.";
	if(get_level(@sd) >= 1 && get_level(@sd) < 5) { set @delay, 8; }
	else if(get_level(@sd) >= 5 && get_level(@sd) < 10) { set @delay, 7; }
	else if(get_level(@sd) >= 10 && get_level(@sd) < 20) { set @delay, 6; }
	else if(get_level(@sd) >= 20 && get_level(@sd) < 30) { set @delay, 5; }
	else if(get_level(@sd) >= 30 && get_level(@sd) < 40) { set @delay, 4; }
	else if(get_level(@sd) >= 40 && get_level(@sd) < 50) { set @delay, 3; }
	else if(get_level(@sd) >= 50 && get_level(@sd) < 60) { set @delay, 3; }
	else if(get_level(@sd) >= 60 && get_level(@sd) < 70) { set @delay, 3; }
	else if(get_level(@sd) >= 70 && get_level(@sd) < 80) { set @delay, 3; }
	else if(get_level(@sd) >= 80 && get_level(@sd) < 90) { set @delay, 3; }
	else if(get_level(@sd) >= 90 && get_level(@sd) < 99) { set @delay, 3; }
	else if(get_level(@sd) == 99) { set @delay, 2; }
	else { set @delay, 2; }
	spelldelay @sd, name2spellid("순보"), @delay;
}

	{
		switch(get_side(@id))
		{
		case 0:
			set_y @id, get_y(@id)-1;
			break;
		case 1:
			set_x @id, get_x(@id)+1;
			break;
		case 2:
			set_y @id, get_y(@id)+1;
			break;
		case 3:
			set_x @id, get_x(@id)-1;
			break;
		}
	}

	updatexy @sd;
	updatelookpc @sd, 0;
	updateallpc @sd;

	message @sd, 3, "강제이동을 외웠습니다.";
}
