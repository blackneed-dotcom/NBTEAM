@노획	{
	set @sd, mysd();
	set @id, myid();
	set @x, get_x(@id);
	set @y, get_y(@id);

	if(get_mapid(@sd) == 2190)
	{
		message @sd, 3, "고래갑판에서는 사용할 수 없습니다.";
		end;
	}

	callfunc "MANA_DELAY", 30, 5, @sd;
	
	switch( get_side(@id) )
	{
	case 0:
		set @y, @y-1;
		set @val, canmove(@id, get_x(@id), get_y(@id)-1);
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2 || get_mobname(@target_id) == "말") { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 1:
		set @x, @x+1;
		set @val, canmove(@id, get_x(@id)+1, get_y(@id));
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2 || get_mobname(@target_id) == "말") { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 2:
		set @y, @y+1;
		set @val, canmove(@id, get_x(@id), get_y(@id)+1);
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2 || get_mobname(@target_id) == "말") { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 3:
		set @x, @x-1;
		set @val, canmove(@id, get_x(@id)-1, get_y(@id));
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2 || get_mobname(@target_id) == "말") { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	}
	getitem @sd, @x, @y;
NOTGET:
	callfunc "BEGINFUNC", 0, 37, 0, 0, 6, 0;
	say @id, 2, "노획!~~";
	//set_mp @sd, get_mp(@sd)-30;	
}

@노획	{
	set @sd, mysd();
	set @id, myid();
	set @x, get_x(@id);
	set @y, get_y(@id);
	
	callfunc "MANA_DELAY", 30, 0, @sd;
	
	switch( get_side(@id) )
	{
	case 0:
		set @y, @y-1;
		set @val, canmove(@id, get_x(@id), get_y(@id)-1);
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2) { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 1:
		set @x, @x+1;
		set @val, canmove(@id, get_x(@id)+1, get_y(@id));
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2) { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 2:
		set @y, @y+1;
		set @val, canmove(@id, get_x(@id), get_y(@id)+1);
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2) { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	case 3:
		set @x, @x-1;
		set @val, canmove(@id, get_x(@id)-1, get_y(@id));
		if(@val)
		{
			set @target_id, obj_getfront();
			if(@target_id > 0) { if(istype(@target_id) != 2) { goto NOTGET; } }
			else { goto NOTGET; }
		}
		break;
	}
	getitem @sd, @x, @y;
NOTGET:
	callfunc "BEGINFUNC", 0, 37, 0, 0, 6, 0;
	say @id, 2, "노획!~~";
	set_mp @sd, get_mp(@sd)-30;	
}
