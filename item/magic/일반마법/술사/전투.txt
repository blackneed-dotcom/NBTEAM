@화염주	{
	set @sd, mysd();
	set @id, myid();
	
	set @user_x, get_x(@id);
	set @user_y, get_y(@id);
	
	set @target_id, script_target(@sd); // 타겟
	set @target_x, get_x(@target_id);
	set @target_y, get_y(@target_id);

	set @gap_x, @user_x - @target_x; // 내 위치와 거리 계산
	set @gap_y, @user_y - @target_y;

	if(@gap_x < 0){ set @gap_x, @gap_x * (-1); } // 거리 차가 음수일경우 양수로 변경
	if(@gap_y < 0){ set @gap_y, @gap_y * (-1); }

	if(@gap_x > 7 || @gap_y > 7)
	{
		message @sd, 3, "마력이 미치지 않습니다."+@gap_x+","+@gap_y;
		end;
	}
	callfunc "MAGICIANSPELL", myid(), mysd(), 40, 320+get_grace(mysd()), 45, "화염주", 120;
}

@지폭지술	{
	set @sd, mysd();
	set @id, myid();
	set @damage, get_mp(@sd)*6/2;
	if($damageup) { set @damage, @damage*6; }
	
	callfunc "MANA_DELAY", 600, 0, @sd;
	
	set @x1, get_x(@id)-8;
	set @x2, get_x(@id)+8;
	set @y1, get_y(@id)-7;
	set @y2, get_y(@id)+7;
	
	if (@x1 < 0) { set @x1, 0; }
	if (@x2 > get_mapwidth(@sd) -1) { set @x2, get_mapwidth(@sd) -1; }
	if (@y1 < 0) { set @y1, 0; }
	if (@y2 > get_mapheight(@sd)-1) { set @y2, get_mapheight(@sd) -1; }
	
	for (set @i, @x1; @i <= @x2; set @i, @i+1)
	{
		for (set @j, @y1; @j<=@y2; set @j, @j+1)
		{
			set @target, obj_getxy(@i, @j);
			if (@target > 0 && @target != @id)
			{
				if (istype(@target) == 1)
				{
					set @target_sd, id2sd(@target);
					damage @id, @target, @damage, 0;
					effect @target, 104;
					if(ismapoption(@sd, 6)) message @target_sd, 3, get_name(@sd) + "님께서 지폭지술을 가합니다.";
				}
				else if (istype(@target) == 2)
				{
					effect @target, 104;
					damage @id, @target, @damage, 0;
				}
			}
		}
	}
	action @id, 6, 30, 0;
	sound @id, 107;
	set_mp @sd, 0;
	if(get_equipitem_name(@sd, 0) == "현성곤봉'무") { set @delay, 195; }
	else if(get_equipitem_name(@sd, 0) == "진혼단봉'이") { set @delay, 185; }
	else { set @delay, 210; }
	spellaether @sd, name2spellid("지폭지술"), @delay;
	spelldelay @sd, name2spellid("지폭지술"), @delay;
	message @sd, 3, "지폭지술을 외웠습니다.";
}
