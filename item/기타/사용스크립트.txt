@캐쉬관리비령	{
	set @sd, mysd();
	set @id, myid();

	if(get_gmlevel(@sd) < 1)
	{
		set @pos, dialog(0,0,1, "운영자가 아니면 사용할수 없습니다.");
		end;
	}
 
	set @pos, dialog(0,0,1, "※ 주의 ※\\n반드시 유저를 관리할때 유저의 장비템을 모두 벗었을때 사용해주세요. 주의사항을 지키지않고 유저를 관리시 불이익은 책임지지 않습니다.");
	if(@pos == 0) end;

	set @user, input1(@username$, "관리할 유저의 이름을 입력해주세요.");
	set @target_id, name2id(@username$);
	set @target_sd, name2sd(@username$);
	if(@target_id > 0)
	{
		set @pos, list2(@list, 0, 0, "기능을 선택해주세요.", "캐쉬 관리");
		if(@pos == 0) end;
		else if(@list == 1) goto cash;
cash:
		runscript @target_sd, "유저캐쉬";
		set @user_cash, #user_cash_p;
		set #user_cash_p, 0;
		set @pos, list2(@select_re, 0, 1, get_name(@target_sd)+"님의 캐쉬는 "+@user_cash+" 입니다. 변경하시겠습니까?", "예", "아니오");
		if(!@pos) end;
		if(@select_re == 2) { end; }
		else if(@select_re == 1) {
			set @pos, input1(@c_cash, "변경할 캐쉬금액를 써주세요.");
			set #change_cash, @c_cash;
			runscript @target_sd, "유저캐쉬변경";
			message @sd, 8, get_name(@target_sd)+" 님의 캐쉬를"+@c_cash+"(으)로 변경하였습니다.";
			end;
		}
	}
	else { message @sd, 8, @username$ + "님은 현재 접속중이지 않습니다."; end; }
}
